//===== rAthena Script =======================================
//= Shadow Blacksmith
//===== Description: =========================================
//= [Official Conversion]
//= Shadow equipments refining NPC.
//===== Changelog: ===========================================
//= 1.0 First version [Aleos]
//= 1.1 Removed re-roll behavior and fetch materials from db
//=     [Secret]
//= 1.2 Added Refine UI [Atemo, Lemongrass]
//= 1.3 Moved to Cash Mall [Lemongrass]
//============================================================
-	script	::ShadowBlacksmith	-1,{
	if( getbattleflag( "feature.refineui" ) ){
		mes "[Ferreiro das Sombras]";
		mes "Agora o Basta é responsável pelo refinamento de itens das sombras.";
		mes "Se você quiser refinar um item das sombras, visite-o.";
		close;
	}

	.@zeny_cost = 200000; // O custo do Zeny é de 200.000 de acordo com o script oficial [Segredo]
	disable_items;
	mes "[Ferreiro das Sombras]";
	mes "Você quer refinar um item das sombras? Escolha seu veneno!";
	next;
	setarray .@indices[1], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;
	.@indlen = getarraysize(.@indices) - 1;
	for(.@i = 1; .@i <= .@indlen; .@i++)
		.@menu$ = .@menu$ + (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Não equipado]") +":";
	.@menu$ = .@menu$ + "Informações Sobre o Refino";
	.@choice = select(.@menu$);
	if (.@choice == .@indlen + 1) { // Informações do Refino
		mes "[Ferreiro das Sombras]";
		mes "O item das sombras irá ganhar bônus extra pelo nível de refinamento, o mesmo que itens como arma/armadura normais.";
		next;
		mes "[Ferreiro das Sombras]";
		mes "A seguir estão os efeitos de refinamento para cada equipamento das Sombras.";
		mes "Arma: ATQ, ATQM + 1 de aumento para cada +1 refinamento com sucesso.";
		mes "Outros: HP + 10 de aumento para cada +1 refinamento com sucesso.";
		next;
		mes "[Ferreiro das Sombras]";
		mes "Você precisa de Oridecon e Elunium para refinar os itens das sombras, junto com uma taxa de refino de 200,000 zenys.";
		next;
		mes "[Ferreiro das Sombras]";
		mes "Quando os itens das Sombras forem refinados até +5 ou superior poderão ser destruídos, por favor, use Oridecon ou Elunium Enriquecido ou PERFEITO para o refinamento.";
		close;
	}

	.@part = .@indices[.@choice];

	if (!getequipisequiped(.@part)) {
		mes "[Ferreiro das Sombras]";
		mes "Não tem nada aqui!";
		close;
	}
	.@equip_id = getequipid(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	while(1) {
		.@equip_refine = getequiprefinerycnt(.@part);
		mes "[Ferreiro das Sombras]";
		mes "Eu preciso de " + callfunc("F_InsertComma", .@zeny_cost) + " zenys como uma taxa para CADA tentativa de refinamento.";
		mes "Escolha seu minério e comece a refinar.";
		next;

		.@material[0] = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
		.@material[1] = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
		.@material[2] = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);
		.@is_enriched = false;

		if (countitem(.@material[0]))
			.@mate$[0] = getitemname(.@material[0]);
		else
			.@mate$[0] = "^8C8C8C"+ getitemname(.@material[0]) +"^000000";
		if (countitem(.@material[1]))
			.@mate$[1] = getitemname(.@material[1]);
		else
			.@mate$[1] = "^8C8C8C"+ getitemname(.@material[1]) +"^000000";
		if (.@equip_refine > 6 && countitem(.@material[2]))
			.@mate$[2] = getitemname(.@material[2]);
		else
			.@mate$[2] = "^8C8C8C"+ getitemname(.@material[2]) +"^000000";

		switch( select( .@mate$[0], .@mate$[1], .@mate$[2], "Cancelar" ) ) {
		case 1:// NORMAL
			.@choose = .@material[0];
			break;
		case 2:// ENRIQUECIDO
			.@is_enriched = true;
			.@choose = .@material[1];
			break;
		case 3:// PERFEITO
			if (.@equip_refine < 7) {
				mes "[Ferreiro das Sombras]";
				mes "Minério PERFEITO só pode ser usado para itens com refinamento +7 ou superior.";
				close;
			}
			.@hoihoi = 1;
			.@choose = .@material[2];
			break;
		case 4:
			mes "[Ferreiro das Sombras]";
			mes "Você cancelou o refinamento.";
			close;
		}
		if (!countitem(.@choose)) {
			mes "[Ferreiro das Sombras]";
			mes "Você não tem os ingrediente adequados para este refinamento.";
			close;
		}
		if (Zeny < .@zeny_cost) {
			mes "[Ferreiro das Sombras]";
			mes "Você não tem zenys suficientes.";
			close;
		}
		if (.@equip_refine > 9) {
			mes "[Ferreiro das Sombras]";
			mes "O refinamento de itens das sombras só será possível até o nível +10...";
			close;
		}
		if (!getequipisenableref(.@part)) {
			mes "[Ferreiro das Sombras]";
			mes "Este item não pode ser refinado.";
			close;
		}
		if (getequippercentrefinery(.@part) < 100) {
			mes "[Ferreiro das Sombras]";
			mes "O limite de refinamento com segurança garantida para itens das sombras é de até +4.";
			if (!.@hoihoi)
				mes "Se você tentar mais refinamentos, o item poderá ser quebrado quando falhar. Ainda quer refinar?";
			else
				mes "Se você tentar mais refinamentos, o nível de refinamento do item poderá diminuir quando falhar. Ainda deseja refinar?";
			next;
			if (select("Proceed","Cancel") == 2) {
				mes "[Ferreiro das Sombras]";
				mes "Heh, eu sabia!";
				close;
			}
		}
		mes "[Ferreiro das Sombras]";
		mes "Pois bem... lá vai nada!";
		next;
		delitem .@choose,1;
		Zeny -= .@zeny_cost;

		// anti-roubo
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
			close;

		if (getequippercentrefinery(.@part, .@is_enriched) > rand(100)) {
			successrefitem .@part;
			mes "[Ferreiro das Sombras]";
			mes "Funcionou! Funcionou!";
			next;
		} else {
			if (.@hoihoi)
				downrefitem .@part;
			else
				failedrefitem .@part;
			mes "[Ferreiro das Sombras]";
			mes "Oh Odin, não... O refineamento falhou.";
			close;
		}
	}
}
