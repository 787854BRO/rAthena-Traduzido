-	script	GuildWarehouse	-1,{
	mes "[Gerente do Armazém]";
	mes "Como vai você? Somos especializados em Armazéns de ClãS. Esta é a nossa ^0000cdHistória^000000:";
	mes "Por que os membros do Clã não podem compartilhar o mesmo Armazém?";
	next;
	mes "[Gerente do Armazém]";
	mes "Você poderá abrir o armazém por 1.000 Zeny ou consultar o histórico de uso pelo Clã.";
	mes "Como posso ajudá-lo?";
	next;
	if( select( "Abrir Armazém do Clã.:Ver histórico de uso do Armazém." ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Gerente do Armazém]";
			mes "O Armazém do Clã está disponível apenas para membros de um Clã.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Gerente do Armazém]";
			mes "Parece que ainda não é possível para seu Clã usar o armazém ou você não tem as permissões de acesso a ele.";
			mes "Por favor, volte depois de verificar se o Clã tem a habilidade necessária e a permissão para acessar o armazém.";
			close;
		}

		if( Zeny < 1000 ){
			mes "[Gerente do Armazém]";
			mes "Receio que você não tenha dinheiro suficiente para pagar a taxa.";
			mes "A taxa é de 1.000 Zenys.";
			close;
		}

		mes "[Gerente do Armazém]";
		mes "Vou abrir o Armazém do Clã para você então!";
		close2;

		if( Zeny < 1000 ){
			// Previne trapaças
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[Gerente do Armazém]";
			mes "Receio que outra pessoa esteja usando o armazém nesse momento.";
			mes "Por favor, tente novamente depois de um tempo.";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Gerente do Armazém]";
			mes "O Armazém do Clã está disponível apenas para membros de um Clã.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Gerente do Armazém]";
			mes "Parece que ainda não é possível para seu Clã usar o Armazém ou você não tem as permissões de acesso a ele.";
			mes "Por favor, volte depois de verificar se o Clã tem a habilidade necessária e a permissão para acessar o armazém.";
			close;
		}

		mes "[Gerente do Armazém]";
		mes "Vou lhe mostrar o histórico de uso do Armazém.";
		mes "Tenha um momento memorável!";
		close2;
		guildopenstorage_log();
		end;
	}
}
