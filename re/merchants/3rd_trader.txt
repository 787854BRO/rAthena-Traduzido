job3_guil01,91,93,3	script	Negociante de Ervas Raras	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "- Espere um minuto! -";
		mes "- Atualmente você está carregando -";
		mes "- muitos itens com você. -";
		mes "- Por favor, volte novamente -";
		mes "- após deixar alguns itens no Armazém. -";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "[Negociante de Ervas Raras]";
		mes "Eu ando ao redor do mundo e recolho ervas venenosas raras. Recentemente, comecei a lidar com uma erva chamada Izidor. Se você estiver interessado, você pode comprá-la.";
		next;
		.@i = select("Como posso comprá-la?:Trocá-la por um Sangue de Animal:Trocá-la por uma Erva Amarga:Trocá-la por uma Erva Letal:Trocá-la por uma Rosa de Gelo:Trocá-la por uma Menta:Trocá-la por uma Hinalle")-2;
		if (.@i == -1) {
			mes "[Negociante de Ervas Raras]";
			mes "Você pode comprá-la por 10,000 zeny e um desses itens raros: Sangue de Animal, Erva Amarga, Erva Letal, Rosa de Gelo, Menta ou Hinalle.";
			next;
			mes "[Negociante de Ervas Raras]";
			mes "Por que eu peço dinheiro? Isso é apenas para pagar um pouco de respeito para um colecionador como eu. Ha ha ha...";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,605,703;
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "[Negociante de Ervas Raras]";
			mes "Quantas você precisa?";
			mes "Você pode comprar no máximo '2000'.";
			mes "Digite '0' se desejar cancelar.";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "[Negociante de Ervas Raras]";
				mes "Você cancelou a negociação.";
				close;
			}
			if (.@amount > 2000) {
				mes "[Negociante de Ervas Raras]";
				mes "O máximo que posso negociar é 2,000!";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "[Negociante de Ervas Raras]";
				mes "Opa, acho que você não tem todos os requerimentos necessários para a negociação.";
				close;
			}
			mes "[Negociante de Ervas Raras]";
			mes "Excelente. Recebi o dinheiro e o item especial.";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "[Negociante de Ervas Raras]";
	mes "Eu ando ao redor do mundo e recolho ervas venenosas raras. Mas eu não posso vender minhas ervas para uma pessoa como você... Ha ha ha...";
	close;
}
