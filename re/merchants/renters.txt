				mes "[^FF0000 Criador de Dragões ^000000]";
				mes "Você já possui uma montaria.";
				close;
			}
			else if (ismounting()) {
				mes "[^FF0000 Criador de Dragões ^000000]";
				mes "Você já possui uma montaria.";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[^FF0000 Criador de Dragões ^000000]";
			mes "Entendo. Então tenha um ótimo dia.";
			close;
		}
	}
	mes "O que você está fazendo aqui?";
	mes "Somente Cavaleiros Rúnicos podem alugar Dragões.";
	close;
}

			close;
		}

// Mestre da Engrenagem Mágica :: M.E.C.H.A
//============================================================
-	script	::mgm	-1,{
	mes "[^FF0000 Mestre da Engrenagem ^000000]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "Bem-vind"+(Sex?"o":"a")+", mecânico.";
		mes "Você gostaria de alugar um Carrinho ou";
		mes "montar em um M.E.C.H.A?";
		next;
		switch(select("Alugar um Carrinho:Alugar um M.E.C.H.A:Cancelar!")) {
		case 1:
			if (checkcart()) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Sinto muito, mas você já";
				mes "possui um carrinho.";
				close;
			}
			setcart;
			mes "[^FF0000 Mestre da Engrenagem ^000000]";
			mes "Ai está!";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Sinto muito, mas você já está";
				mes "montado em um M.E.C.H.A.";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Por favor, aprenda a habilidade Licença de Pilotagem primeiro.";
				close;
			}
			else if(ismounting()) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Por favor, remova a sua montaria atual primeiro.";
				close;
			}
			setmadogear;
			mes "[^FF0000 Mestre da Engrenagem ^000000]";
			mes "Divirta-se, e por favor, volte sempre que precisar!";
			close;
		case 3:
			mes "[^FF0000 Mestre da Engrenagem ^000000]";
			mes "O Robô de Emergência é realmente útil para situações de emergência e é vendido por 1.000.000 de Zenys.";
			next;
			if (select("Comprar:Cancelar") == 2) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Entendo. Sinta-se à vontade para falar comigo";
				mes "se mudar de ideia.";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Sinto muito, mas você já tem um Robô de Emergência.";
				close;
			}
			if (Zeny < 1000000) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Sinto muito, mas você não tem Zeny suficiente para comprar o Robô de Emergência.";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Robô de Emergência
			mes "[^FF0000 Mestre da Engrenagem ^000000]";
			mes "Aí está!";
			close;
		case 4:
			mes "[^FF0000 Mestre da Engrenagem ^000000]";
			mes "Qual dispositivo você quer atualizar?";
			next;
			if (select("Termoregulador:Resfriador Avançado") == 1) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Atualizar o Termoregulador para um Resfriador Avançado requer 1 Termoregulador e 2.000.000 de Zenys.";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Atualizar um Resfriador Avançado para um Resfriador requer 1 Resfriador Avançado e 4.000.000 de Zenys.";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("Atualizar:Cancelar") == 2) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Entendo. Sinta-se à vontade para falar comigo";
				mes "se mudar de ideia.";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Sinto muito, mas você não tem o " + getitemname(.@itemid) + ".";
				close;
			}
			if (Zeny < .@cost) {
				mes "[^FF0000 Mestre da Engrenagem ^000000]";
				mes "Desculpe, mas você não tem Zenys o suficiente para atualizar o dispositivo.";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //Resfriador Avançado, Resfriador.
			mes "[^FF0000 Mestre da Engrenagem ^000000]";
			mes "Aqui está você! Seu próprio " + getitemname(.@itemid) + ".";
			close;
		case 5:
			close;
		}
	}
	mes "Como posso ajudar você?";
	mes "Os M.E.C.H.A.S estão disponíveis apenas para Mecânicos.";
	close;
}
