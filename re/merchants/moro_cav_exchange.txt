moro_cav,32,73,5	script	Trocar Marcas do Herói#1	4_F_SHADOWCHASER,{
		disable_items;
		mes "[Trocador de Marcas do Herói]";
		mes "Se você lutou contra Morroc";
		mes "e coletou";
		mes "Marcas do Herói,";
		mes "posso trocá-los por um anel especial para você.";
		mes "Gostaria de ver meu anel?";
		while(true) {
			next;
			if (checkweight(2163,2) == 0) {
				mes "[Trocador de Marcas do Herói]";
				mes "Sua mochila está com muitos itens,";
				mes "por favor volte depois de organizar!";
				close;
			}
			switch( select( "Descrição do Anel.", "Trocar pelo Anel.", "Encerrar conversa." ) ) {
			case 1:
				mes "[Trocador de Marcas do Herói]";
				mes "Este Anel do Herói";
				mes "pode não parecer";
				mes "impressionante,";
				next;
				mes "[Trocador de Marcas do Herói]";
				mes "mas leve-o até";
				mes "o Encantador Especialista Byrnes,";
				mes "ele pode Encantar";
				mes "os 4 slots dele";
				mes "e liberar";
				mes "suas habilidades ocultas.";
				next;
				mes "[Trocador de Marcas do Herói]";
				mes "O Anel do Herói";
				mes "pode ajudar você";
				mes "a subjugar";
				mes "Morroc.";
				mes "Mais alguma dúvida?";
				continue;
			case 2:
				mes "- Anel do Herói -";
				mes "^006400Um anel feito para guerreiros^000000";
				mes "^006400na missão de derrotar Morroc...^000000";
				mes "^006400Quando encantado,^000000";
				mes "^006400concede ao seu portador^000000";
				mes "^006400o poder para confrontar Morroc.^000000";
				mes "^006400Classe: Acessório / DEF: 0^000000";
				mes "^006400Peso: 20 / Nível necessário: 160^000000";
				mes "^006400Slot: 0 / Equipável em: Todas as classes^000000";
				next;
				mes "[Trocador de Marcas do Herói]";
				mes "O Anel do Herói requer";
				mes "30 Marcas do Herói";
				mes "para a troca.";
				mes "Deseja trocar?";
				next;
				if (select( "Menu anterior.", "Trocar 30 Marcas do Herói." ) == 1) {
					mes "[Trocador de Marcas do Herói]";
					mes "Pense";
					mes "melhor.";
					continue;
				}
				if (countitem(6684) < 30) {
					mes "[Trocador de Marcas do Herói]";
					mes "^006400Marca do Herói^000000";
					mes "é concedida a";
					mes "aqueles que lutaram bravamente contra Morroc.";
					mes "Você não possui as";
					mes "30 ^006400Marcas do Herói^000000,";
					mes "portanto não posso ajudá-lo.";
					continue;
				}
				mes "[Trocador de Marcas do Herói]";
				mes "Irei trocar";
				mes "30 ^006400Marcas do Herói^000000";
				mes "por 1 ^006400Anel do Herói^000000.";
				mes "Mais alguma coisa?";
				delitem 6684,30;	// Marca do Herói
				getitem 2981,1;		// Anel do Herói
				continue;
			case 3:
				mes "[Trocador de Marcas do Herói]";
				mes "Um dia";
				mes "você perceberá";
				mes "que precisa do Anel do Herói.";
				close;
			}
		}
	}

moro_cav,34,65,5	script	Byrnes, Encantador Esp.	1_M_03,{
		disable_items;
		mes "[Byrnes]";
		mes "Já ouviu falar do Anel do Herói? Apenas verdadeiros heróis merecem possuir esse magnífico anel, e eu estive esperando por um.";
		mes "O que posso fazer por você?";
		next;
		switch( select( "Encantar Anel do Herói.", "Redefinir Encantamentos do Anel do Herói.", "Cancelar." ) ) {
		case 1:
			mes "[Byrnes]";
			mes "Só posso encantar anéis equipados. Qual dos anéis equipados deseja encantar?";
			next;
			setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
			.@s = select( "Esquerdo.", "Direito." ) - 1;
			switch( getequipid(.@eqi_slot[.@s]) ) {
			case -1:
				mes "[Byrnes]";
				mes "Por favor, equipe o Anel do Herói que deseja encantar.";
				close;
			case 2981:
				break;
			default:
				mes "[Byrnes]";
				mes "Por favor, equipe seu Anel do Herói. Não posso encantar outra coisa.";
				close;
			}
			// personalizado neste ponto
			setarray .@card[0],
				getequipcardid(.@eqi_slot[.@s],0),
				getequipcardid(.@eqi_slot[.@s],1),
				getequipcardid(.@eqi_slot[.@s],2),
				getequipcardid(.@eqi_slot[.@s],3);

			if (.@card[0] > 0) {
				mes "[Byrnes]";
				mes "Um Anel do Herói completamente encantado não pode receber mais encantamentos.";
				close;
			}
			if (.@card[3] == 0) {
				setarray .@enchant_list$[0], "DEF", "DEFM";
				.@slot = 4;
			}
			else if (.@card[2] == 0) {
				setarray .@enchant_list$[0], "HP Máx.", "SP Máx.";
				.@slot = 3;
			}
			else if (.@card[1] == 0) {
				setarray .@enchant_list$[0], "ATQ", "ATQ Mágico";
				.@slot = 2;
			}
			else {
				setarray .@enchant_list$[0], "FOR", "INT", "AGI", "VIT", "DES", "SOR";
				.@slot = 1;
			}
			mes "[Byrnes]";
			mes "Qual habilidade deseja como encantamento no slot " + .@slot + "?";
			next;
			.@enchant_list_size = getarraysize(.@enchant_list$);
			for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
				.@menu$ += .@enchant_list$[.@i] + ":";
			.@type = select(.@menu$) - 1;
		switch(.@slot) {
		case 4:
			switch(.@type) {// lista de encantamentos
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// DEF Mágica
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// HP Máx.
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// SP Máx.
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// ATQ
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// ATQ Mágico
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// FOR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DES
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default: // SOR
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

// verificação de anti-roubo
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// taxas desconhecidas
		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "Encantamento concluído!";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[Byrnes]";
		mes "Para remover os encantamentos do seu Anel do Herói, eu preciso de 3 Marcas de Herói.";
		next;
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Você não tem Marcas de Herói suficientes. Não posso remover o encantamento do seu Anel do Herói.";
			close;
		}
		mes "[Byrnes]";
		mes "Eu só posso remover os encantamentos de anéis equipados. Qual dos anéis equipados você deseja redefinir?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "Esquerdo.", "Direito." ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[Byrnes]";
			mes "Você precisa equipar o item para redefinir seu estado de encantamento.";
			close;
		case 2981:
			break;
		default:
			mes "[Byrnes]";
			mes "Equipe seu Anel do Herói. Eu não posso redefinir outro item.";
			close;
		}
		// parte personalizada
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Você não tem Marcas de Herói suficientes. Não posso redefinir o seu Anel do Herói.";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[Byrnes]";
			mes "Seu Anel do Herói não possui encantamentos no momento.";
			close;
		}

		// verificação de anti-roubo
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "Encantamentos removidos com sucesso.";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
