//= Empacotador de Gemas
//============================================================
-	script	#EmpacotadorGemas	HIDDEN_NPC,{
	mes "[Empacotador de Gemas]";
	mes "Nós empacotamos gemas em bolsas de forma fácil para carregá-las.";
	mes "O que você gostaria de fazer?";
	next;
	switch(select("Trocar Gemas.:Cancelar.")) {
	case 1:
		mes "[Empacotador de Gemas]";
		mes "Vou preparar uma bolsa para as gemas.";
		close2;
		callshop "Gemstone_Bagger";
		break;
	case 2:
		mes "[Empacotador de Gemas]";
		mes "Volte se precisar de ajuda.";
		close;
	}
}
