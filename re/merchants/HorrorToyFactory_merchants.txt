		else if (.@r < 1931) .@enchant = 4750;	// Sorte1
		else if (.@r < 1951) .@enchant = 4751;	// Sorte2
		else if (.@r < 1971) .@enchant = 4752;	// Sorte3
		else .@enchant = 9;
		break;
	default:
		break;
	}
	if (.@enchant == 9) {
		mes "Há um problema, por favor volte novamente!";// personalizado
		close;
	}
	if (.@card[3] == 0 && .@sc_type < 4)
		.@index_slot = 3;
	else if (.@card[2] == 0 && .@sc_type < 3)
		.@index_slot = 2;
	else if (.@card[1] == 0 && .@sc_type < 2)
		.@index_slot = 1;
	else {
		mes "Eu não sei o que é isso, mas acho que não é algo que eu possa lidar.";// personalizado
		close;
	}
	if (countitem(7642) > 14) {// Moeda Sangrenta
		delitem 7642,15;// Moeda Sangrenta

		// anti-roubo
		if (callfunc("F_IsEquipIDHack", .@eq_num, .@item_id) || callfunc("F_IsEquipCardHack", .@eq_num, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@eq_num, .@refine))
			close;

		delequip .@eq_num;
		if (.@enchant == 0) {
			mes "Ai! Que pena, o encantamento falhou e seu equipamento quebrou. Que azar!";
			close2;
			specialeffect2 EF_LORD;
			end;
		}
		.@card[ .@index_slot ] = .@enchant;
		getitem2 .@item_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
		mes "Hmm... Foi um sucesso. Dê uma olhada nisso.";
		specialeffect2 EF_REPAIRWEAPON;
		close;
	}
	mes "Hmm... Aliás, você precisa juntar mais Moedas Sangrentas. Você nem consegue pagar a taxa.";
	close;
}
