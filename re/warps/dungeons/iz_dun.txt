iz_dun04,130,234,5	script	Guardião da Caverna#iz_dun	413,{
	mes "[Guardião da Caverna]";
	mes "Pare!";
	mes "O que te traz aqui, aventureiro?";
	mes "Descobrimos um novo andar desta cidade subaquática.";
	next;
	mes "[Guardião da Caverna]";
	mes "Você acha que está à altura do desafio?";
	mes "Os monstros aqui estão entre os níveis 110 e 124.";
	mes "E você deve tomar cuidado com o Kraken!";
	next;
	switch(select("Soltem o Kraken!:Ficarei aqui.")) {
	case 1:
		mes "[Guardião da Caverna]";
		mes "Seja cauteloso e boa sorte.";
		mes "Se você vir o Kraken, certifique-se de estar com seus amigos para lha ajudar ou faça o que fazemos e corra.";
		close2;
		warp "iz_dun05",141,187;
		end;
	case 2:
		mes "[Guardião da Caverna]";
		mes "É sempre melhor estar preparado antes de explorar uma nova área.";
		mes "Volte quando estiver pronto para ir.";
		close;
	}
}
