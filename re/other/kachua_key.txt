//===== rAthena Script =======================================
//= Dynamic NPC: Kachua's Secret Box
//===== Description: =========================================
//- [Official conversion]
//= Kachua's Secret Box is a Gachapon NPC.
//= It lets the player exchange a Kachua's Secret Key for a random item.
//= It also gives a Kachua's Mileage Coupon for each pull.
//===== Changelogs: ==========================================
//= 1.0 First version. [secretdataz]
//============================================================

sec_in02,126,178,3	script	Caixa Secreta da Kachua#bm	4_TREASURE_BOX,{
	.@key$ = "K_Secret_Key";
	.@keyname$ = getitemname( .@key$ );
	.@box$ = "Main_Lucky_Box";
	.@boxname$ = getitemname( .@box$ );
	mes "Uma Caixa Secreta cujo conteúdo é um mistério para todos.";
	mes "^4d4dffVocê pode abrir esta caixa consumindo uma <ITEM>Chave da Kachua<INFO>.@key$</INFO></ITEM>.^000000";
	next;
	switch(select("^4d4dffAbrir caixa 1 vez (1 " + .@keyname$ + ")^000000","^4d4dffAbrir caixa 10 vezes (10 " + .@keyname$ + ")^000000")) {
		case 1:
			mes "^FF0000[Aviso]^000000";
			mes "^FF0000Troque um item aleatório por um(a) " + .@keyname$ + ".^000000";
			mes "^FF0000O item trocado não poderá ser recuperado, nem convertido de volta em " + .@keyname$ + ".^000000";
			next;
			if(select("Continuar.","Encerrar conversa.") == 2) {
				mes "Você decidiu não abrir a Caixa.";
				close;
			} else if (countitem(.@key$) < 1) {
				mes "Você não possui <ITEM>Chave da Kachua<INFO>.@key$</INFO></ITEM> o suficiente.";
				close;
			} else {
				if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
					mes "^4d4dffCertifique-se de que há espaço suficiente no seu inventário.^000000";
					close;
				}
				delitem(.@key$, 1);
				consumeitem(.@box$);
				mes .@boxname$ + " foi aberta!";
				mes "Será que você teve sorte?";
				specialeffect2 EF_VALLENTINE;
				close;
			}
		case 2:
			mes "^FF0000[Aviso]^000000";
			mes "^FF0000Troque 10 itens aleatórios por 10 " + .@keyname$ + ".^000000";
			mes "^FF0000Os itens trocados não poderão ser recuperados, nem convertidos de volta em " + .@keyname$ + ".^000000";
			next;
			if(select("Continuar.","Encerrar conversa.") == 2) {
				mes "Você decidiu não abrir a Caixa.";
				close;
			} else if (countitem(.@key$) < 10) {
				mes "Você não possui <ITEM>Chave da Kachua<INFO>.@key$</INFO></ITEM> o suficiente.";
				close;
			} else {
				for (.@i = 1; .@i <= 10; ++.@i) {
					progressbar "4d4dff",2;
					if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
						mes "^4d4dffCertifique-se de que há espaço suficiente no seu inventário.^000000";
						close;
					}
					if (countitem(.@key$) < 1) { // Verificação adicional, por precaução
						close;
					}
					delitem(.@key$, 1);
					consumeitem(.@box$);
					dispbottom .@boxname$ + " foi aberta " + .@i + " vez(es). Outra está sendo aberta.",0xFFFFFF;
					specialeffect2 EF_VALLENTINE;
				}
				mes .@boxname$ + " foi aberta 10 vezes!";
				mes "Será que você teve sorte?";
				close;
			}
	}
}
