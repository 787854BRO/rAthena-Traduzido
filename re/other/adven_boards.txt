function	script	F_AvenBoard	{
	.@mo_board = select( "Procurar monstros por andar.",
					"Sistema de Informações de Monstros.",
					( countitem(909) == 30000 ? "Solicitar" : "" ),
					"Cancelar" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[Sistema de Informações Sobre Monstros]";
			mes "Este é o Sistema de Informações Sobre Monstros. É possível pesquisar informações dos monstros que aparecem nesta masmorra. Ao pesquisar, digite o Nome do Monstro, incluindo espaços.";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutro","Água","Terra","Fogo","Vento","Veneno","Sagrado","Sombrio","Fantasma","Morto-Vivo";
			setarray .@size$[0],"Médio","Pequeno","Grande";
			setarray .@race$[0],"Amorfo","Morto-Vivo","Animal","Planta","Inseto","Peixe","Demônio","Humanoide","Anjo","Dragão";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[Sistema de Informações Sobre Monstros]";
			mes "Pesquisando...";
			next;
			mes "[Sistema de Informações Sobre Monstros]";
			mes "A pesquisa foi concluída.";
			mes "Exibindo os resultados.";
			next;
			mes "[Sistema de Informações Sobre Monstros]";
			if (.@i >= .@size) {
				mes "Não há informações disponíveis para esse monstro.";
				mes "Por favor, verifique o nome do monstro.";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- Nome : "+ getarg(.@i+1);
				mes "- Elemento : "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- Tamanho : "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- Raça : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[Sistema de Informações Sobre Monstros]";
			mes "Deseja procurar informações de outro monstro?";
			next;
			if (select( "Pesquisar.","Cancelar." ) == 1)
				continue;
			mes "[Sistema de Informações Sobre Monstros]";
			mes "O sistema de busca de monstros será encerrado agora.";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "A informação solicitada aparecerá - Não tem problema se aparecer";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "Erro de operação.";
		close;
	}
}

function	script	F_InfoBoard	{
	next;
	mes "[Grupo de Aventureiros]";
	mes "Para mais informações sobre os monstros,";
	mes "o Sistema de Informações Sobre Monstros";
	mes "deve ser utilizado.";
	next;
	return;
}
