ecl_in02,164,56,3	script	Recepcionista#Laphine	835,{
	mes "[Recepcionista}";
	mes "Bem-vind"+(Sex?"o":"a")+"(a) à Clínica de Eclage.";
	next;
	switch (select("Salvar Localização:Restauração:Terminar a conversa")) {
		case 1:
			mes "[Recepcionista}";
			mes "A clínica foi definida como o seu ponto de salvamento. Tenha cuidado~";
			savepoint "ecl_in02",162,50;
			close;
		case 2:
			mes "[Recepcionista}";
			mes "Restaure sua mente e corpo em nossa clínica. Ela está cheia da energia limpa de Eclage.";
			mes "Nós cobramos 1 Moeda Esplendor normalmente, mas nós oferecemos um desconto especial para os viajantes. Seu total é 5.000 zeny.";
			next;
			switch (select("Pague com Moeda Esplendor:Pague com Zeny:Esqueça")) {
				case 1:
					if (!countitem(6081)) {
						mes "[Recepcionista}";
						mes "Parece que você não tem moedas suficientes. Você verificaria sua conta novamente, por favor?";
						close;
					}
					mes "[Recepcionista}";
					mes "Sinta-se em casa~";
					close2;
					delitem 6081,1; //Moeda Esplendor
					percentheal 100,100;
					warp "ecl_in02",167,49;
					end;
					
				case 2:
					if (Zeny < 5000) {
						mes "[Recepcionista}";
						mes "Parece que você não tem recursos suficientes para prosseguir. Você se importa de verificar sua conta novamente?";
						close;
					}
					mes "[Recepcionista}";
					mes "Sinta-se em casa~";
					close2;
					Zeny = Zeny - 5000;
					percentheal 100,100;
					warp "ecl_in02",167,49;
					end;
				case 3:
					close;
			}
		case 3:
			close;
	}
}
