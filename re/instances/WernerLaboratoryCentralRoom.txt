1@slw,155,191,3	script	Dispositivo de Segurança#switch01	CLEAR_NPC,{
	if (is_party_leader() == false)
		end;
	.@number = atoi( replacestr( strnpcinfo(2), "switch", "" ) );
	.@type = (.@number < 10) ? 0 : 1;
	.@npc_num = (.@number < 10) ? .@number : (.@number-10);
	if ('device_state[.@type] & .@npc_num) {
		mes "Já está ativado.";
		close;
	}
	callsub S_Code, .@string$, .@answer;
	mes "Há códigos de segurança anotados.";
	mes "[" + .@string$ + "]";
	mes "O que você vai inserir?";
	next;
	if (select( "^ff0000" + .@string$ + "^000000", "^4d4dff" + .@string$ + "^000000" ) != .@answer) {
		'device_state[.@type] = 0;
		mes "O dispositivo foi inicializado.";
		close;
	}
	if (is_party_leader() == false)
		end;
	'device_state[.@type] = 'device_state[.@type] | .@npc_num;
	if ('device_state[.@type] != 3) {
		mes "O Dispositivo de Segurança foi operado.";
		mes "Parece que há mais um Dispositivo de Segurança.";
		close;
	}
	mes "O Dispositivo de Segurança foi operado.";
	mes "O portão foi ativado.";
	switch( .@type ) {
	case 0:
		enablenpc instance_npcname("#slwwarp01");
		if (instance_live_info(ILI_NAME) == "Laboratório Werner#1")	// terra_gloria_main == 21
			enablenpc instance_npcname("Eisen Werner#werner03");
		else {
			enablenpc instance_npcname("Guarda#gard05");
			enablenpc instance_npcname("Guarda#gard06");
			enablenpc instance_npcname("Guarda#gard07");
		}
		break;
	case 1:
		if (instance_live_info(ILI_NAME) == "Laboratório Werner#1")
			enablenpc instance_npcname("Eisen Werner#werner04");
		else
			enablenpc instance_npcname("Entrada Central#door");
		enablenpc instance_npcname("#slwwarp02");
		break;
	}
	close;

S_Code:
	setarray .@sample$[0],
		"Pureza", 2,
		"Eternidade", 2,
		"Amanhecer", 2,
		"Rosa", 1,
		"Contradição", 1,
		"Alegria", 1,
		"Mar", 1,
		"Retorno", 1,
		"Solidão", 1,
		"Brilho", 1,
		"Crepúsculo", 1;
	.@size = getarraysize(.@sample$) / 2;
	.@r = rand(.@size) * 2;
	set getarg(0), .@sample$[.@r];
	set getarg(1), .@sample$[.@r+1];
	return;
}
