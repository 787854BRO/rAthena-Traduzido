wolfvill,79,260,4	script(CLOAKED)	Aira#con	4_M_BRZ_JACI,{
	if (ep18_1_main < 57)
		end;
	switch( checkquest(12522,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[Aira]";
		mes "Por favor, aguarde o fim do tempo de espera da missão.";
		close;
	case 2:
		erasequest 12522;
		mes "^0000ffTodos os efeitos colaterais do acesso desapareceram.^000000";
		close;
	}
	mes "[Aira]";
	mes "" + strcharinfo(0) + "Graças ao Sr. consegui chegar em segurança na vila.";
	mes "Mas parece que algo foi revivido na vila. Acho que vou precisar de ajuda mais uma vez.";
	next;
	.@md_name_normal$ = "Vila da Decepção";
	.@md_name_hard$ = "Vila da Decepção - Modo Avançado";

	if (is_party_leader() == true) {
		.@create_normal$ = "^000044Criar " + .@md_name_normal$ + "^000000";
		if (countitem(1000471) > 0)
			.@create_hard$ = "^000044Criar " + .@md_name_hard$ + "^000000";
	}
	switch( select( "Entrar na", .@create_normal$, .@create_hard$ ) ) {
	case 1:
		if (select( .@md_name_normal$, .@md_name_hard$ ) == 1) {
			if (BaseLevel < 170) {
				mes "[Aira]";
				mes "Você precisa no mínimo estar no Nível de Base 170 para entrar.";
				close;
			}
			.@enter_to$ = .@md_name_normal$;
		}
		else {
			if (BaseLevel < 200) {
				mes "[Aira]";
				mes "Você precisa estar no mínimo no Nível de Base 200 para entrar.";
				close;
			}
			.@enter_to$ = .@md_name_hard$;
		}

		switch( instance_enter(.@enter_to$) ) {
		case IE_OTHER:
			mes "^ff0000Ocorreu um erro desconhecido.^000000";
			close;
		case IE_NOINSTANCE:
			mes "^0000ff A instância não foi criada ou você está tentando entrar em uma instância diferente da do líder do grupo.^000000";
			close;
		case IE_NOMEMBER:
			mes "^ff0000Apenas membros do grupo podem entrar na Instância Memorial.^000000";
			close;
		case IE_OK:
			mapannounce "wolfvill", "" + strcharinfo(0) + " do grupo " + getpartyname( getcharid(1) ) + " está entrando na " + .@enter_to$ + ".",bc_map,"0x00ff99",FW_NORMAL,12;
			if (.@enter_to$ == .@md_name_normal$)
				setquest 12520;
			else
				setquest 12521;
			setquest 12522;
			// warp "1@advs",111,19;
			end;
		}
		end;
	case 2:
		if (BaseLevel < 170) {
			mes "[Aira]";
			mes "Você precisa estar no mínimo no Nível de Base 170 para criar a instância.";
			close;
		}
		if (instance_create(.@md_name_normal$) < 0) {
			mes "Grupo: " + getpartyname(getcharid(1)) + "";
			mes "Líder: " + strcharinfo(0) + "";
			mes "^0000ff" + .@md_name_normal$ + " ^000000- Erro desconhecido";
			close;
		}
		end;
	case 3:
		if (BaseLevel < 200) {
			mes "[Aira]";
			mes "Você precisa estar no mínimo no Nível de Base 200 para criar a instância.";
			close;
		}
		if (countitem(1000471) < 1) {
			mes "[Aira]";
			mes "Onde está o item " + getitemname(1000471) + "?";
			close;
		}
		if (instance_create(.@md_name_hard$) < 0) {
			mes "Grupo: " + getpartyname(getcharid(1)) + "";
			mes "Líder: " + strcharinfo(0) + "";
			mes "^0000ff" + .@md_name_hard$ + " ^000000- Erro desconhecido";
			close;
		}
		delitem 1000471,1;
		end;
	}
	end;
}




1@advs,1,1,0	script	advs_init	-1,{
	end;
OnInstanceInit:
	//--------------------------------------------
	// inicialização de variáveis
	//--------------------------------------------

	'advs_step = 0;
	'map_advs$ = instance_mapname("1@advs");

	// variáveis da etapa 1
	'sw_timer = 0;
	deletearray 'sw[0], 2;

	// variáveis da etapa 3
	deletearray 'x_reaper[0], 'size_reaper;
	deletearray 'y_reaper[0], 'size_reaper;
	'size_reaper = 0;

	//--------------------------------------------
	// Configurações opcionais
	//--------------------------------------------

	// Número mínimo de ceifadores para aparecer na etapa 3
	'min_reaper = 3;

	// Configuração de acordo com o modo
	if (instance_live_info(ILI_NAME) == "Vila da Decepção - Modo Avançado") {
		'advs_difficult = 1;
		'sanare_id = 21377;	// EP18_MD_SANARE_L
		'hunter_id = 21378;	// EP18_MD_HEARTHUNTER_L
		'monster_relieve = 10;	// precisa conferir, mas os danos são muito baixos comparados ao modo normal

		// Total de monstros para matar nas etapas 1, 2 e 3
		'total_monster_step_1 = 80;
		'total_monster_step_2 = 70;
		'total_monster_step_3 = 80;

		// Número de monstros que aparecem no início das etapas 1, 2 e 3
		'init_monster_step_1 = 40;
		'init_monster_step_2 = 30;
		'init_monster_step_3 = 40;

		'schulang_id = 21360;	// EP18_MD_SCHULANG_L
		'freyja_id = 21361;	// EP18_MD_DEMI_FREYJA_L
	
		// Não mexa aqui a menos que saiba o que está fazendo
		.@interval = 5;	// distância entre cada ceifador
		.@max_reaper = 48;	// número máximo de ceifadores que podem aparecer
	}
	else {
		'advs_difficult = 0;
		'sanare_id = 21318;	// EP18_MD_SANARE_R
		'hunter_id = 21319;	// EP18_MD_HEARTHUNTER_R2
		'monster_relieve = 9;

		// Total de monstros para matar nas etapas 1, 2 e 3
		'total_monster_step_1 = 35;
		'total_monster_step_2 = 35;
		'total_monster_step_3 = 35;

		// Número de monstros que aparecem no início das etapas 1, 2 e 3
		'init_monster_step_1 = 35;
		'init_monster_step_2 = 25;
		'init_monster_step_3 = 24;

		'schulang_id = 21316;	// EP18_MD_SCHULANG_R
		'freyja_id = 21317;	// EP18_MD_DEMI_FREYJA_R

		// Não mexa aqui a menos que saiba o que está fazendo
		.@interval = 10;
		.@max_reaper = 16;
	}

	for ( .@i = 109; .@i <= 139; .@i += .@interval ) {
		for ( .@j = 219; .@j <= 249; .@j += .@interval ) {
			if (.@i == 124 && .@j == 234)
				continue;
			'x_reaper[ 'size_reaper ] = .@i;
			'y_reaper[ 'size_reaper ] = .@j;
			'size_reaper++;
		}
	}
	'total_reaper = min(.@max_reaper, 'size_reaper);

	// quantidade total de comida necessária para reduzir o número de ceifadores para 'min_reaper
	'max_food = ('total_reaper * 2) - ('min_reaper * 2);

	// debug
	'total_monster_step_1 = max('init_monster_step_1, 'total_monster_step_1);
	'total_monster_step_2 = max('init_monster_step_2, 'total_monster_step_2);
	'total_monster_step_3 = max('init_monster_step_3, 'total_monster_step_3);

	//--------------------------------------------
	// gerenciamento de npcs
	//--------------------------------------------
	disablenpc instance_npcname("#boss2");
	disablenpc instance_npcname("#box_admin");

	// Portais
	disablenpc instance_npcname("#wp1");
	disablenpc instance_npcname("#wp2");
	disablenpc instance_npcname("#wp3");
	disablenpc instance_npcname("#wp4");

	disablenpc instance_npcname("#food1");
	disablenpc instance_npcname("#food2");
	disablenpc instance_npcname("#food3");
	disablenpc instance_npcname("#food4");

	// círculos da etapa final
	disablenpc instance_npcname("#b_tr1");
	disablenpc instance_npcname("#b_tr2");
	disablenpc instance_npcname("#b_tr3");
	disablenpc instance_npcname("#b_tr4");

	// parar os círculos de se moverem
	disablenpc instance_npcname("#b_sw1");
	disablenpc instance_npcname("#b_sw2");
	disablenpc instance_npcname("#b_sw3");
	disablenpc instance_npcname("#b_sw4");

	disablenpc instance_npcname("#boss1");
	disablenpc instance_npcname("#boss2_sw");

	donpcevent instance_npcname("#advs_step_1") + "::OnStart";
	end;
}

1@advs,1,1,0	script	#advs_step_1	-1,{
	end;
OnStart:
	'mob_to_respawn = 'total_monster_step_1 - 'init_monster_step_1;

	// As coordenadas podem estar imprecisas (talvez não seja aleatório no oficial)
	for ( .@i = 0; .@i < 'init_monster_step_1; ++.@i )
		callsub( S_Spawn );
	end;
	
OnMobDead:
	if ('mob_to_respawn > 0)
		'mob_to_respawn--;
	initnpctimer;
	sleep 2000;
	if ('advs_step == 0 && 'mob_to_respawn > 0) {
		if ('mob_to_respawn > 0)
			callsub( S_Spawn );
		if ('mob_to_respawn > 1)
			callsub( S_Spawn );
	}
	end;
OnTimer2000:
	stopnpctimer;
	.@alive = mobcount('map_advs$, instance_npcname("#advs_step_1") + "::OnMobDead");
	mapannounce 'map_advs$, "Número de monstros restantes: " + .@alive, bc_map, 0xAAAAFF;
	end;
S_Spawn:
	.@label$ = instance_npcname("#advs_step_1") + "::OnMobDead";
	if (rand(2))
		.@mob_id = 'hunter_id;
	else
		.@mob_id = 'sanare_id;
	.@r = rand(3);
	if (.@r == 0)
		areamonster 'map_advs$,63,89,72,182,"--en--",.@mob_id,1, .@label$;
	else if (.@r == 1)
		areamonster 'map_advs$,63,89,184,101,"--en--",.@mob_id,1, .@label$;
	else
		areamonster 'map_advs$,184,89,174,182,"--en--",.@mob_id,1, .@label$;
	unitskilluseid $@mobid[0], 771, 'monster_relieve;	// NPC_RELIEVE_ON
	return;
}
