moro_cav,50,64,5	script	Semente Amarela#bios	CLEAR_NPC,{
	if (BaseLevel < 160) {
		mes "^ff0000Personagens abaixo do Nível 160 não podem entrar neste Calabouço Memorial.^000000";
		close;
	}
	if (isbegin_quest(15006) == 0) {
		mes "^ff0000Você deve primeiro encontrar o Velho Errante para ter acesso a este calabouço.^000000";
		close;
	}
	.@party_id = getcharid(1);
	if (.@party_id < 1) {
		mes "^ff0000Você deve primeiro formar/entrar em um grupo de pelo menos 1 pessoa para entrar neste calabouço.^000000";
		close;
	}
	switch( checkquest(15005,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		if (checkquest(15007,PLAYTIME) == 2) {// 5 minutos para ré-entrar
			mes "^ff0000Este calabouço de instância não poderá ser acessado por 23 horas após sua última visita.^000000";
			close;
		}
		switch( checkquest(15008,PLAYTIME) ) {
		case -1:
			break;
		case 0:
		case 1:
			mes "^0000ffEste calabouço de instância não pode ser acessado por 23 horas após sua última visita.^000000";
			close;
		case 2:
			break;
		}
		break;
	case 2:
		erasequest 15005;// Calabouço Memorial: Ilha Bios
		erasequest 15007;// Calabouço Memorial: Ilha Bios
		if (isbegin_quest(15008) > 0)
			erasequest 15008;// Calabouço Memorial: Ilha Bios
		break;
	}
	if (is_party_leader() == true)
		.@create$ = "Criar Calabouço de Instância.";
	switch( select( .@create$, "Entrar na Ilha Bios.", "Cancelar." ) ) {
	case 1:
		instance_create("Ilha Bios");
		mes "^ff0000Calabouço de Instância criado.^000000";
		mes "^ff0000Por favor, clique e pressione Enter novamente.^000000";
		close;
	case 2:
		switch( instance_enter("Ilha Bios") ) {
		case IE_NOMEMBER:
		case IE_NOINSTANCE:
		case IE_OTHER:
			if (is_party_leader(.@party_id) == false) {
				mes "^ff0000Atenção: Este Calabouço de Instância não poderá ser acessado por 23 horas após sua última visita.^000000";
				next;
			}
			mes "^ff0000O líder do seu grupo ainda não criou uma Fenda do Tempo.^000000";
			close;
		case IE_OK:
			mapannounce "moro_cav", "" + strcharinfo(0) + " do grupo, " + getpartyname( getcharid(1) ) + ", está entrando na Ilha Bios.",bc_map,"0x00FF99";
			if (isbegin_quest(15005) == 0) {
				setquest 15005;// Calabouço de Instância: Ilha Bios
				setquest 15007;// Calabouço de Instância: Ilha Bios
			}
			end;
		}
	case 3:
		end;
	}
	end;

OnInit:
	while(true) {
		sleep 500;
		specialeffect EF_ENHANCE;
	}
	end;
}
