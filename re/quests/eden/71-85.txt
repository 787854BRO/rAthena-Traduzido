moc_para01,44,38,3	script	Missões [71 - 85]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Apenas membros do -";
		mes "- Grupo do Éden podem ler -";
		mes "- este Quadro de Avisos. -";
		close;
	}

	// Limpa os dados originais da missão (não mais utilizados).
	if (slv_quest) set slv_quest,0;
	if (checkquest(10102) > -1 || checkquest(10103) > -1 || checkquest(10104) > -1 || checkquest(10105) > -1 || checkquest(10106) > -1) {
		for(set .@quest,10102; .@quest<=10106; set .@quest,.@quest+1) {
			if (checkquest(.@quest) > -1)
				erasequest .@quest;
		}
		mes "- Desde que o cliente se mudou -";
		mes "- para outro lugar, -";
		mes "- o formulário de candidatura foi retirado. -";
		next;
	}

	if (BaseLevel < 71) {
		mes "- Apenas membros com o -";
		mes "- nível adequado podem ler -";
		mes "- este Quadro de Avisos. -";
		mes "- Tente encontrar um Quadro -";
		mes "- que se adeque ao seu nível. -";
		close;
	} else if (BaseLevel > 85) {
		// IDs daa Missões: 10107-10123, 5055-5057
		setarray .@quests[0],
			10107,10108,10109,
			10110,10111,10112,10113,
			10114,10115,10116,10117,
			10118,10119,10120,
			10121,10122,10123,5055,5056;
		for(set .@i,0; .@i<getarraysize(.@quests); set .@i,.@i+1) {
			if (checkquest(.@quests[.@i],HUNTING) == 2)
				set .@complete[getarraysize(.@complete)], .@quests[.@i];
		}
		if (checkquest(5057) > -1 && countitem(7187) >= 30)
			set .@complete[getarraysize(.@complete)],5057;
		if (getarraysize(.@complete)) {
			mes "Você tem missões em curso.";
			mes "Quer entregá-las agora?";
			next;
			if(select("Claro.:Não.") == 2)
				close;
			for(set .@i,0; .@i<getarraysize(.@complete); set .@i,.@i+1)
				callsub L_Quest,.@complete[.@i];
			close;
		}
		mes "- Apenas membros com o -";
		mes "- nível adequado podem ler -";
		mes "- este Quadro de Avisos. -";
		mes "- Tente encontrar um Quadro -";
		mes "- que se adeque ao seu nível. -";
		close;
	}
	mes "- Escolha onde você -";
	mes "- gostaria de caçar. -";
	next;
	switch(select("Calabouço da Esfinge:Glast Heim:Área de Juno:Torre do Relógio:Ilhas Localizadas")) {
	case 1:
		mes "- Há vários pedidos -";
		mes "- relacionados com o Calabouçu da Esfinge -";
		mes "- neste Quadro de Missões. -";
		next;
		set .@str1$,"Os aldeões de Morroc estão preocupados com o número crescente de monstros no Calabouço da Esfinge. ";
		set .@str2$,"- Meidi de Morroc -";
		switch(select("Caça Réquiem:Caçar Marduk:Caçar Pasana")) {
			case 1: callsub L_Quest,10107,.@str1$+"Caçar 10 Réquiems.",.@str2$;
			case 2: callsub L_Quest,10108,.@str1$+"Caçar 10 Réquiems.",.@str2$;
			case 3: callsub L_Quest,10109,.@str1$+"Caçar 10 Réquiems.",.@str2$;
		}
	case 2:
		mes "- Há vários pedidos -";
		mes "- relacionados com Glast Heim -";
		mes "- neste Quadro de Missões. -";
		next;
		set .@str1$,"Tenho muito medo que os monstros de Glast Heim nos ataquem. ";
		set .@str2$,"- Ancélia de Geffen -";
		switch(select("Caçar Moldura Maldita:Caçar Druida Maligno:Caçar Alma Penada:Caçar Raydric Arqueiro")) {
			case 1: callsub L_Quest,10110,.@str1$+"Caçar 10 Molduras Malditas.",.@str2$;
			case 2: callsub L_Quest,10111,.@str1$+"Caçar 10 Druidas Malignos.",.@str2$;
			case 3: callsub L_Quest,10112,.@str1$+"Caçar 10 Almas Penadas.",.@str2$;
			case 4: callsub L_Quest,10113,.@str1$+"Caçar 10 Raydrics Arqueiros.",.@str2$;
		}
	case 3:
		mes "- Há vários pedidos -";
		mes "- relacionados com a área -";
		mes "- em torno de Juno. -";
		next;
		set .@str1$,"Aqueles que não podem lutar, como eu, têm muito mais dificuldade em viajar devido ao grande número de monstros. ";
		set .@str2$,"- Jeanbai, viajante -";
		switch(select("Caçar Grand Peco:Caçar Rochoso:Caçar Bode:Caçar Harpia")) {
			case 1: callsub L_Quest,10114,.@str1$+"Caçar 20 Grands Pecos.",.@str2$;
			case 2: callsub L_Quest,10115,.@str1$+"Caçar 20 Rochosos.",.@str2$;
			case 3: callsub L_Quest,10116,.@str1$+"Caçar 20 Bodes.",.@str2$;
			case 4: callsub L_Quest,10117,.@str1$+"Caçar 20 Harpias.",.@str2$;
		}
	case 4:
		mes "- Há vários pedidos -";
		mes "- relacionados com a Torre do Relógio -";
		mes "- neste Quadro de Missões. -";
		next;
		set .@str1$,"Eu adorava ouvir a doce melodia da Torre do Relógio em Al De Baran, mas já não consigo ouvir nada devido a todos os ruídos que os monstros estão fazendo. ";
		set .@str2$,"- Rizingsetter, Presidente da Comunidade do Relógio do Amor -";
		switch(select("Caçar Relógio:Caçar Punk:Caçar Rideword")) {
			case 1: callsub L_Quest,10118,.@str1$+"Caçar 15 Relógios antes de quebrar.",.@str2$;
			case 2: callsub L_Quest,10119,.@str1$+"Caçar 15 Punks antes de quebrar.",.@str2$;
			case 3: callsub L_Quest,10120,.@str1$+"Caçar 15 Ridewords antes de quebrar.",.@str2$;
		}
	case 5:
		mes "- Há vários pedidos -";
		mes "- relacionados com as Ilhas Localizadas -";
		mes "- neste Quadro de Missões. -";
		next;
		set .@str1$,"Quero partir para uma aventura, mas há tantas coisas com que tenho de lidar. Se eu não posso, então outros também não podem! ";
		set .@str2$,"- Funfy, quem quer partir para uma aventura -";
		switch(select("Caçar Kikimora:Caçar Boneco de Miyabi:Caçar Mi Gao:Caça à Mula sem Cabeça:Caçar Tamruan:Reunir Máscaras do Festival")) {
			case 1: callsub L_Quest,10121,.@str1$+"Caçar 15 Kikimoras em Moscóvia!",.@str2$;
			case 2: callsub L_Quest,10122,.@str1$+"Caçar 15 Bonecos de Miyabi em Amatsu!",.@str2$;
			case 3: callsub L_Quest,10123,.@str1$+"Caçar 15 Mi Gaos em Luoyang!",.@str2$;
			case 4: callsub L_Quest,5055,"Sei que há muitos monstros aterradores fora da cidade, mas o que me aterroriza são estas Mula sem Cabeça à espreita. Espero que elas nunca se aventurem nas cidades. Caçar 15 Mula sem Cabeça!","- Mariana de Brasilis -";
			case 5: callsub L_Quest,5056,"Os Tamruans fazem muito barulho. Não me importo de manhã, mas eles impedem-me de adormecer à noite. Eu quero dormir. Caçar 15 Tamruans!","- Shaman Pumo de Ayothaya -";
			case 6: callsub L_Quest,5057,"Os Pirralhos roubaram todas as Máscaras que eu tinha preparado para o festival. Tentem obter 30 Máscaras! Ficarei te devendo~","- SaYumMoon Chefe de Kunlun -";
		}
	}
	end;

//callsub L_Quest,<ID da missão>{,"<descrição1>","<descrição2>"};
L_Quest:
	set .@quest, getarg(0);
	switch(.@quest) {
		case 10107: set .@name$,"Caça a Réquiem"; set .@exp,15000; break;
		case 10108: set .@name$,"Caça a Marduk"; set .@exp,15000; break;
		case 10109: set .@name$,"Caça a Pasana"; set .@exp,15000; break;
		case 10110: set .@name$,"Caça a Moldura Maldita"; set .@exp,20000; break;
		case 10111: set .@name$,"Caça ao Druida Maligno"; set .@exp,20000; break;
		case 10112: set .@name$,"Caça a Alma Penada"; set .@exp,20000; break;
		case 10113: set .@name$,"Caça ao Raydric Arqueiro"; set .@exp,20000; break;
		case 10114: set .@name$,"Caça ao Grand Peco"; set .@exp,30000; break;
		case 10115: set .@name$,"Caça ao Rochoso"; set .@exp,30000; break;
		case 10116: set .@name$,"Caça ao Bode"; set .@exp,30000; break;
		case 10117: set .@name$,"Caça a Harpia"; set .@exp,30000; break;
		case 10118: set .@name$,"Caça ao Relógio"; set .@exp,25000; break;
		case 10119: set .@name$,"Caça ao Punk"; set .@exp,25000; break;
		case 10120: set .@name$,"Caça ao Rideword"; set .@exp,25000; break;
		case 10121: set .@name$,"Caça ao Kikimora"; set .@exp,30000; break;
		case 10122: set .@name$,"Caça ao Boneco de Miyabi"; set .@exp,30000; break;
		case 10123: set .@name$,"Caça ao Mi Gao"; set .@exp,30000; break;
		case 5055: set .@name$,"Caça a Mula sem Cabeça"; set .@exp,30000; break;
		case 5056: set .@name$,"Caça ao Tamruan"; set .@exp,30000; break;
		case 5057: set .@name$,"Reunir Máscaras do Festival"; set .@exp,30000; break;
	}
	if (getargcount() == 1) {
		mes "'"+.@name$+"' concluída. Quer entregar e receber sua recompensa?";
		next;
		if(select("Entregar a missão.:Cancelar.") == 1) {
			if (.@quest == 5057) delitem 7187,30; //Máscara do Festival
			erasequest .@quest;
			getexp .@exp,0;
			mes "Você recebeu EXP por concluir a missão.";
			next;
			return;
		}
		close;
	} else {
		if (.@quest == 5057) {
			// A missão 5057 é a única missão de coleta, portanto condições separadas são desnecessárias.
			if (checkquest(5057) == -1)
				set .@hunting,-1;
			else if (countitem(7187) >= 30)
				set .@hunting,2;
		} else
			set .@hunting, checkquest(.@quest,HUNTING);
		if (.@hunting == 0 || .@hunting == 1) {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "Volte aqui quando tiver terminado a missão.";
		} else if (.@hunting == 2) {
			mes "Concluiu a missão.";
			next;
			if(select("Entregar a missão.:Cancelar.") == 1) {
				if (.@quest == 5057) delitem 7187,30; //Máscara do Festival
				erasequest .@quest;
				getexp .@exp,0;
				mes "Você recebeu EXP por concluir a missão.";
			}
		}/* else if (.@hunting == 3) {
//			recall_completequest .@quest
			erasequest .@quest;
			mes "Você aceitou novamente a missão '"+.@name$+"'.";
		}*/ else {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "Você quer aceitar a missão?";
			next;
			if(select("Sim.:Não.") == 1) {
				mes "Você recebeu a missão '"+.@name$+"'. Veja as informações da sua missão para mais detalhes.";
				setquest .@quest;
			}
		}
		close;
	}
}
