	// Se o sistema VIP estiver habilitado, os jogadores pagaram como taxa uma 'Pedra de Reset de Atributos' por 5 entradas (com escolha de teleporte).
	// Se estiver desabilitado, os jogadores pagaram a taxa de 5.000z por entrada.
	if (VIP_SCRIPT) {
		mes "raiva contra mim. Vai te custar 1 'Pedra de Reset de Atributos' para 5 acessos.";
		next;
		mes "[Riss]";
		if (bossnia_event > 0) {
			set .@type,1;
			mes "Acessos restantes: "+bossnia_event;
		} else if (countitem(6320)) {
			set .@type,2;
			mes "Você quer ir?";
		} else {
			mes "Você não tem um ingresso agora.....";
			mes "Então venha até mim novamente com uma 'Pedra de Reset de Atributos' mais tarde.";
			close;
		}
		next;
		if(select("Entrar:Sair") == 2)
			close;
		set .@i, select("Primeiro:Segundo:Terceiro:Quarto");
		switch(select("Teleporte 1:Teleporte 2:Teleporte 3:Teleporte 4")) {
			case 1: set .@x,31; set .@y,208; break;
			case 2: set .@x,31; set .@y,31; break;
			case 3: set .@x,208; set .@y,31; break;
			case 4: set .@x,208; set .@y,208; break;
		}
		if (.@type == 1)
			set bossnia_event, bossnia_event-1;
		else {
			delitem 6320,1; //Pedra de Reset de Atributos
			set bossnia_event,4;
		}
		specialeffect2 EF_MAXPOWER;
		warp "bossnia_0"+.@i,.@x,.@y;
		close;
	} else {
	mes "mágoas de mim. Isso irá te custar 5,000 zeny.";
	next;
	if (Zeny > 4999) {
		mes "[Riss]";
		mes "Você realmente quer aceitar o desafio?";
		mes "Ok, apenas escolha seu caminho.";
		next;
		set .@i, select("Primeiro","Segundo","Terceiro","Quarto");
		mes "[Riss]";
		mes "Se cuida, cara~";
		mes "Não se magoe comigo.";
		close2;
		set Zeny, Zeny - 5000;
		warp "bossnia_0"+.@i,rand(202,204),rand(202,204);
		end;
	} else {
		mes "[Riss]";
		mes "Você não tem zeny suficiente...";
		mes "Volte quando possuir ao menos 5,000 zeny.";
		close;
		}
	}
}
