function	script	Q_TeletransportadoraRA	{

		warp $QW_SP_WarpMap$, $QW_SP_WarpX, $QW_SP_WarpY;
		close2;

	case 4: goto L_dungeon;
	case 5: goto L_FewWarps;
	case 6: goto L_NoUnlock;
	case 7: goto L_heal_Full;
	case 8: goto L_heal_Part;

	case 10: goto L_GStorage;
	case 11:
	default:
		goto L_end;

	case 2: set $QW_DW_OFF,($QW_DW_OFF)?0:1; goto GM_Menu;
	case 3: goto DungeonLevelLimit;
	case 4: set $QW_MapUserShow,($QW_MapUserShow)?0:1; goto GM_Menu;

		if ($QW_HF == 0) {
			set $QW_HF,1;
			set $QW_HP,0;
		}
		else set $QW_HF,0;
		goto GM_Menu;
	case 6:
		if ($QW_HP == 0) {
			set $QW_HP,1;
			set $QW_HF,0;
		}
		else set $QW_HP,0;
		goto GM_Menu;
	case 7: set $QW_Stor,($QW_Stor)?0:1; goto GM_Menu;
	case 8: set $QW_GStor,($QW_GStor)?0:1; goto GM_Menu;
	case 9: set $QW_KPoint,($QW_KPoint)?0:1; goto GM_Menu;
	case 10: goto Setprice;
	case 11: goto SpecialWarpMenu;
	default:
		close;
		end;
	}

	if ($QW_S_PRICE == 60) mes "Armazém = ^0000FF Modo Kafra ^000000";
	mes "Armazém de Clãs = "+(($QW_GS_PRICE)?"^00FF00"+$QW_GS_PRICE:"^FF0000Grátis")+"^000000";
	
	switch(select("Teletransporte - Básico","Teletransporte - Avançado","Teletransporte - Ultra Avançado","Taxa do teletransporte Básico de Cavernas","Taxa do teletransporte Avançado de Cavernas","Taxa do teletransporte Ultra Avançado de Cavernas","Cura Completa","Cura Parcial","Armazém","Armazém de Clãs","Voltar","Sair")) {

		next;
		mes "Teletransporte - Básico = "+(($QW_BW_PRICE)?"^00FF00"+$QW_BW_PRICE:"^FF0000Grátis")+"^000000";
		mes "Básico - Teletransportes para as cidades iniciais e cavernas relacionadas.";
		input $QW_BW_PRICE;
		goto Setprice;
	case 2:
		next;
		mes "Teletransporte - Avançado = "+(($QW_AW_PRICE)?"^00FF00"+$QW_AW_PRICE:"^FF0000Grátis")+"^000000";
		mes "Avançado - Teletransportes para cidades e cavernas que se encontram na mesma ilha, mas não perto de nenhuma cidade inicial."; 
		input $QW_AW_PRICE;
		goto Setprice;
	case 3:
		next;
		mes "Teletransporte - Ultra Avançado = "+(($QW_OW_PRICE)?"^00FF00"+$QW_OW_PRICE:"^FF0000Grátis")+"^000000";
		mes "Ultra Avançado - Teletransportes para cidades e cavernas no exterior acessíveis por barcos de Alberta."; 
		input $QW_OW_PRICE;
		goto Setprice;
	case 4:
		next;
		mes "Taxa de Telestransporte de nível Básico de Cavernas = "+(($QW_BW_FEE)?"^00FF00"+$QW_BW_FEE:"^FF0000NO")+"^000000 zeny por nível.";
		mes "Básico - Teletransportes para cavernas relacionadas à cidades iniciais.";
		mes "As taxas dos teletransportes das cavernas são calculadas por mapas que ficam longe da entrada das cavernas vezes a taxa de teletransporte das cavernas.";
		mes "Esses custos estão acima dos custos regulares dos Teletransportes.";
		input $QW_BW_FEE;
		goto Setprice;
	case 5:
		next;
		mes "Taxa de Telestransporte de nível Avançado de Cavernas = "+(($QW_AW_FEE)?"^00FF00"+$QW_AW_FEE:"^FF0000NO")+"^000000 zeny por nível.";
		mes "Avançado - Teletransportes de cavernas que não ficam perto de nenhuma cidade inicial.";
		mes "As taxas dos teletransportes das cavernas são calculadas por mapas que ficam longe da entrada das cavernas vezes a taxa de teletransporte das cavernas.";
		mes "Esses custos estão acima dos custos regulares dos Teletransportes.";
		input $QW_AW_FEE;
		goto Setprice;
	case 6:
		next;
		mes "Ultra Avançado Dungeon level warp fee = "+(($QW_OW_FEE)?"^00FF00"+$QW_OW_FEE:"^FF0000NO")+"^000000 zeny por nível.";
		mes "Ultra Avançado - Teletranspotes de cavernas relacionadas à cidades que ficam no exterior, e que são acessíveis pelos barcos de Alberta.";
		mes "As taxas dos teletransportes das cavernas são calculadas por mapas que ficam longe da entrada das cavernas vezes a taxa de teletransporte das cavernas.";
		mes "Esses custos estão acima dos custos regulares dos Teletransportes.";
		input $QW_OW_FEE;
		goto Setprice;
	case 7:
		next;
		mes "Cura Completa = "+(($QW_HF_PRICE)?"^00FF00"+$QW_HF_PRICE:"^FF0000Grátis")+"^000000";
		mes "Preço completo da Cura instantânea 1.";
		input $QW_HF_PRICE;
		goto Setprice;
	case 8:
		next;
		mes "^FF0000HP ^000000Cura Parcial = "+(($QW_HP_H_PRICE)?"^00FF00"+$QW_HP_H_PRICE+"^000000 um ponto":"^FF0000Grátis^000000");
		mes "^FF0000SP ^000000Cura Parcial = "+(($QW_HP_S_PRICE)?"^00FF00"+$QW_HP_S_PRICE+"^000000 um ponto":"^FF0000Grátis^000000");
		mes "Preço de cura por 1 HP.";
		mes "preço de cura por 1 SP.";
		mes "2 entradas, primeiro HP e depois SP.";
		input $QW_HP_H_PRICE;
		input $QW_HP_S_PRICE;

		next;
		if ($QW_S_PRICE && $QW_S_PRICE != 60) mes "Armazém = ^00FF00 "+$QW_S_PRICE+" ^000000";

		if ($QW_S_PRICE == 60) mes "Armazém = ^0000FF Modo Kafra ^000000";

//======================Menu=Teletransporte=Especial==========================

	if ($QW_SP_Warpname$ == "") set $QW_SP_Warpname$,"Special Warp";
	next;
	mes "Role para baixo para ver todas as informações.";
	mes "A opção do menu Teletransporte Especial será mostrada no menu príncipal quando todas as cidades e cavernas estiverem desbloqueadas.";
	mes "E quando o mapa para o Teletransporte Especiall for definido.";
	mes "Nome da opção do menu definido como ^0000FF"+$QW_SP_Warpname$+"^000000";
	if ($QW_SP_WarpMap$ == "") mes "o mapa está atualmente ^FF0000não^000000 definido, e o menu de Teletransporte Especial está desativado.";
	if ($QW_SP_WarpMap$ != "") mes "o mapa está atualmente definido como ^0000FF"+$QW_SP_WarpMap$+"^000000 e o menu de Teletransporte Especial foi";
	mes "definida nas seguintes coordenadas ^0000FF"+$QW_SP_WarpX+","+$QW_SP_WarpY+"^000000";

		next;
		mes "Defina o nome para mostrar no menu como opção.";

	case 2:
		next;
		mes "Configure o mapa no formato ^0000FFmapname^000000.";
		mes "Quando este Mapa estiver definido, a opção para os jogadores será mostrada assim que eles atenderem aos requísitos.";

		next;

		mes "Segunda Coordenada = Ycoord";

		goto GM_Menu;

		end;
	}

//======================Limite=Níveis=Cavernas========================

DungeonLevelLimit:
	next;
	mes "Limite de profundidade das cavernas está "+(($QW_DL)?"^00FF00Ativado":"^FF0000Desativado")+"^000000";
	mes "Se^00FF00 Ativado ^000000O limite de Profundidade das cavernas é ajustado para ^0000FF"+$QW_DDL+"^000000.";

	switch(select("Alternar Limite de Profundidade de Cavernas","Definir Limite de Profundidade de cavernas","Voltar para o Menu GM","Sair")){

//===========================Cidades=================================

L_town:
	callfunc "QWS_Tarray";
	freeloop(1);
	set .@menu$,"";
	for(set .@i,0; .@i<35; set .@i,.@i+1)	// O máximo esperado é 35
		set .@menu$,.@menu$+@Tmenulist$[.@i]+":";
	freeloop(0);
	set @TWMenu,select(.@menu$);

	if (@Tmenuref[@TWMenu-1] == 57005) goto L_End; // 57005='morto' em hexadecimal

	if(Zeny<@pTprice[@Tmenuref[@TWMenu-1]]) callsub L_Short_on_zeny,0;
	set Zeny, Zeny-@pTprice[@Tmenuref[@TWMenu-1]];
	if ($QW_KPoint == 1) set RESRVPTS, RESRVPTS + (@pTprice[@Tmenuref[@TWMenu-1]]/16);
	warp @pTmap$[@Tmenuref[@TWMenu-1]],@pTXcoords[@Tmenuref[@TWMenu-1]],@pTYcoords[@Tmenuref[@TWMenu-1]];
	close2;
	set Zeny, Zeny+@pTprice[@Tmenuref[@TWMenu-1]];
	end;

//=========================Cavernas================================

L_dungeon:
	callfunc "QWS_Darray";
	freeloop(1);
	set .@menu$,"";
	for(set .@i,0; .@i<35; set .@i,.@i+1)	// O máximo esperado é 35
		set .@menu$,.@menu$+@Dmenulist$[.@i]+":";
	freeloop(0);
	set @DWMenu,select(.@menu$);

	if (@Dmenuref[@DWMenu-1] == 57005) goto L_End; // 57005='morto' em hexadecimal
	set @DwarpMenu, (@Dmenuref[@DWMenu-1]);
	callfunc "QWS_DLarray";

	next;
	mes "[TeletransportadoraRA]";
	mes "Por favor, selecione onde você deseja ir:";
	freeloop(1);
	set .@menu$,"";
	for(set .@i,0; .@i<18; set .@i,.@i+1)	// O máximo esperado é 18
		set .@menu$,.@menu$+@DWLmenulist$[.@i]+":";
	freeloop(0);
	set @DWLMenu,select(.@menu$);

	if (@DWLmenuref[@DWLMenu-1] == 57005) goto L_End; // 57005='morto' em hexadecimal

	set @Darrayref, @DWLmenuref[@DWLMenu-1];
	set @warpprice, @pDprice[@Dmenuref[@DWMenu-1]]+(getd(@pDfee$[@Dmenuref[@DWMenu-1]])*(@DDepth[@Darrayref]));

	if(Zeny<@warpprice) callsub L_Short_on_zeny,1;
	set Zeny, Zeny-(@warpprice);
	if ($QW_KPoint == 1) set RESRVPTS, RESRVPTS + (@warpprice/16);
	warp @DGat$[@Darrayref],@DXcoords[@Darrayref],@DYcoords[@Darrayref];
	close2;
	set Zeny, Zeny+@pTprice[@Tmenuref[@DWMenu-1]];
	end;

//=============================Healing=============================

L_heal_Full:
	set @healfee, $QW_HF_PRICE;
	if(Zeny<@healfee) callsub L_Short_on_zeny,4;
	set Zeny, Zeny-@healfee;
	if ($QW_KPoint == 1) set RESRVPTS, RESRVPTS + (@healfee/500);
	next;
	mes "[TeletransportadoraRA]";
	mes "Feche a janela e eu irei te curar.";
	close2;
	percentheal 100,100;
	end;

L_heal_Part:
	set @healchoice,select("Cura completa","Apenas pontos de cura","Apenas pontos de habilidade","Sair");
	if (@healchoice == 1) callsub PHeal,1,1;
	if (@healchoice == 2) callsub PHeal,1,0;
	if (@healchoice == 3) callsub PHeal,0,1;
	goto L_end;
	
PHeal:
	next;
	set @Hp, MaxHp-Hp;
	set @Sp, MaxSp-Sp;
	set @HpPrice, @hp*$QW_HP_H_PRICE;
	set @SpPrice, @sp*$QW_HP_S_PRICE;
	mes "[TeletransportadoraRA]";
	if(getarg(0) == 1) mes ""+@HpPrice+" Zeny para "+@Hp+" pontos de cura.";
	if(getarg(1) == 1) mes ""+@SpPrice+" Zeny para "+@Sp+" pontos de habilidade.";
	set @total, @HpPrice+@SpPrice;
	mes "para um total de "+@total+" zeny";
	if (select("Cura-me","Deixe-me ver as opções novamente")==2) goto L_heal_Part;

	if(getarg(0) == 1)set @HpPrice, (MaxHp-Hp)*$QW_HP_H_PRICE;
	if(getarg(1) == 1)set @SpPrice, (MaxSp-Sp)*$QW_HP_S_PRICE;
	set @healfee, @HpPrice+@SpPrice;
	if (getarg(0) == 1)&&(getarg(1) == 1)&&(Zeny<@healfee) goto Zeny_Short_Both;
	if (getarg(0) == 1)&&(Zeny<@healfee) goto Zeny_short_HP;
	if (getarg(1) == 1)&&(Zeny<@healfee) goto Zeny_short_SP;
	set Zeny, Zeny-@healfee;
	if (getarg(0) == 1)&&(getarg(1) == 1) percentheal 100,100;
	if (getarg(0) == 1) percentheal 100,0;
	if (getarg(1) == 1) percentheal 0,100;
	close;
	end;

Zeny_Short_Both:
	mes "[TeletransportadoraRA]";
	mes "Escolha outra opção, você pode pagar por ambos.";
	mes "Eu posso curar tanto quanto você pode pagar também.";
	if (select("OK","Sair") == 2) goto L_end;
	goto PHeal;

Zeny_short_HP:
	mes "[TeletransportadoraRA]";
	mes "Você quer que eu cure parcialmente seu HP?";
	if (select("Sim","Não") == 2) goto L_end;
	set @Hp, Zeny/$QW_HP_H_PRICE;
	set @HpPrice, @Hp*$QW_HP_H_PRICE;
	if (@Hp == 1) mes "Você não vale o esforço.";
    	if (@Hp == 1) goto L_end;
    	set Zeny, Zeny-@HpPrice;
	heal @Hp,0;
	close;

function	script	QWS_TownStamps	{
	set @Tstamp,0;
	set @MaxTstamp,30; //número máximo de cidades
	set @binvalue,1;
	set @Tstamploop,0;
	do {
		if ((@binvalue & ##QWS_T_Unlock) == @binvalue) set @Tstamp,@Tstamp+1;
		set @binvalue, @binvalue *2;
		set @Tstamploop, @Tstamploop + 1;
	} while (@Tstamploop < @MaxTstamp);
	return;
}

	set @Dstamp,0;

		set @binvalue, @binvalue *2;
		set @Dstamploop, @Dstamploop + 1;
	} while (@Dstamploop < @MaxDstamp);
	return;
}

//----------------Make Town Menu Function

function	QWS_Make_Town_Menu	{
	set @temptownmenubin,1;
	if (getarg(0) == 0) goto menu_item;
	set @templooptownmenu,0;
	do {
		set @temptownmenubin, @temptownmenubin * 2;
		set @templooptownmenu, @templooptownmenu + 1;
		}while (getarg(0) > @templooptownmenu);
// checa o marcador e menu de item 
menu_item:
	if ((@temptownmenubin & ##QWS_T_Unlock) != @temptownmenubin) {
		set @Ti,@Ti+1;
		return;
	}
	if (@pTprice[@Ti] != 0)&&($QW_MapUserShow == 0) set @Tmenulist$[@Tj], @pTmenuitems$[@Ti]+" -> "+@pTprice[@Ti];
	if (@pTprice[@Ti] == 0)&&($QW_MapUserShow == 0) set @Tmenulist$[@Tj], @pTmenuitems$[@Ti];
	if (@pTprice[@Ti] != 0)&&($QW_MapUserShow == 1) set @Tmenulist$[@Tj], @pTmenuitems$[@Ti]+" ["+getmapusers(@pTmap$[@Ti])+"]"+" -> "+@pTprice[@Ti];
	if (@pTprice[@Ti] == 0)&&($QW_MapUserShow == 1) set @Tmenulist$[@Tj], @pTmenuitems$[@Ti]+" ["+getmapusers(@pTmap$[@Ti])+"]";
	set @Tmenuref[@Tj],@Ti;
	set @Tj,@Tj+1;
	set @Ti,@Ti+1;
	return;
}
}

//----------------Torna a função de menu do calabouço

function	QWS_Make_Dungeon_Menu	{
	set @tempdungeonmenubin,1;
	if (getarg(0) == 0) goto menu_item;
	set @temploopdungeonmenu,0;
	do {
		set @tempdungeonmenubin, @tempdungeonmenubin * 2;
		set @temploopdungeonmenu, @temploopdungeonmenu + 1;
		}while (getarg(0) > @temploopdungeonmenu);
// check marker and make menu item
menu_item:
	if ((@tempdungeonmenubin & ##QWS_D_Unlock) != @tempdungeonmenubin) {
		setarray @DLocRef[@Di], @Dref;
		set @Dref, @Dref+@DLevels[@Di];
		set @Di,@Di+1;
		return;
	}
	if ($QW_MapUserShow == 1) {
		set @tempmapusers,0;
		set @mapusersloop,0;
		do {
			set @tempmapusers,(getmapusers(@DGat$[@Dref+@mapusersloop]) + @tempmapusers);
			set @mapusersloop, (@mapusersloop + 1);
		} while (@DLevels[@Di] > @mapusersloop);
		if (@pDprice[@Di] != 0) set @Dmenulist$[@Dj], @pDmenuitems$[@Di]+" ["+@tempmapusers+"] -> "+@pDprice[@Di];
		if (@pDprice[@Di] == 0) set @Dmenulist$[@Dj], @pDmenuitems$[@Di]+" ["+@tempmapusers+"]";
		set @Dmenuref[@Dj],@Di;
		set @Dj,@Dj+1;
		setarray @DLocRef[@Di], @Dref;
		set @Dref, @Dref+@DLevels[@Di];
		set @Di,@Di+1;
		return;
	}
	if (@pDprice[@Di] != 0) set @Dmenulist$[@Dj], @pDmenuitems$[@Di]+" -> "+@pDprice[@Di];
	if (@pDprice[@Di] == 0) set @Dmenulist$[@Dj], @pDmenuitems$[@Di];
	set @Dmenuref[@Dj],@Di;
	set @Dj,@Dj+1;
	setarray @DLocRef[@Di], @Dref;
	set @Dref, @Dref+@DLevels[@Di];
	set @Di,@Di+1;

function	script	QWS_DLarray	{
//----------------Inicia o menu de construção
	set @DWref,@DLocRef[@DwarpMenu]; // referência aos arrays com leveldata
	set @DWi,0;  // loop counter
	set @DWj,0; // menu lines counter
	cleararray @DWLmenulist$[0],"",20; // Limpando a matriz para se livrar de itens fantasma no menu
	do {
		if ($QW_DL == 0)||($QW_DDL >= @DDepth[@DWref]) {
			set @warpprice, (@pDprice[@DwarpMenu]+ (getd(@pDfee$[@DwarpMenu])*@DDepth[@DWref]));
			if (@warpprice != 0)&&($QW_MapUserShow == 0) set @DWLmenulist$[@DWj], @DLevelName$[@DWref]+" -> "+@warpprice;
			if (@warpprice == 0)&&($QW_MapUserShow == 0) set @DWLmenulist$[@DWj], @DLevelName$[@DWref];
			if (@warpprice != 0)&&($QW_MapUserShow == 1) set @DWLmenulist$[@DWj], @DLevelName$[@DWref]+" ["+getmapusers(@DGat$[@DWref])+"] -> "+@warpprice;
			if (@warpprice == 0)&&($QW_MapUserShow == 1) set @DWLmenulist$[@DWj], @DLevelName$[@DWref]+" ["+getmapusers(@DGat$[@DWref])+"]";
			set @DWLmenuref[@DWj],@DWref;
			set @DWj,@DWj+1;
		}
		set @DWref, @DWref+1;
		set @DWi,@DWi+1;
	} while (@DWi < @DLevels[@DwarpMenu]);
	set @DWLmenulist$[@DWj], "Exit";
	set @DWLmenuref[@DWj],57005; // 57005='morto' em hexadecimal
	return;
}
