lighthalzen,108,229,2	script	La Fenn	766,{
	mes "[La Fenn]";
	if(BaseLevel<99) {
		if(Q_99LVL) {
			mes "Você... Você ficou mais forte!!! Mas você é ainda definitivamente VOCÊ!";
			mes "Eu não poderia imaginar que você fosse capaz de cumprir minha condição...";
			emotion ET_SCRATCH;
			next;
			mes "[La Fenn]";
			mes "Aqui está seu prêmio. É seu para sempre.";
			set Q_99LVL,0;
			getitem 12106,1; //Caixa de Jóias
			emotion ET_CONGRATULATION;
		} else {
			mes callfunc("F_Hi");
			mes "...";
			emotion ET_SLEEPY;
			next;
			mes "[La Fenn]";
			mes "Eu tenho um enigma para homens fortes de 2ª Classe e Nível de Base 99.";
			mes "*Bocejo* Se você conhece alguns homens assim, por favor, conte a eles sobre o meu enigma.";
		}
		close;
	}
	if(BaseLevel==99 && Upper==0) {
		if(Q_99LVL) {
			mes "Agora vá em frente e fique mais forte!";
			mes "Se você resolver o meu enigma, então eu te darei um estojo mágico com um acessório raro.";
		} else {
			mes callfunc("F_Hi")+" O Herói!";
			if(BaseExp<1000000) {
				mes "Agora você está mais forte.";
				mes "Mas... não forte o suficiente para o meu enigma.";
				if(rand(2)) {
					mes "Matar mais "+((1000000-BaseExp)/3765+1)+" selos e retorne.";
				} else {
					mes "Matar mais "+((1000000-BaseExp)/81+1)+" Poporings e retorne.";
				}
			} else {
				mes "Agora você está REALMENTE forte.";
				mes "E se você ficar mais forte do que está agora, eu lhe darei meu tesouro.";
				set Q_99LVL,1;
				emotion ET_BEST;
			}
		}
		close;
	}
	emotion ET_SCRATCH;
	mes "...";
	close;
}
