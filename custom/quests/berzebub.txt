prontera,165,178,7	script	Mulher Velha	103,{  //Local Original

	setarray .@Items[0],714,25,1061,100,1095,200,618,10,1551,2,1614,1,727,20;
	set .@n$,"[Mulher Velha]";

	mes .@n$;
	if(BerzQuest == 1) {
	mes "Ei, eu lembro de você! Eu já te disse que você só pode completar essa missão uma vez.";
	close; }
	mes "Saudações, meu rapaz. Eu encontrei um item muito bacana enquanto eu estava colhendo cogumelos perto de Glast Heim.";
	next;
	if(select("Continuar conversando.:Sair")==2) goto M_Leave;

	mes .@n$;
	mes "Eu nunca vi um item como este antes. Me disseram que ele pode ser colocado dentro de qualquer acessório com slot disponível.";
	next;
	mes .@n$;
	mes "Se você estiver interessado neste item, eu lhe direi o que você precisa fazer para obtelo. Eu só posso dar um desses para cada jogador, então uma vez que você tenha completado esta missão, você não poderá mais fasela novamente.";
	next;
	if(select("Fale mais sobre isso.:Não, eu não me importo com isso.")==2) goto M_Leave;

	mes .@n$;
	mes "Eu acredito que o item é chamado de ^FF0000Carta Belzebu^000000. Faz você lançar feitiços muito mais rápido!";
	next;
	mes .@n$;
	mes "Eu precisarei de todos os itens a seguir:";
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		mes " ~ "+.@Items[.@i+1]+"x "+getitemname(.@Items[.@i]);
		if (countitem(.@Items[.@i]) < .@Items[.@i+1]) set .@nr,1; }
	next;
	if(select("Eu tenho tudo isso!:Eu conseguirei aqueles ASAP.")==2) goto M_Leave;

	mes .@n$;
	if (.@nr) {
		mes "Você mente! Saia daqui e pegue esses itens, e me desculpe sua estrela do rock.";
		close; }
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2)
		delitem .@Items[.@i], .@Items[.@i+1];
	getitem 4145,1;
	set BerzQuest, 1;
	mes "Parabéns! Você tem todos os itens. Aqui está a sua ^FF0000Carta Belzebu^000000, como prometido.";
	emotion ET_CONGRATULATION;
	close;

M_Leave:
	mes .@n$;
	mes "Que ótimo item eu tenho aqui...";
	close;
}
