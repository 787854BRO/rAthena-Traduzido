	mes "[Mensageiro Real]";
	mes "Bem-vind"+(Sex?"o":"a")+" a Prontera.";
	mes "Eu sou o mensageiro responsável pelas Missões Reais.";
	next;
	if(select("Me fale mais...:Vou dar um passeio")==2) close;
	mes "[Mensageiro Real]";
	mes "Existem duas missões.";
	mes "Escolha o que você mais gosta.";
	next;
	set .@i, select("Balmung:Chifre de Lord Kaho:^777777Não importa.^000000");
	mes "[Mensageiro Real]";
	if (.@i==3) {
		mes "Tenha um bom dia.";
		close; }
	mes "Você precisará dos seguintes itens:";
	for(set .@j,1; .@j<13; set .@j,.@j+2) {
		mes " ~ "+getd(".@Req"+.@i+"["+(.@j+1)+"]")+"x ^0055FF"+getitemname(getd(".@Req"+.@i+"["+.@j+"]"))+"^000000";
		if (!.@nr && countitem(getd(".@Req"+.@i+"["+.@j+"]")) < getd(".@Req"+.@i+"["+(.@j+1)+"]")) set .@nr,1; }
	mes " ~ 1,000,000 de zenys.";
	next;
	mes "[Mensageiro Real]";
	if (.@nr || Zeny < 1000000) {
		mes "Volte quando você atender a todos os requisitos.";
		close; }
	for(set .@j,1; .@j<13; set .@j,.@j+2)
		delitem getd(".@Req"+.@i+"["+.@j+"]"), getd(".@Req"+.@i+"["+(.@j+1)+"]");
	set Zeny, Zeny-1000000;
	mes "Eu vejo que você já tem todos os itens que eu preciso. Aqui está o seu "+getitemname(getd(".@Req"+.@i+"[0]"))+", como prometido.";
	getitem getd(".@Req"+.@i+"[0]"),1;
	close;
