umbala,126,129,4	script	Homem Barbudo#bandit	120,{
	if(BEARD_QUEST == 4 ) goto L_MAKE;
	if(BEARD_QUEST >= 5 ) goto L_DONE;
	if(BEARD_QUEST >= 1 ) goto L_THREAD;

	mes "[Homem Barbudo]";
	mes "Barbas! Peguem suas maravilhosas barbas!";
	mes "Você gostaria de uma barba?";
	next;
	menu "^0000ff•^000000 Sim, eu gostaria de alguns pêlos faciais!",-,"Não, obrigado, isso é nojento!",M_NO;

	mes "[Homem Barbudo]";
	mes "HoHo~ Então você quer alguns bigodes.";
	emotion ET_SMILE;
	mes "Bem, eu daria a você minhas barbas, mas não é barato.";
	mes "De fato eu tenho apenas uma sobrando...";
	mes "mas eu não estou vendendo essa.";
	mes "Mas não fique bravo! Eu posso fazer uma para você.";
	next;
	set BEARD_QUEST,1;
L_THREAD:
	mes "[Homem Barbudo]";
	mes "Eu precisaria de algumas ^8080FFLinhas^000000 para a Barba.";
	mes "Fale com meu amigo em Izlude por alguumas ^8080FFLinhas^000000.";
	next;

	mes "[Homem Barbudo]";
	mes "Eu também precisaria que você me trouxesse algumas coisas para faser a barba.";
	mes "Elas são:";
L_LIST:
	mes "^8080FFElástico^000000";
	mes "^8080FF5 Couro de Animal^000000";
	mes "^8080FFTintura Preta^000000";
	mes "^8080FFBigode Grisalho^000000";
	mes "^8080FF100 Muco Pegajoso^000000";
	mes "^8080FFAntígeno^000000";
	mes " e um ^8080FFDiamante de 2 quilates^000000.";
	close;

M_NO:
	mes "[Homem Barbudo]";
	mes "Bem, que assim seja!";
	emotion ET_ANGER;
	close;

L_DONE:
	mes "[Homem Barbudo]";
	mes "Desculpe, eu só posso fazer uma por pessoa.";
	mes "Nós não queremos barbas demais agora. Certo?";
	emotion ET_KIK;
	close;

L_MAKE:
	mes "[Homem Barbudo]";
	mes "Vamos fazer sua barba!";
	next;
	if(countitem(7200) < 1 || countitem(919) < 1 ||	countitem(983) < 1 || countitem(2241) < 1 ||
	countitem(938) < 100 || countitem(973) < 1 || countitem(731) < 1) goto L_NOITEMS;
	delitem 7200, 1;
	delitem 919, 1;
	delitem 983, 1;
	delitem 2241, 1;
	delitem 938, 100;
	delitem 973, 1;
	delitem 731, 1;
	mes "^8080FF~Você o vê cortando e costurando a barba~^000000";
	next;
	mes "^8080FF~Ele te entrega a barba já acabada~^000000";
	next;
	getitem 2237, 1;
	set BEARD_QUEST, 5;
	mes "[Homem Barbudo]";
	mes "Tenha um bom dia!";
	emotion ET_KIK;
	close;

L_NOITEMS:
	mes "[Homem Barbudo]";
	mes "Que diabos! Eu não posso fazer barba sem os itens!";
	mes "Aqui está a lista novamente:";
	emotion ET_HUK;
	goto L_LIST;
}

	if(BEARD_QUEST != 3) mes "O que devo fazer?";
	if(BEARD_QUEST != 3) close;
