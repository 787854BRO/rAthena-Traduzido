// Simple item trade-in quest, a person can get a Dead Branch
// by simply providing the items: Log x3 (7201), Wooden Heart x1 (7189), 
// Trunk x5 (1019), and Wooden Gnarl x1 (7222).
//===== Additional Comments: ================================= 
//= 1.1 Optimized, added Bloody Branch with 0.01% chance [Lupus]
//============================================================

niflheim,204,179,3	script	Sr. Mobry	121,{

	mes "[Sr. Mobry]";
	mes "As árvores possuem espíritos que você conhece...";
	next;

	mes "[Sr. Mobry]";
	mes "Na verdade eu sei um segredo sobre esses espíritos que podem de alguma forma... talvez te ajudar...";
	next;

	mes "[Sr. Mobry]";
	mes "Eu posso criar um Galho Espíritual... isto é, se você puder me fornecer os materiais necessários...";
	next;
	goto L_MENU;

L_MENU:
	menu "O que eu preciso?",-,"Onde posso encontrar essa porcaria?",M_FIND,"Faça-me um Galho Seco!",M_CREATE,"Esqueça isso...",M_END;

	mes "[Sr. Mobry]";
	mes "Mwehehe... Eu vejo que você quer controlar os espíritos das árvores, estou certo? Ok... eu precisarei:";
	mes "^1354453 Tora^000000";
	mes "^1354451 Coração de Madeira^000000";
	mes "^1354455 Tronco^000000";
	mes "^1354451 Coluna de Madeira^000000";
	next;
	goto L_MENU;

M_FIND:
	mes "[Sr. Mobry]";
	mes "Você poderá encontrar os seguintes itens de certos monstros heheh...";
	mes "^135445Toras podem ser encontradas próximo as Árvores dos Golens.^000000";
	mes "^135445Corações de Madeira é a essência das Árvores dos Golens.^000000";
	mes "^135445Troncos? Sério, pense em madeira...^000000";
	mes "^135445Coluna de Madeira... Gibbet...^000000";
	next;
	goto L_MENU;

L_NOLOG:
	mes "[Sr. Mobry]";
	mes "^135445Toras^000000 são o corpo... eu preciso de mais delas.";
	mes "Me traga três toras e eu te farei o espírito...";
	close;

L_NOHEART:
	mes "[Sr. Mobry]";
	mes "O ^135445Coração de Madeira^000000 é a essência do espírito.";
	mes "Traga-me um coração de madeira e farei de você o espírito...";
	close;

L_NOTRUNK:
	mes "[Sr. Mobry]";
	mes "O quê? Você não conseguiu encontrar os ^1354455 Troncos^000000?";
	mes "Mate os ''Elder Willows'' ou algo assim... Nossa... nenhum tronco hehe... patético.";
	close;

L_NOGNARL:
	mes "[Sr. Mobry]";
	mes "Como posso fazer o item sem ^135445Coluna de Madeira^000000?";
	mes "Eu disse procure o ''Gibbet''... eles estão em todos os lugares...";
	close;

M_CREATE:
	mes "[Sr. Mobry]";
	mes "Bem... vamos ver o que você me trouxe...";
	next;
	if(countitem(7201)<3) goto L_NOLOG;
	if(countitem(7189)<1) goto L_NOHEART;
	if(countitem(1019)<5) goto L_NOTRUNK;
	if(countitem(7222)<1) goto L_NOGNARL;
	delitem 7201,3;
	delitem 7189,1;
	delitem 1019,5;
	delitem 7222,1;
	mes "[Sr. Mobry]";
	mes "Heh... Espero que você se divirta com os espíritos das árvores... Cuidado agora...";
	mes "Heh... Cuidado agora? É que eu me preocupo se você di... Não se preocupe.";
	if(rand(1000)==0) goto L_GIVE2;
	getitem 604,1; //Galho Seco
	close;
L_GIVE2:
	getitem 12103,1; //Galho Sangrento
	close;

M_END:
	mes "[Sr. Mobry]";
	mes "Sem espírito para você... então vá!";
	close;
}
