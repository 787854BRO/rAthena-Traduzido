	mes "Agora que finalmente encontrei a fórmula para o elmo perfeito, estou disposto a compartilhar meu tempo e talento.";

		mes "1x ^0055FF"+getitemname(.@Items[.@i])+"^000000 (do monstro "+strmobinfo(1,.@Drops[.@i])+").";

	mes .@n$;

		mes .@n$;

		delitem .@Items[.@i],1;

	set Zeny, Zeny-5000000;
	mes "Uau! Você é realmente corajoso!";
