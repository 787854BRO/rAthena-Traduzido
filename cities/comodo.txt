	mes "rota 'Lutifus'? Você";
	mes "pode viajar na Lutifus";

	next;

		case 1:
			if (Zeny < 600) break;
			set Zeny, Zeny-600;
			warp "alberta",192,169;
			end;
		case 2:
			if (Zeny < 800) break;
			set Zeny, Zeny-800;
			if (checkre(0))
			warp "izlude",195,212;
			else
			warp "izlude",176,182;
			end;

			mes .@n$;
			mes "Viajar de navio é ainda um";
			mes "dos métodos mais seguros e";
			mes "confiáveis de transporte.";
			mes "Eu convido você a experimentar";
			mes "a viagem da rota 'Lutifus' em breve~";
			close;
	}
	mes .@n$;
	mes "Me desculpe, mas você não";
	mes "tem dinheiro suficiente";
