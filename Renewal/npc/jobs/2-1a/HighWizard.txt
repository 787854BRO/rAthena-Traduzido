valkyrie,44,47,5	script	Arquimago#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Arquimago]";
			mes "Parabéns!";
			mes "Honra aos guerreiros!";
			close;
		}
		mes "[Arquimago]";
		mes "Nós Arquimagos temos";
		mes "a responsabilidade de";
		mes "usar nossos poderes destrutivos";
		mes "para o bem.";
		next;
		mes "[Arquimago]";
		mes "Uma vida inteira de treinamento";
		mes "é necessária antes de se tornar";
		mes "um Arquimago. Você pode imaginar";
		mes "o que aconteceria se nossos poderes";
		mes "fossem colocados em mãos erradas?!";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[Arquimago]";
		mes "Chegou a hora.";
		mes "A cidade de Rune-Midgard está";
		mes "precisando daqueles que podem";
		mes "exercer o poder da magia...";
		next;
		mes "[Arquimago]";
		mes "Gostaria de se";
		mes "tornar um Arquimago?";
		next;
		if (select("Não.:Sim.") == 1) {
			mes "[Arquimago]";
			mes "Quando estiver pronto,";
			mes "sinta-se livre para voltar.";
			mes "Honra aos guerreiros!";
			close;
		}
		if (SkillPoint) {
			mes "[Arquimago]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "todos os seus Pontos de Habilidade antes de voltar a falar comigo.";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[Arquimago]";
		mes "Parabéns!";
		mes "Como um Arquimago,";
		mes "espero que você use";
		mes "seus poderes para trazer";
		mes "paz aos necessitados.";
		close;
	}
	mes "[Arquimago]";
	mes "Bem-vind"+((Sex) ? "o":"a")+"";
	mes "à Valhalla,";
	mes "o Salão de Honras.";
	next;
	mes "[Arquimago]";
	mes "Por favor, sinta-se";
	mes "a vontade enquanto";
	mes "estiver por aqui.";
	mes "Honra aos guerreiros.";
	close;
}
