	function ChkResult {
		.@tid = getarg(0);
		.@expected = getarg(1);
		.@ret = getarg(2);
		.@success = (.@ret==.@expected);
		debugmes "Teste "+.@tid+" = "+(.@success?"Sucesso":"Falhou");
		return .@success;
	}

	function FinalReport {
		.@tdone = getarg(0);
		.@success = getarg(1);
		debugmes "Resultado = Passar : "+.@success+"/"+.@tdone+" Falhou : "+(.@tdone-.@success)+"/"+.@tdone;
		if(.@success != .@tdone) { debugmes "Ocorreram uma ou mais falhas. Habilite a impressão chkresult para encontrar testes com falhas."; }
		return;
	}
