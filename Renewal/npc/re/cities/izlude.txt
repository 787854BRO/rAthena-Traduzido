// Canal do Teletransporte :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[Funcionário]";
	mes "Izlude é um lugar muito vivo não é mesmo?";
	mes "Estou aqui para te ajudar a se sentir mais á vontade em Izlude!";
	mes "Eu posso o enviar a outra cópia de Izlude.";
	mes "Você quer ir para outra cópia?";
	next;
	set .@i, select(getarg(0)+":Qualquer uma");
	if (.@i < 6) {
		setarray .@maps$[1],"izlude","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
