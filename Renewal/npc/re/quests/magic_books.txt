function	script	Magic_Book_Essence	{
	if (MaxWeight - Weight < 1000) {
		mes "Por favor, alivie o peso da sua bolsa.";
		close;
	}
	.@item_req = getarg(1);
	.@req_amount = getarg(2);
	if (checkquest(getarg(0)) == 1 && countitem(.@item_req) >= .@req_amount) {
		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		delitem .@item_req,.@req_amount;
		if (rand(2)) {
			mes "Você usou "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" para criar o ^999900Elixir "+ getarg(3) +"^000000.";
			next;
			mes "^008800A Essência de "+getarg(3)+" durará permanentemente, a menos que haja algum problema.^000000";
			completequest getarg(0);
			specialeffect2 EF_LORD;
			close;
		}
		mes "^008888Você falhou ao refinar o Elixir usando "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +".^000000";
		close;
	}
	if (checkquest(getarg(0)) == 2) {
		mes "Você já criou o Elixir "+getarg(3)+" nesta área.";
		close;
	}
	mes "Alguém já usou esta área.";
	close;
}
