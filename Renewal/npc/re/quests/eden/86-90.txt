		// IDs de Missões: 4167-4180

		setarray .@names$[0],

		setarray .@exp[0],
			76000,82000,82000,

			72000,72000,72000,86000,80000;
		mes "Deve-se estar dentro do nível 86-90 para aceitar estas missões.";
		for(set .@quest,4167; .@quest<=4180; set .@quest,.@quest+1) {
			set .@hunting, checkquest(.@quest,HUNTING);
			if (.@hunting == 0 || .@hunting == 1) {
				next;
				mes ":: Você não pode prosseguir caçando o";

		mes "[Quadro de Missões 86-90]";
		mes "Aqui esta uma lista de pedidos aleatorios para cacar monstros em outras regioes de Midgard.";

			case 1: callsub L_Quest,4176,"Petite",72000,"??????","Existem monstros do tipo Dragão invadindo a região ocidental de Geffen.";
			case 2: callsub L_Quest,4177,"Rafflesia",72000,"???????","Existem monstros do tipo Planta infestando os campos de Lighthalzen.";
			case 3: callsub L_Quest,4178,"Nuvem Venenosa",72000,"????","Existem monstros do tipo Amorfo flutuando em torno dos campos de Einbroch.";

//callsub L_Quest,<ID da missão>,"<nome do mostro>",<recompensa de EXP>,"<imagem cutin>","<descrição>"{,"s"};
L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[Quadro de Missões 86-90]";
		mes "Tem de esperar 3 horas antes de poder assumir novamente esta missão.";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[Quadro de Missões 86-90]";
		mes "Você já terminou de caçar todos os 30 "+getarg(1)+getarg(5,"")+"?";
		close;
	} else if (.@hunting == 2) {
		mes "[Quadro de Missões 86-90]";
		mes "Você completou a caça.";
		mes "Por favor, aceite esta recompensa como compensação.";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;
		getexp getarg(2),0;
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1
		erasequest .@quest1;
		mes "[Quadro de Missões 86-90]";
		mes "Você aceitou novamente a missão de caçar o monstro "+getarg(1)+".";
		close;
	}*/ else {
		mes "[Quadro de Missões 86-90]";
		mes getarg(4);
		next;
		if (.@cutin$ != "") cutin .@cutin$,3;
		mes "[Quadro de Missões 86-90]";
		mes "Cace 30 monstros "+getarg(1)+" e publique seu sucesso neste Quadro.";
		next;
		if (.@cutin$ != "") cutin .@cutin$,255;
		if(select("Vou caçá-los.:Não obrigado.") == 1) {
			mes "[Quadro de Missões 86-90]";
			mes "Quando tiver completado a missão, publique neste quadro e receba a sua recompensa.";
			if (.@playtime > -1) erasequest .@quest2;
			setquest .@quest1;
		}
		close;
	}
}
