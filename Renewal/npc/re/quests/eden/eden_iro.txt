function	script	VIP_iRO_Acolyte	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes getarg(0);
		mes "Mas, meus plenos poderes só estão disponíveis para aqueles que são VIPs.";
		mes "Ainda posso lhe oferecer um portal para "+getarg(1)+" e ir embora.";
		next;
		switch(select(getarg(0)+" -> 1800z.:Cancelar.")) {
		case 1:
			if (Zeny < 1800) {
				mes getarg(0);
				mes "Sinto muito, mas você não tem";
				mes "bastante zeny para o Serviço de";
				mes "Teletransporte". A taxa do Teletransporte";
				mes "para "+getarg(1)+" é de 1.800 zenys.";
				close;
			}
			specialeffect2 EF_READYPORTAL;
			specialeffect2 EF_TELEPORTATION;
			specialeffect2 EF_PORTAL;
			next;
			set Zeny, Zeny - 1800;
			if (getarg(1) == "Juno")
				warp "yuno",158,125;
			else
				warp "prontera",116,72;
			close;
		case 2:
			close;
		}
	}
	return;
}
