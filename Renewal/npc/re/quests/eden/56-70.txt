			set .@items,1; //incompleta
		else
			set .@items,2; //completa

		mes "Você tem uma missão em curso. Deseja verificar os detalhes?";
		next;

		mes "Estou me saindo muito bem na missão. Relatar agora?";
		next;

			erasequest .@quest_id;
			mes "Concluiu a missão com êxito. Receba a recompensa.";
