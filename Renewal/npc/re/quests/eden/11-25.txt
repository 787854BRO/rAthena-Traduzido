moc_para01,36,38,3	script	Missões [11 - 25]#Tuto	4_BOARD3,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Você precisa ter um -";
		mes "- ^4d4dff'Emblema do Éden'^000000 -";
		mes "- para receber essas missões. -";
		close;
	}
	for ( .@i = 11124; .@i < 11134; .@i++ )
		if (checkquest(.@i,PLAYTIME) == 2) erasequest .@i;
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			mes "- Você deve recolher sua -";
			mes "- recompensa antes de começar -";
			mes "- uma nova missão. -";
			mes "";
			mes "- Gerente: Spike -";
			close;
		}
		else if (.@j >= 0) {
			mes "- Você só pode solicitar -";
			mes "- uma missão de cada vez. -";
			mes " ";
			mes "- Gerente: Spike -";
			close;
		}
	}
	if (BaseLevel < 11) {
		mes "- Não há tarefas -";
		mes "- disponíveis para você agora. ";
		close;
	}
	if (BaseLevel > 25) {
		mes "- Essas missões são fáceis -";
		mes "- demais para o seu calibre, -";
		mes "- procure por missões de -";
		mes "- níveis mais apropriados para você. ";
		close;
	}
	mes "- Quadro de Avisos de missões -";
	mes "- para iniciantes. -";
	mes " ";
	mes "- Gerente: Spike -";
	next;
	switch(select("Caçar Zangão:Caçar Condor:Caçar Rockers:Caçar Rabo de Verme:Caçar Esporo:Caçar Ovo de Besouro-Ladrão:Caçar Muka:Caçar Familiar:Caçar Peco-Peco:Caçar Esporo Venenoso:Cancelar")) {
		case 1: callsub L_Quest,11114,"Caçar Zangão"," - Cace 10 ^4d4dff'Zangões'^000000 - "," - Noroeste de Prontera. - ";
		case 2: callsub L_Quest,11115,"Caçar Condor"," - Cace 20 ^4d4dff'Condors'^000000 - "," - Sul de Morroc. - ";
		case 3: callsub L_Quest,11116,"Caçar Rockers"," - Cace 10 ^4d4dff'Rockers'^000000 - "," - Sudoeste de Prontera e - "," - traga-me 10 ^4d4dffPernas de Gafanhoto^000000. - ";
		case 4: callsub L_Quest,11117,"Caçar Rabo de Verme"," - Cace 20 ^4d4dff'Rabos de Verme'^000000 - "," - Sudoeste de Payon. - ";
		case 5: callsub L_Quest,11118,"Caçar Esporo"," - Cace 30 ^4d4dff'Esporos'^000000 - "," - Ao redor de Payon. - ";
		case 6: callsub L_Quest,11119,"Caçar Ovo de Besouro-Ladrão"," - Cace 20 ^4d4dffOvos de Besouro-Ladrão^000000 - "," - no esgoto de Prontera - "," - e traga-me 10 ^4d4dffChrysalis^000000. - ";
		case 7: callsub L_Quest,11120,"Caçar Muka"," - Cace 20 ^4d4dff'Mukas'^000000 - "," - Sul de Prontera. - ";
		case 8: callsub L_Quest,11121,"Caçar Familiar"," - Cace 20 ^4d4dffFamiliares^000000 - "," - nos esgotos - "," - Oeste de Prontera. - ";
		case 9: callsub L_Quest,11122,"Caçar Peco-Peco"," - Cace 30 ^4d4dff'Peco-Pecos'^000000 - "," - Sul de Prontera. - ";
		case 10: callsub L_Quest,11123,"Caçar Esporo Venenoso"," - Cace 30 ^4d4dffEsporos Venenosos^000000 - "," - dentro da Torre de Geffen e - "," - Traga-me 5 ^4d4dffEsporos de Venenosos^000000. - ";
		case 11:
			mes "- Pare de ler o quadro de avisos.";
			close;
	}

L_Quest:
	.@quest_status = checkquest(getarg(0)+10,PLAYTIME);
	if (.@quest_status == 0 || .@quest_status == 1) {
		mes "- A Missão ^4d4dff"+ getarg(1) +"^000000 -";
		mes "- já está finalizada por hoje. ";
		mes "";
		mes "- Gerente: Spike -";
		close;
	}
	mes "- Missão '"+ getarg(1) +"'-";
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	mes "";
	mes "- Gerente: Spike -";
	next;
	if (select("Eu farei esta missão.:Procurar outras missões.") == 2) {
		mes "- Vamos procurar -";
		mes "- outras missões. ";
		close;
	}
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	setquest getarg(0);
	close;
}

moc_para01,32,30,6	script	Spike	4_F_CHILD,{
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			switch(.@i) {
				case 11114: callsub L_Quest,.@i,"Caçar Zangão",900;
				case 11115: callsub L_Quest,.@i,"Caçar Condor",1800;
				case 11116: callsub L_Quest,.@i,"Caçar Rockers",1400,940,10;
				case 11117: callsub L_Quest,.@i,"Caçar Rabo de Verme",2600;
				case 11118: callsub L_Quest,.@i,"Caçar Esporo",3900;
				case 11119: callsub L_Quest,.@i,"Caçar Ovo de Besouro-Ladrão",2900,915,10;
				case 11120: callsub L_Quest,.@i,"Caçar Muka",3200;
				case 11121: callsub L_Quest,.@i,"Caçar Familiar",3200;
				case 11122: callsub L_Quest,.@i,"Caçar Peco-Peco",5400;
				case 11123: callsub L_Quest,.@i,"Caçar Esporo Venenoso",6000,7033,5;
			}
		}
		else if (.@j >= 0) {
			mes "[Spike]";
			mes "Você ainda não terminou sua missão.";
			mes "Continue com o bom trabalho. Esta quase terminada.";
			close;
		}
	}
	mes "[Spike]";
	mes "Você está procurando um emprego?";
	mes "Sempre temos muito trabalho a fazer.";
	next;
	mes "[Spike]";
	mes "Por que você não olha em volta e encontra algumas missões interessantes?";
	close;

L_Quest:
	.@quest_id = getarg(0);
	if (checkquest(.@quest_id,HUNTING) == 2) {
		.@item_req = getarg(3,0);
		.@req_amount = getarg(4,0);
		if (.@item_req) {
			if (countitem(.@item_req) < .@req_amount) {
				mes "[Spike]";
				mes "Você não tem "+ getitemname(.@item_req) +" o suficiente.";
				mes "Você precisa me trazer "+ .@req_amount +" "+ getitemname(.@item_req) +".";
				close;
			}
		}
		mes "[Spike]";
		mes "Hmm... deixe-me ver~";
		mes "Você completou a missão de '"+ getarg(1) +"'.";
		next;
		mes "[Spike]";
		mes "Bom trabalho~!!";
		mes "Nosso cliente está muito feliz com seu trabalho. Ele apreciou muito o que você fez.";
		mes "Por favor, ajude-nos novamente. Obrigado!";
		if (.@item_req)
			delitem .@item_req,.@req_amount;
		erasequest .@quest_id;
		setquest .@quest_id+10;
		getexp getarg(2),0;
		close;
	}
}
