malangdo,220,167,5	script	Trocador de Moeda CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "Você tem muitos tipos de objetos. Vamos continuar depois que você reduzir esses objetos.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Não podemos continuar porque você tem muitos objetos pesados. Vamos tentar continuar depois que você reduzir o peso.";
		close;
	}
	mes "[Trocador de Moeda CX-1]";
	mes "Bem-vind"+(Sex?"o":"a")+" ao CX-1. Se você tiver moedas, você poderá trocá-las por moedas de alta ou baixa qualidade.";
	next;
	setarray .@coins[1],
		6417, //Silvervine  -> Fruta dos Gatos Madura
		6418, //Agrade_Coin -> Moeda Classe A
		6419, //Bgrade_Coin -> Moeda Classe B
		6420, //Cgrade_Coin -> Moeda Classe C
		6421, //Dgrade_Coin -> Moeda Classe D
		6422; //Egrade_Coin -> Moeda Classe E
	.@menu$ = "Parar:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+" (nenhuma)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[Trocador de Moeda CX-1]";
		mes "Obrigado por ter vindo.";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[Trocador de Moeda CX-1]";
		mes "Você escolheu uma moeda que você não tem.";
		mes "Obrigado por ter vindo.";
		close;
	}
	mes "[Trocador de Moeda CX-1]";
	mes "------ Taxa de Troca ------";
	mes "Maior Troca - 30 para 10";
	mes "Menor Troca - 10 para 30";
	mes "------------------------------";
	mes "^ff3333Quando você usa o serviço de troca, cobramos uma taxa de acordo com o preço atual de mercado.^000000";
	next;
	setarray .@exchange_name$[0],"maior","menor";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Valor deduzido por troca.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0; // Não é possível trocar por Fruta dos Gatos Madura.
	.@menu$ = "Parar:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff3333Troca disponível por moeda "+.@exchange_name$[.@i]+"^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaaTroca por moeda "+.@exchange_name$[.@i]+" (vazio)^000000:";
		else
			.@menu$ = .@menu$ + "Troca por moeda "+.@exchange_name$[.@i]+" - "+getitemname(.@coin)+" ("+.@exchange_rate[.@i]+" necessário):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[Trocador de Moeda CX-1]";
		mes "Obrigado por ter vindo.";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[Trocador de Moeda CX-1]";
		mes "Esta moeda não está disponível para trocar por moeda de nível "+.@exchange_name$[.@i]+".";
		mes "Obrigado por ter vindo.";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[Trocador de Moeda CX-1]";
		mes "Você não tem moedas suficientes para trocar por moeda de nível "+.@exchange_name$[.@i]+".";
		mes "Obrigado por ter vindo.";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[Trocador de Moeda CX-1]";
	mes "^003399"+.@exchange_rate[.@i]+"^000000 unidade(s) de ^003399"+getitemname(.@coin)+"^000000 foram tomadas.";
	mes "Eu as troquei pela ^003399moeda "+.@exchange_name$[.@i]+"^000000, depois de deduzir ^003399"+.@exchange_loss[(!.@i)]+"^000000 unidade(s), ^003399"+.@exchange_total+"^000000 unidade(s) foram obtidas.";
	close;
}
