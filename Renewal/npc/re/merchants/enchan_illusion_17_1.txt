		for (.@i = 0; .@i < 4; .@i++)
			.@card[.@i] = getequipcardid(.@part,.@i);
		copyarray .@check[0],.@card[0],4;
		if (!.@equip_id || .@id[0] != .@equip_id && .@id[1] != .@equip_id) {
			mes "[RS26]";
			mes "O equipamento precisa ser verificado. Esse item não é compatível com este dispositivo.";
		close;
	}
	.@var$ = module_check(.@module_id[.@s],.@part,.@index);
	explode(.@s$,.@var$,",");
	switch (atoi(.@s$[0])) {
		case 1:
			mes "[RS26]";
			mes "O módulo precisa ser verificado. Este dispositivo não consegue detectá-lo.";
			close;
		case 2:
			mes "[RS26]";
			mes "Parece que o equipamento que você está tentando aprimorar já atingiu o limite.";
			close;
		case 3:
			mes "[RS26]";
			mes "Você precisa ter um módulo compatível com a eficácia necessária para continuar com o encantamento.";
			close;
		case 4:
			mes "[RS26]";
			mes "Com este módulo você poderá encantar até " + .@s$[1] + " " + (atoi(.@s$[1]) > 1 ? "vezes" : "vez") + ". Parece que o limite já foi alcançado.";
			close;
		case 5:
			.@item_name$ = getitemname(.@module_id[.@s]);
			.@item_name$ = replacestr(.@item_name$,"Módulo de Modificação (","");
			.@item_name$ = replacestr(.@item_name$,")","");
			mes "[RS26]";
			mes "^0000FFVocê selecionou " + getitemname(.@equip_id) + " " + .@item_name$ + "^000000";
			mes "------------------";
			mes "Com o módulo selecionado você poderá encantar até ^0000FF" + .@s$[2] + " " +  (atoi(.@s$[2]) > 1 ? "vezes" : "vez") +  "^000000 neste equipamento. Atualmente ele está encantado ^0000FF" + .@s$[1] + "^000000 vez(es) com este módulo.";
			mes "Deseja continuar?";
			next;
			if (select("Vou pensar melhor...:Vamos prosseguir.") == 1) {
				mes "[RS26]";
				mes "Então, volte a me procurar se precisar de ajuda~";
				close;
			}
			for (.@i = 1; .@i < 4; .@i++) {
				if (.@card[.@i] == 0) {
					.@augment_slot = .@i;
					break;
				}
			}
			.@card[.@augment_slot] = atoi(.@s$[3]);
			delitem .@module_id[.@s],1;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				end;
			specialeffect2 EF_REPAIRWEAPON;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			mes "[RS26]";
			mes "Tentando aprimorar o slot ^990000" + .@augment_slot + "^000000 do equipamento.";
			close;
	}
}

function	module_check	{
	if(!countitem(getarg(0)))
		return "1";
	for (.@i = 0; .@i < 4; .@i++)
		.@card[.@i] = getequipcardid(getarg(1),.@i);
	if(.@card[3]) return "2";
	explode(.@T$,.module$[getarg(2)],",");
	if(getarg(0) >= 25700 && getarg(0) <= 25705){
		explode(.@TT$,.module$[4],",");
		for (.@i = 0; .@i < getarraysize(.@TT$); .@i++) {
			explode(.@module$,.@TT$[.@i],":");
			if(getarg(0) == atoi(.@module$[0]))
				.@enchant_id = atoi(.@module$[1]);
			.@module = atoi(.@module$[0]);
			.@enchant = atoi(.@module$[1]);
			for (.@j = 1; .@j < 4; .@j++) {
				if (.@card[.@j] == .@enchant)
					return "4,1";
			}
		}
		.@max = 1;
	} else {
		for (.@i = 0; .@i < getarraysize(.@T$); .@i++) {
			explode(.@module$,.@T$[.@i],":");
			if(getarg(0) == atoi(.@module$[0])){
				.@enchant_id = atoi(.@module$[1]);
				.@max = atoi(.@module$[2]);
				break;
			}
		}
	}
	if(!.@enchant_id)
		return "3";
	for (.@j = 1; .@j < 4; .@j++) {
		if (.@card[.@j] == .@enchant_id)
			.@count++;
	}
	if (.@count == .@max)
		return "4," + .@max;
	return "5," + .@count + "," + .@max + "," + .@enchant_id;
	}
}
