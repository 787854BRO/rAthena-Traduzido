-	script	ZondaTeleportationServices	HIDDEN_NPC,{
	cutin "zonda_do01", 2;
	mes "[Funcionária Zonda]";
	mes "Bem-vind"+((Sex)?"o":"a")+".";
	mes "Nossa equipe está sempre trabalhando para superar suas expectativas em relação à qualidade do nosso serviço.";
	mes "Por favor, escolha o seu destino.";
	next;
	switch( select( "Norte da Bifrost", "Eclage", "Esplendor", "Acampamento da Expedição Midgard", "Manuk", "Arredores da Montanha Kamidal", "Cancelar.") ) {
	case 1:
		callsub( S_Zone, "Norte da Bifrost", 17000,850, 7213 ); // Valor de teletransporte não confirmado
		warp "bif_fild02",160,230; // Coordenadas não verificadas
		end;
	case 2:
		callsub( S_Zone, "Eclage", 17000,850, 11311 );
		warp "eclage",111,39;
		end;
	case 3:
		callsub( S_Zone, "Esplendor", 17000,850, 10078 );
		warp "splendide",171,115;
		end;
	case 4:
		callsub( S_Zone, "Acampamento da Expedição Midgard", 19000,950, 10078 );
		warp "mid_camp",210,288;
		end;
	case 5:
		callsub( S_Zone, "Manuk", 22000,1100, 10078 );
		warp "manuk",263,177;
		end;
	case 6:
		callsub( S_Zone, "Arredores da Montanha Kamidal", 24000,1200, 7184 );
		warp "dic_fild01",188,187;
		end;
	case 7:
		mes "[Funcionária Zonda]";
		mes "Estamos sempre nos esforçando para garantir os melhores serviços aos nossos clientes.";
		mes "Obrigado por utilizar nossos serviços.";
		close3;
	}

S_Zone:
	.@zeny_req = getarg(1);
	.@item_req = getarg(2);
	.@quest_req = getarg(3);
	
	if (isbegin_quest(.@quest_req) == 0) {
		mes "[Funcionária Zonda]";
		//mes getarg(0);
		mes "Você deve completar a missão de acesso para poder se teletransportar para "+getarg(0)+" antes de usar nossos serviços de teletransporte.";
		close3;
	}
	mes "[Funcionária Zonda]";
	mes getarg(0) + " está disponível por ^ff0000"+.@zeny_req +" Zenys^000000 ou ^ff0000"+.@item_req+" Peixes Enlatados de Malangdo^000000.";
	next;
	switch( select( .@zeny_req + "Zeny", .@item_req + "Peixes Enlatados de Malangdo", "Cancelar" ) ) {
	case 1:
		if (Zeny < .@zeny_req)
			break;
		Zeny -= .@zeny_req;
		return;
	case 2:
		if (countitem(12636) < .@item_req)
			break;
		delitem 12636,.@item_req; // Peixes Enlatados de Malangdo
		return;
	case 3:
		mes "[Funcionária Zonda]";
		mes "Você pode me perguntar sobre outras áreas de serviço?";
		close3;
	}
	mes "[Funcionária Zonda]";
	mes "Senhor! ";
	mes "Verifique novamente.";
	close3;
}
