	}
	if (isbegin_quest(13181) != 1) {

		close3;

	case 0:

		mes "^ff0000Este portal dimensional não pode ser acessado por 20 horas após seu último acesso.^000000";
		close3;
	case 2:
		mes "[Runane]";

		close3;
	}

		.@menu$ = "Abrir o portal dimensional.";
	switch( select( .@menu$, "Entrar no " + .@md_name$ + ".", "Cancelar." ) ) {
	case 1:
		if (instance_create(.@md_name$) < 0) {
			mes "Nome do Grupo: " + getpartyname( getcharid(1) );
			mes "Líder do Grupo: " + strcharinfo(0);
			mes "^0000ff" + .@md_name$ + " ^000000- Falha na Reserva!";
			close3;
		}
		mes "[Runane]";
		mes "^ff0000Preparando o portal dimensional...^000000";
		mes "^ff0000Por favor, aguarde.^000000";
		close3;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			close3;
		case IE_NOINSTANCE:
			if (is_party_leader() == false) {
				mes "[Runane]";
				mes "Teleportando...";
				next;
				mes "[Runane]";
				mes "^ff0000Por favor, note que este portal não pode ser acessado por 23 horas após seu último acesso.^000000";
				next;
			}
			mes "[Runane]";
			mes "^ff0000O líder do seu grupo ainda não criou uma Fenda Temporal.^000000";
			close3;
		case IE_OTHER:
			mes "Ocorreu um erro desconhecido.";
			close3;
		case IE_OK:
			setquest 13182;// Pesadelo de Requiém: Esperando
			mapannounce "moc_para01", "" + strcharinfo(0) + " do grupo, " + getpartyname( getcharid(1) ) + ", está entrando no " + .@md_name$ + ".", bc_map, 0xFF99;
			specialeffect2 EF_READYPORTAL2;
			specialeffect2 EF_PORTAL2;
			// warp "1@jtb",16,17;
			end;
		}
	case 3:
		cutin "",255;
		end;
	}
}
