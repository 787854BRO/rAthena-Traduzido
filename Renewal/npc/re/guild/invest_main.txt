-	script	#invest_timer	-1,{
OnClock0000:	// Abri os investimentos em (1 hora após GdE)
	if (gettime(DT_DAYOFWEEK) == WEDNESDAY && !agitcheck()) {
		set $2011_agit_invest,1;
		donpcevent "#fund_master::OnInvest_start";
	}
	end;
OnClock1200:	// Fecha os investimentos na Fri (60 horas após abertura dos investimentos)
	if (gettime(DT_DAYOFWEEK) == FRIDAY && !agitcheck()) {
		set $2011_agit_invest,2;
		donpcevent "#fund_master::OnInvest_stop";
	}
	end;
OnClock1235:	// Abre as masmorras em Fri (pelo menos 31 minutos após os investimentos fecharem)
	if (gettime(DT_DAYOFWEEK) == FRIDAY && !agitcheck())
		donpcevent "#fund_master::OnResult";
	end;
OnClock2000:	// Fechar as masmorras às terças (1 hora antes da GdE)
	if (gettime(DT_DAYOFWEEK) == TUESDAY)
		donpcevent "#fund_master::OnReset";
	end;
}

function	script	F_Invest_Status	{
	set .@day, gettime(DT_DAYOFWEEK);
	set .@hour, gettime(DT_HOUR);
	set .@minute, gettime(DT_MINUTE);

	// Inativo.
	if (agitcheck())
		return 0;

	// Aberto para investimentos.
	if (.@day >= WEDNESDAY && (.@day < FRIDAY || (.@day == FRIDAY && .@hour <= 12)))
		return 1;

	// Investimentos fechados, calculando resultados.
	if (.@day == FRIDAY && .@hour == 12 && .@minute < 35)
		return 2;

	// Cálculos completos, masmorras abertas.
	if ((.@day == FRIDAY && (.@hour > 12 || (.@hour == 12 && .@minute >= 35))) || .@day > FRIDAY ||
	    .@day < TUESDAY || (.@day == TUESDAY && .@hour < 20))
		return 3;

	// Masmorras fechadas.
	return 0;
}

		case 1:
			set .@units,1;
			break;
		case 2:
			set .@units,10;
			break;
		case 3:
			set .@units,50;
			break;

			mes "Por favor, digite 0 para cancelar.";

				mes "Você excedeu o valor de investimentos.";
				mes "Por favor, tente novamente.";
				close;
			} else if (.@units == 0) {

				close;
			} else
				break;
		case 5:

			mes "Cancelado!";

		if (Zeny < .@zeny) {

			mes "Você não tem dinheiro suficiente.";
			close;

		mes .@npc_name$;
		mes "Você investiu "+callfunc("F_InsertPlural",.@units,"unit")+" "+callfunc("F_InsertComma",.@zeny)+" Zeny(s)"+getarg(2,"")+".";
		mes "Não é muito, mas este é um pequeno presente. Você pode trocá-lo por itens em Malangdo.";
		set Zeny, Zeny-.@zeny;
		setd getarg(0), getd(getarg(0)) + .@units;
		getitem 6488,.@units; //Thanks_Invest_Ticket
		close;
	case 2:
		mes .@npc_name$;
		mes "Uau, você trouxe Certificados de Investimento.";
		mes "Quantos certificados você vai investir?";
		next;
		switch(select("1 Certificado de Investimento:5 Certificados de Investimento:Outra Quantia:Cancelar")) {
		case 1:
			set .@tickets,1;
			break;
		case 2:
			set .@tickets,5;
			break;
		case 3:
			mes .@npc_name$;
			mes "Por favor, informe o número de Certificados de Investimento que você deseja investir.";
			mes "Por favor, escolha entre 1 e 20.";
			mes "Por favor, digite 0 para cancelar.";
			next;
			input .@tickets;
			if (.@tickets < 0 || .@tickets > 20) {
				mes .@npc_name$;
				mes "Você excedeu o valor de investimentos.";
				mes "Por favor, tente novamente.";
				close;
			} else if (.@tickets == 0) {
				mes .@npc_name$;
				mes "Cancelado!";
				close;
			} else
				break;
		case 4:
			mes .@npc_name$;
			mes "Cancelado!";
			close;
		}
		if (countitem(6489) < .@tickets) {
			mes .@npc_name$;
			mes "Parece que você tem menos Certificados de Investimento do que você deseja investir.";
			close;
		}
		set .@units, .@tickets * 10;
		mes .@npc_name$;
		mes .@tickets+" Certificado(s) de Investimento"+((.@tickets == 1)?" é":" são")+" o mesmo que um investimento de "+.@units+" unidade(s) ("+((.@units < 100)?.@units:insertchar(""+.@units,",",1))+"0,000 Zenys)"+getarg(2,"")+".";

		setd getarg(0), getd(getarg(0)) + .@units;
		getitem 6488,.@units; //Thanks_Invest_Ticket
		close;

		mes .@npc_name$;
		mes "Cancelado!";
		close;
	}
}
