//Assassina Inútil;
yuno_in01,22,162,6	script	Assassina	725,{
	mes "[Sharlet]";
	if (Event_THQS > 0) goto N_Member;
	mes "Hum, me desculpe, mas eu não vou perder meu tempo falando com você se você nem sequer é um membro!";
	close;
N_Member:
	set @TEMP,rand(1,2);
	if (@TEMP == 1) goto N_Chat1;
	if (@TEMP == 2) goto N_Chat2;
	mes "Como você está lendo isso???";
	mes "Bem, a menos que você esteja lendo o código :)";
	close;
N_Chat1:
	mes "Ahh, Bem-vind"+(Sex?"o":"a")+" companheiro Caçador de Tesouros "+strcharinfo(0)+".";
	mes "Você ouviu falar sobre aquele homem corajoso que entrou no Glast Heim sozinho?! Ele deve estar louco.";
	close;
N_Chat2:
	mes ""+strcharinfo(0)+" você não tem algo melhor para faser? Como uma missão.";
	close;
}
