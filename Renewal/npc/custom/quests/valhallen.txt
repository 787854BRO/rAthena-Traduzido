prontera,147,171,5	script	Lenneth	811,{
	mes "[Lenneth]";
	mes "Eu venho de Valhalla para ajudar você.";
	mes "Eu posso transmutar para você muitos dos itens do Valhallen,";
	mes "embora eles exijam muitos elementos raros para fazer.";
	next;
	switch(select("Mjolnir:Sleipnir:Brisingamen:Gleipnir:Megingjard:Manto de Asfrika:Armadura de Brynhildr:Hervor:Jormungand:Muspellium:Essência de Runa:Sair")) {
		case 1:
			mes "[Lenneth]";
			mes "^3355FFMjolnir^000000 é o poderoso Martelo de Thor.";
			setarray .@Items[0],1530,1522,2,984,20,985,5,969,40,7074,2,7075,4,7078,5,7091,5,7089,5;
			break;
		case 2:
			mes "[Lenneth]";
			mes "^3355FFSleipnir^000000 são botas feitas a partir do Cavalo de Guerra de Odin.";
			setarray .@Items[0],2410,2406,2,969,20,985,10,7076,3,7079,5,7083,3,7086,4,7087,3;
			break;
		case 3:
			mes "[Lenneth]";
			mes "^3355FFBrisingammen^000000 é o Colar mágico de Freyja, deusa da Beleza.";
			setarray .@Items[0],2630,2603,1,726,2,722,3,727,10,724,5,969,20,7073,4,7077,4,7088,3,7090,3,7092,3;
			break;
		case 4:
			mes "[Lenneth]";
			mes "O ^3355FFGleipnir^000000 é uma corda leve, porém forte, necessária para fazer ^3355FFMegingjard^000000.";
			setarray .@Items[0],7058,7080,4,7081,5,7082,4,7084,3,7085,3;
			break;
		case 5:
			mes "[Lenneth]";
			mes "O ^3355FFMegingjard^000000 é o poderoso Cinturão de Thor.";
			setarray .@Items[0],2629,7058,1,2627,1,969,20,726,10,984,10;
			break;
		case 6:
			mes "[Lenneth]";
			mes "A ^3355FFManto de Asfrika^000000 é a luz dos deuses.";
			setarray .@Items[0],2541,7835,1,7836,1,7837,1,7838,1,2513,1,7063,100,7291,10,7293,10,985,20;
			break;
		case 7:
			mes "[Lenneth]";
			mes "A ^3355FFArmadura de Brynhildr^000000 é a armadura da guerreira.";
			setarray .@Items[0],2383,7830,1,7831,1,7832,1,7833,1,7834,1,7510,100,2357,1,969,10,985,20;
			break;
		case 8:
			mes "[Lenneth]";
			mes "O ^3355FFHervor^000000 tem o nome da valquíria que garante a vitória de seus guerreiros.";
			setarray .@Items[0],2181,6595,2,6596,1,6594,4,6597,3,6602,1,6604,1,6605,1,2115,1;
			break;
		case 9:
			mes "[Lenneth]";
			mes "O ^3355FFJormungand^000000 é famoso por sua habilidade de envenenar o inimigo.";
			setarray .@Items[0],2020,6603,4,6599,1,6598,1,6601,4,6600,1,6604,1,6605,1,1473,1;
			break;
		case 10:
			mes "[Lenneth]";
			mes "O ^3355FFMuspellium^000000 é necessário para fazer o ^3355FFJormungand^000000 e o ^3355FFHervor^000000.";
			setarray .@Items[0],6605,984,50,985,50;
			break;
		case 11:
			mes "[Lenneth]";
			mes "A ^3355FFEssência de Runa^000000 é necessário para fazer o ^3355FFJormungand^000000 e o ^3355FFHervor^000000.";
			setarray .@Items[0],6604,12735,200,12736,100;
			break;
		default:
			close;
	}
	
	mes "^FF0000Os itens que eu preciso são os seguintes:^000000";
	set .@itemmissing,0;
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		mes " ~ "+.@amount+"x "+((countitem(.@id)<.@amount)?("^dd3355"):("^33dd55"))+getitemname(.@id)+((getitemslots(.@id)>0)?" ["+getitemslots(.@id)+"]":"")+"^000000";
		if (countitem(.@id) < .@amount) 
			set .@itemmissing,1;
	}
	if (.@itemmissing) {
		next;
		mes "[Lenneth]";
		mes "Vá em frente e encontre todos esses itens.";
		mes "Eu esperarei aqui para ajudá-lo.";
		close;
	}
	next;
	mes "[Lenneth]";
	mes "Você deseja o item ^3355FF"+getitemname(.@items[0])+"^000000?";
	next;
	if(select("Sim:Não")==2) {
		mes "[Lenneth]";
		mes "Por favor, retorne se você mudar de idéia.";
		close;
	}
	mes "[Lenneth]";
	mes "Aqui está o seu ^3355FF"+getitemname(.@items[0])+"^000000.";
	mes "Que te sirva bem.";
	//segunda verificação para que se o jogador tentar ignorar o primeira.
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		if (countitem(.@id) < .@amount) 
			close;
	}
	//Exclui os itens _após_ a segunda verificação.
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		delitem .@id, .@amount;
	}
	getitem .@items[0],1;
	close;
}
