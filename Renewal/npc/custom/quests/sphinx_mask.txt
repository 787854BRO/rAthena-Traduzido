morocc,208,90,6	script	Ladrão de Turbante	58,{
	mes "[Ladrão de Turbante]";
	if(event_umbala < 2) {
		mes "O que você quer comigo? vá embora!";
		close; }
	if(sphmask_q) {
		mes "Você não tem mais negócios comigo, vá embora!";
		close; }
	mes "Olá meu amigo, você se interessaria em obter uma máscara rara? Seu valor eu o asseguro é bem real meu amigo. Este item é magnífico!";
	next;
	callsub L_Menu,1,"1,000,000",1000000;
	callsub L_Menu,2,"750,000",750000;
	callsub L_Menu,3,"500,000",500000;
	next;
	mes "[Ladrão de Turbante]";
	mes "Ack! Esqueça isso! Eu posso conseguir a máscara em outro lugar!";
	set sphmask_q,1;
	close;

L_Menu:
	mes "[Ladrão de Turbante]";
	switch(getarg(0)) {
		case 1: mes "Wah? O ho ho ho, então você sabe que eu roubo disem daqueles selvagens? Bem, eu não vou devolver nada. Mas eu vou por um preço... então o que me diz?"; break;
		case 2: mes "Ahh você não é um homem de negócios? Tudo bem, que tal..."; break;
		case 3: mes "Hmmm... você quer barganhar pagando apenas isso, ok... minha oferta final..."; break; }
	next;
	if(select("Pagar "+getarg(1)+"z.:Sem acordo.")==2) return;
	mes "[Ladrão de Turbante]";
	if (Zeny < getarg(2)) {
		mes "Você está brincando comigo? Você não tem dinheiro!";
		close; }
	set Zeny, Zeny-getarg(2);
	mes "Ah ho, é um negócio, então!";
	getitem 7114,1;
	set sphmask_q,1;
	close;
}
