// Verifica se o jogador tem o nível requerido.
// fecha se não, continua se sim.
function	Require_Level	{
	if (BaseLevel < getarg(0) || JobLevel < getarg(1)) {
		.@blvl = getarg(0) - BaseLevel;
		.@jlvl = getarg(1) - JobLevel;
		mes "Nível Requerido:";
		mes ((getarg(0)>1)? 
			"^bb0000"+getarg(0)+"^000000 (^bb0000Base^000000) / ":"")+"^00bb00"+
			getarg(1)+"^000000 (^00bb00Classe^000000)";
		mes "Você precisa de mais " +
			((.@blvl > 0) ? "^bb0000"+.@blvl+"^000000 níveis de Base " + 
				((.@jlvl > 0) ? "e " : "") : "") +
			((.@jlvl > 0) ? "^00bb00"+.@jlvl+"^000000 níveis de Classe " : "") +
			"para continuar.";
		close;
	}
	return;
}

// Verifica se o dado eac é uma classe Mini
function Is_Baby	{
	return ((getarg(0, eaclass())&EAJL_BABY)>0);
}

// Verifica se o jogador pode mudar para a terceira classe.
// Nota: Isto não inclui as verificações de nível.
function	Can_Change_Fourth	{
	// Para mudar para a quarta classe você precisará estar:
    // * Na Terceira Classe Transcendental
	if( !.FourthClass )
		return false; // Fourth job change disabled
	if( (eaclass()&(EAJL_THIRD|EAJL_UPPER)) != (EAJL_THIRD|EAJL_UPPER) )
		return false; // Terceira classe não transcendental
	if( eaclass()&EAJL_FOURTH )
		return false; // Já é quarta classe
	if( roclass(eaclass()|EAJL_FOURTH) < 0 )
		return false; // Quarta classe não tem JOB
	return true;
}

// Verifica se o jogador já pode mudar para terceira classe.
// Nota: Isto não inclui as verificações de nível.
function	Can_Change_Third	{
	// Para mudar para terceira classe, você precisa ser:
	// * Segundas Classes
	// * Segundas Classes Transcendentais
	// * Segundas Classes Mini
	if( !.ThirdClass )
		return false; // Troca para Terceira Classe Desativada
	if( !(eaclass()&EAJL_2) )
		return false; // Não é segunda classe
	if( eaclass()&EAJL_THIRD )
		return false; // É terceira classe
	if( roclass(eaclass()|EAJL_THIRD) < 0 )
		return false; // Não é terceira classe
	if( (eaclass()&EAJ_UPPERMASK) == EAJ_SUPER_NOVICE )
		return false; // Exp. Super Aprendiz é igual a 3° Cls, mas tem seu próprio case
	if( Is_Baby() && (!.BabyClass || !.BabyThird) )
		return false; // Não é Mini (Terceira) mudança permitida
	return true;
}

function	Can_Rebirth	{
	// Para renascer, você precisa ser:
	// * Segunda Classe
	if( !.RebirthClass )
		return false; // Renascimento desativado
	if( !(eaclass()&EAJL_2) )
		return false; // Não é segunda classe
	if( eaclass()&(EAJL_UPPER|EAJL_THIRD) )
		return false; // Já renasceu/ Terceira Classe
	if( roclass(eaclass()|EAJL_UPPER) < 0 )
		return false; // A classe não é transcedental
	if( Is_Baby() && !.BabyClass )
		return false; // Nenhuma mudança para mini permitida
	return true;
}

// Verifica se o dado eac é primeira classe
function	Is_First_Cls	{
	return (getarg(0) <= EAJ_TAEKWON);
}

function	Check_Riding	{
	// Nota: Porque devemos sempre checar as Montarias:
	// As montarias são consideradas como outra classe, e isso é o que
	// torna esse NPC útil, apenas para lidar com
	// esses casos especiais.
	if (checkfalcon() || checkcart() || checkriding() || ismounting()) {
		mes "Por favor, remova " +
			((checkfalcon()) ? "seu Falcão" : "") +
			((checkcart()) ? "seu Carrinho" : "") +
			((checkriding()) ? "seu Peco Peco" : "") +
			((ismounting()) ? "sua Montaria" : "") +
			" antes de continuar.";
		close;
	}
	return;
}
function	Check_SkillPoints	{
	if (.SkillPointCheck && SkillPoint) {
		mes "Por favor, use todos os seus pontos de habilidade antes de prosseguir.";
		close;
	}
	return;
}

// addJobOptions é essencialmente o mesmo
// setarray .@array[getarraysize(.@array)],opt1,opt2,...;
// É apenas mais fácil de ler, já que estamos usando isso com muita frequência
function	Job_Options	{
	.@argcount = getargcount();
	.@arr_size = getarraysize(getarg(0));
	for( .@i = 1; .@i < .@argcount; .@i++) {
		setarray getelementofarray(getarg(0), .@arr_size++),getarg(.@i);
	}
}

// Início do NPC
	mes .NPCName$;
	Check_Riding();
	Check_SkillPoints();

	// Inicialização
	.@eac = eaclass();
	.@fourth_possible = Can_Change_Fourth();
	.@third_possible = Can_Change_Third();
	.@rebirth_possible = Can_Rebirth();

	// BaseClass = roclass(.@eac&EAJ_BASEMASK) são os jogadores da Primeira Classe
	// BaseJob = roclass(.@eac&EAJ_UPPERMASK) são os jogadores da Segunda Classe
	//dispbottom "Debug: eac ("+.@eac+"), terceira ("+.@third_possible+"), rebirth("+.@rebirth_possible+"), BaseClass ("+BaseClass+"), BaseJob ("+BaseJob+")";
		
	// A partir daqui, a Mestra das Classes verifica a classe atual
	// e preenche o array `.@job_opt` com possíveis
	// opções de classes para os jogadores.
	
	if( .@rebirth_possible ) {
		// Opção de Renascimento (exibido no topo do menu)
		Require_Level(.Req_Rebirth[0], .Req_Rebirth[1]);
		Job_Options(.@job_opt, Job_Novice_High);
	}
	if( .@third_possible ) {
		// Mudança para terceira classe (exibida após o renascimento)
		Require_Level(.Req_Third[0], .Req_Third[1]);
		Job_Options(.@job_opt, roclass(.@eac|EAJL_THIRD));
	}
	if( .@fourth_possible ) {

		Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
		Job_Options(.@job_opt, roclass(.@eac|EAJL_FOURTH));
	}
	 
	if (.SecondExpanded && 
	   (.@eac&EAJ_UPPERMASK) == EAJ_SUPER_NOVICE && // é Super Aprendiz
	   !(eaclass()&EAJL_THIRD) ) {				 	// ainda não expandido SN
		// (Mini) Super Aprendiz para Expansão de (Mini) Super Aprendiz
	   	if( !Is_Baby(.@eac) || (.BabyClass && .BabyExpanded) ) {
			// .BabyClass & .BabyExpanded irá ser ativado se o for um mini
			Require_Level(.Req_Exp_SNOVI[0], .Req_Exp_SNOVI[1]);
			Job_Options(.@job_opt,roclass(.@eac|EAJL_THIRD)); // Expandido SN se for "terceira" cls
		}
	}

		// (Mini) Justiceiro para (Mini) Rebelde
		if( !Is_Baby(.@eac) || (.BabyClass && .BabyExpanded) ) {

		}

			case Job_Novice_High:

						Job_Acolyte_High, Job_Merchant_High, Job_Thief_High);

		if(.LastJob && lastJob && (.@eac&EAJL_UPPER)) {

			if( .@class1 > 0 ) { // 2-1

				Job_Options(.@job_opt, .@class1);
			}
			if( .@class2 > 0 ) { // 2-2
				Require_Level(.Req_Second[0], .Req_Second[1]);
				Job_Options(.@job_opt, .@class2);
			}
		}
	}

// Exibe o menu de tarefas
function	Job_Menu	{
	// getarg(0) é a array .@job_opt que contém todas as mudanças de classes disponíveis.
	function Confirm_Change;
	while(true) {
		.@opt_cnt =  getarraysize(getarg(0));
		if( .@opt_cnt <= 0 ) {
			mes "Não há mais classes disponíveis.";
			close;
		}

		.@selected = 0; // Apenas uma única mudança de classe é dada, nenhuma seleção é necessária
		if (.@opt_cnt > 1) {
			// Várias classes são dadas. Selecione uma e a mesma será salva em. @Class
			// Depois disso confirma o .@class
			mes "Selecione uma classe:";
			.@menu$ = "";
			for (.@i = 0; .@i < .@opt_cnt; .@i++) {
				if( getelementofarray(getarg(0), .@i) == Job_Novice_High)
					.@jobname$ = "^0055FFRenascido^000000";
				else
					.@jobname$ = jobname(getelementofarray(getarg(0), .@i));
				.@menu$ = .@menu$ + " ~ " + .@jobname$ + ":";
			}
			.@menu$ = .@menu$+" ~ ^777777Cancelar^000000";
			.@selected = select(.@menu$) - 1;
			if( .@selected < 0 || .@selected >= .@opt_cnt )
				close;
			next;
			mes .NPCName$;
		}
		.@class = getelementofarray(getarg(0), .@selected);
		if ((.@class == Job_Super_Novice || .@class == Job_Super_Baby) &&
				BaseLevel < .SNovice) {
			// Requisito de Nível Especial porque Super Aprendiz e
			// Super Bebê pode ser selecionado em uma das primeiras mudanças
			// É por isso que o Requisito de Nível é antes e não depois
			// da seleção.
			mes "O Nível de Base de " + .SNovice +
				" é necessário para se transformar em um " + jobname(.@class) + ".";
			return;
		}
		// Confirma a classe
		Confirm_Change(.@class, .@opt_cnt > 1);
		next;
		mes .NPCName$;
	}
	return;
}

// Executa a mudança de classe atual e fecha.
function	Job_Change	{
	.@previous_class = Class;
	.@to_cls = getarg(0);
	next;
	mes .NPCName$;
	mes "Você agora é " + callfunc("F_InsertArticle", jobname(.@to_cls)) + "!";
	if (.@to_cls == Job_Novice_High && .LastJob)
		lastJob = Class; // Salva a última escolha para o renascimento
	jobchange .@to_cls;
	if (.@to_cls == Job_Novice_High)
		resetlvl(1);
	else if (.@to_cls == Job_Baby) {
		resetstatus;
		resetskill;
		set SkillPoint,0;
	}
	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;
	if (.@previous_class != Class) {
		if (.Platinum)
			callfunc "F_GetPlatinumSkills";
		if (.GetJobEquip)
			Get_Job_Equip();
	}
	close; // Sempre fecha após a mudança
}

function	Confirm_Change	{
	// O Jogador confirma que ele quer mudar para. .@class
	.@class = getarg(0, -1);
	.@back = getarg(1, false);
	if( .@class < 0 || eaclass(.@class) == -1 ) {
		mes "Erro de Classe Desconhecido.";
		close;
	}
	mes "Você tem certeza que quer mudar para a classe ^0055FF"+jobname(.@class)+"^000000?";
	.@job_option$ = " ~ Mudar para a classe ^0055FF"+jobname(.@class)+"^000000";
	if( .@class == Job_Novice_High)
		.@job_option$ = " ~ ^0055FFRenascer^000000";
	
	if (select(.@job_option$+": ~ ^777777" + 
			((.@back) ?"Voltar" : "Cancelar") + "^000000") == 1) {
		Job_Change(.@class);
	}
	if (!.@back)
		close; // "Cancelar" for pressionado
	return;
}
