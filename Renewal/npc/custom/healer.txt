-	script	Curandeira	-1,{

	.@Price = 100000;	// Zeny necessário para curar um jogador (Padrão: 0 Desativado)
	.@Buffs = 1;	    // Aplicar Buffs em jogadores? (1: Ativado / 0: Desativado)
	.@Delay = 5;	    // Aplicar tempo de espera(Em segundos) para usar o npc? (Padrão: 0 Desativado)

	if (@HD > gettimetick(2))
		end;
	if (.@Price) {
		message strcharinfo(0), "Para te curar terei que cobrar uma taxa de " + callfunc("F_InsertComma",.@Price) + " zenys.";
		if (Zeny < .@Price)
			end;
		if (select("^0055FFCurar HP/SP^000000:^777777Nada não, obrigado!^000000") == 2)
			end;
		Zeny -= .@Price;
	}
	specialeffect2 EF_HEAL2;
	percentheal 100,100;
	if (.@Buffs) {
		specialeffect2 EF_INCAGILITY;
		sc_start SC_INCREASEAGI,240000,10;
		specialeffect2 EF_BLESSING;
		sc_start SC_BLESSING,240000,10;
	}
	if (.@Delay)
		@HD = gettimetick(2) + .@Delay;
		
//=================== Essa parte foi adicionada pela Equipe de Desenvolvimento rAthena! ===================
	dispbottom "Curandeira : Você deve aguardar "+.@Delay+" segundo(s) para usar meus serviços novamente!";
//==========================================================================================================
		
	end;
}
