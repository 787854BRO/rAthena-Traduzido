//////////////////////////////////////////////////////////////////////////////////
// Retorna a string de nome do item expandido.
// -- callfunc "F_ItemName",<ID do item>,{<elemento>{,<VVS metros 0..3>{,<taxa de refino>}}}
//    Elementos: 0=nenhum, 1=Gelo, 2=Terra, 3=Fogo, 4=Vento
// Exemplo:
//    // Exibe "Mostre-me sua ^000090+5 VVS Faca de Fogo^000000..."
//    mes "Mostre-me sua "+callfunc("F_ItemName",1201,1,2,5)+"...";
//////////////////////////////////////////////////////////////////////////////////
function	script	F_ItemName	{
	if (getarg(3,0))
		set .@t$,"+"+getarg(3)+" ";
	switch (getarg(2,0)) {
		case 0: break;
		case 1: set .@t$,.@t$+"VS "; break;
		case 2: set .@t$,.@t$+"VVS "; break;
		case 3: set .@t$,.@t$+"VVVS "; break;
		default: set .@t$,.@t$+getarg(2)+"xVS "; break;
	}
	switch (getarg(1,0)) {
		case 0: break;
		case 1: set .@t$,.@t$+"Gelo "; break;
		case 2: set .@t$,.@t$+"Terra "; break;
		case 3: set .@t$,.@t$+"Fogo "; break;
		case 4: set .@t$,.@t$+"Vento "; break;
		default: set .@t$,.@t$+"Estranho "; break;
	}
	return "^000090"+.@t$+getitemname(getarg(0))+"^000000";
}

//////////////////////////////////////////////////////////////////////////////////
// Armazena habilidades de missão aprendidas como variáveis ADV_QSK e ADV_QSK2.
// -- callfunc "F_SaveQuestSkills"
//
// Restaura habilidades de missão de 1ª classe aprendidas.
// -- callfunc "F_Load1Skills"
//
// Restaura habilidades de missão de 2ª classe aprendidas.
// -- callfunc "F_Load2Skills"
//////////////////////////////////////////////////////////////////////////////////
function	script	F_SaveQuestSkills	{
	set ADV_QSK,0; set ADV_QSK2,0;
	//Habilidades de missão de 1ª classe
	for(set .@i, 0; .@i < 14; set .@i, .@i + 1){
		if(getskilllv(144+.@i)) set ADV_QSK,ADV_QSK|pow(2,.@i);
	}
	//Habilidades de missão de 2ª classe
	for(set .@i, 0; .@i < 19; set .@i, .@i + 1){
		if(getskilllv(1001+.@i)) set ADV_QSK2,ADV_QSK2|pow(2,.@i);
	}
	return;
}
function	script	F_Load1Skills	{
	for(set .@i, 0; .@i < 14; set .@i, .@i + 1){
		if(ADV_QSK|pow(2,.@i) == ADV_QSK) skill 144+.@i,1,0;
	}
	set ADV_QSK,0; //Limpa a variável
	return;
}
function	script	F_Load2Skills	{
	for(set .@i, 0; .@i < 19; set .@i, .@i + 1){
		if(ADV_QSK2|pow(2,.@i) == ADV_QSK2) skill 1001+.@i,1,0;
	}
	set ADV_QSK2,0; //Limpa a variável
	return;
}

//////////////////////////////////////////////////////////////////////////////////
// Retorna o tempo restante em um formato legível.
// O argumento é subtraído do tempo de época UNIX, ou 'gettimetick(2)'.
// -- callfunc "Time2Str",<tick no tempo da época UNIX>
// Exemplo:
//    // Exibe "Tempo restante: 1 dia, 1 hora, 8 minutos, 20 segundos"
//    set .@Timer, gettimetick(2) + 90500;
//    mes "Tempo restante: " + callfunc("Time2Str",.@Timer);
//////////////////////////////////////////////////////////////////////////////////
function	script	Time2Str	{
	set .@Time_Left, getarg(0) - gettimetick(2);
	
	set .@Days, .@Time_Left / 86400;
	set .@Time_Left, .@Time_Left - (.@Days * 86400);
	set .@Hours, .@Time_Left / 3600;
	set .@Time_Left, .@Time_Left - (.@Hours * 3600);
	set .@Minutes, .@Time_Left / 60;
	set .@Time_Left, .@Time_Left - (.@Minutes * 60);
	
	set .@Time$, "";
	if( .@Days > 1 )
		set .@Time$, .@Time$ + .@Days + " dias, ";
	else if( .@Days > 0 )
		set .@Time$, .@Time$ + .@Days + " dia, ";

	if( .@Hours > 1 )
		set .@Time$, .@Time$ + .@Hours + " horas, ";
	else if( .@Hours > 0 )
		set .@Time$, .@Time$ + .@Hours + " hora, ";

	if( .@Minutes > 1 )
		set .@Time$, .@Time$ + .@Minutes + " minutos, ";
	else if( .@Minutes > 0 )
		set .@Time$, .@Time$ + .@Minutes + " minuto, ";

	if( .@Time_Left > 1 || .@Time_Left == 0 )
		set .@Time$, .@Time$ + .@Time_Left + " segundos";
	else if( .@Time_Left == 1 )
		set .@Time$, .@Time$ + .@Time_Left + " segundo";
	
	return .@Time$;
}

//////////////////////////////////////////////////////////////////////////////////
// Verifica se o equipamento foi trocado (por exemplo, por meio de hacks).
// A função verifica o equipamento atual em uma posição em relação aos valores
// fornecidos e registra quaisquer incompatibilidades.
// Retorna 0 se houver correspondência, 1 se houver incompatibilidade.
// -- callfunc "F_IsEquipIDHack",<posição do equipamento>,<ID do equipamento>
// -- callfunc "F_IsEquipRefineHack",<posição do equipamento>,<refinamento>
// -- callfunc "F_IsEquipCardHack",<posição do equipamento>,<Carta 0>,<Carta 1>,<Carta 2>,<Carta 3>
//////////////////////////////////////////////////////////////////////////////////
function	script	F_IsEquipIDHack	{
	set .@id_chk, getequipid(getarg(0));
	set .@id, getarg(1);
	if (.@id != .@id_chk) {
		logmes "Roubo: Tentou trocar o equipamento " + getitemname(.@id) + " por " + getitemname(.@id_chk) + ".";
		return 1;
	}
	return 0;
}
function	script	F_IsEquipRefineHack	{
	set .@refine_chk, getequiprefinerycnt(getarg(0));
	set .@refine, getarg(1);
	if (.@refine != .@refine_chk) {
		logmes "Roubo: Tentou trocar o equipamento com refino +" + .@refine + " por +" + .@refine_chk + ".";
		return 1;
	}
	return 0;
}
function	script	F_IsEquipCardHack	{
	set .@pos, getarg(0);
	for (set .@i,0; .@i < 4; set .@i, .@i+1) {
		set .@card, getarg(.@i + 1);
		set .@card_chk, getequipcardid(.@pos, .@i);
		if (.@card != .@card_chk) {
			logmes "Roubo: Tentou trocar a carta " + getitemname(.@card) + " por " + getitemname(.@card_chk) + ".";
			return 1;
		}
	}
	return 0;
}

//////////////////////////////////////////////////////////////////////////////////
// Retorna o plural de um substantivo - funciona na maioria dos casos, mas não para tudo!
// -- callfunc "F_GetPlural","<quantidade>"{,<0:normal/1:maiúsculas>}
// Examples:
//    callfunc("F_GetPlural","cachorro")            // retorna "cachorros"
//    callfunc("F_GetPlural","fox/raposa",1)          // retorna "FOXES/RAPOSAS"
//    callfunc("F_GetPlural","faca")          // retorna "facas"
//    callfunc("F_GetPlural","Pedaço de Bolo")  // retorna "Pedaços de Bolo"
//
// Retorna o plural de um substantivo se o número fornecido não for 1.
// -- callfunc "F_InsertPlural",<número>,"<substantivo>"{,<0:normal/1:maiúscula>{,"<formato da string>"}}
// O formato da string usa sprintf() e DEVE conter %d (arg0) e %s (arg1), nessa ordem.
// Exemplos:
//    callfunc("F_InsertPlural",1,"cachorro")    // retorna "1 cachorro"
//    callfunc("F_InsertPlural",3,"fox/raposa",1)  // retorna "3 FOXES/RAPOSAS"
//    // retorna "^FF00005^000000 facas"
//    callfunc("F_InsertPlural",5,"faca",0,"^FF0000%d^000000 %s")
//////////////////////////////////////////////////////////////////////////////////
function	script	F_GetPlural	{
	set .@str$, getarg(0);

	if (countstr(.@str$," ")) {  // várias palavras
		explode(.@tmp$,.@str$," ");
		set .@size, getarraysize(.@tmp$);

		// se o formato for "... de|em|no ...", pluraliza-se a palavra precedente; caso contrário, pluraliza-se a última palavra.
		if (compare(.@str$," de ") || compare(.@str$," em ") || compare(.@str$," no ")) {
			for (set .@i,1; .@i<.@size; set .@i,.@i+1) {
				if (getstrlen(.@tmp$[.@i]) == 2 && compare("de|em|no",.@tmp$[.@i]))
					break;
				set .@index, .@index + 1;
			}
		} else
			set .@index, .@size - 1;

		set .@str$, .@tmp$[.@index];
		set .@tmp$[.@index],"%s";
		set .@format$, implode(.@tmp$," ");
	} else
		set .@format$, "%s";

	set .@strlen, getstrlen(.@str$);
	if (.@strlen < 3)  // previne erros
		return ((getarg(1,0)) ? strtoupper(sprintf(.@format$,.@str$)) : sprintf(.@format$,.@str$));

	setarray .@suffix$[0], charat(.@str$,.@strlen - 1), substr(.@str$,.@strlen - 2,.@strlen - 1);
	if (!compare("abcdefghijklmnopqrstuvwxyz",.@suffix$[0])) {  // quando o último caractere não é uma letra
		set .@result$, .@str$;
	}

	// exceções comuns --> forma singular == forma plural
	else if (compare("fish|glasses|sunglasses|clothes|boots|shoes|greaves|sandals|wings|ears",.@str$)) {
		set .@result$, .@str$;
	}

	// termina em -s, -x, -z, -ch, -sh --> adiciona -es
	else if (.@suffix$[0] == "s" || .@suffix$[0] == "x" || .@suffix$[0] == "z" ||
	    .@suffix$[1] == "ch" || .@suffix$[1] == "sh") {
		set .@result$, .@str$ + "es";
	}

	// termina em -f, -fe --> remove -f, -fe --> adiciona -ves
	else if ((.@suffix$[0] == "f" || .@suffix$[1] == "fe") && .@suffix$[1] != "ff") {
		if (compare("belief|cliff|chief|dwarf|grief|gulf|proof|roof",.@str$))
			set .@result$, .@str$ + "s";  // exceções --> adiciona -s
		else
			set .@result$, substr(.@str$,0,.@strlen - 2 - (.@suffix$[1] == "fe")) + "ves";
	}
	
	// termina em consoante + -y --> remove -y --> adicona -ies
	else if (.@suffix$[0] == "y" && !compare("aeiou",charat(.@suffix$[1],0))) {
		set .@result$, delchar(.@str$,.@strlen - 1) + "ies";
	}

	// termina em -o --> exceções --> adiciona -es
	else if (.@suffix$[0] == "o" &&
	         compare("buffalo|domino|echo|grotto|halo|hero|mango|mosquito|potato|tomato|tornado|torpedo|veto|volcano",.@str$)) {
		set .@result$, .@str$ + "es";
	}

	// padrão --> adiciona -s
	else {
		set .@result$, .@str$ + "s";
	}

	return ((getarg(1,0)) ? strtoupper(sprintf(.@format$,.@result$)) : sprintf(.@format$,.@result$));
}
function	script	F_InsertPlural	{
	return sprintf(getarg(3,"%d %s"), getarg(0), ((getarg(0) == 1) ? getarg(1) : callfunc("F_GetPlural",getarg(1),getarg(2,0))));
}

//////////////////////////////////////////////////////////////////////////////////
// Retorna 'um' ou 'uma' com base na palavra.
// -- callfunc "F_GetArticle","<quantidade>";
// Exemplos:
//    callfunc("F_GetArticle","Maçã")  // retorna "uma"
//    callfunc("F_GetArticle","cachorro")    // retorna "um"
//
// Retorna 'um' ou 'uma' com base em uma palavra, seguido por outra palavra.
// -- callfunc "F_InsertArticle","<palavra>"{,<0:menúscula um/1:maiúscula Um>}
// Exemplos:
//    callfunc("F_InsertArticle","Maçã")  // returns "uma Maçã"
//    callfunc("F_InsertArticle","cachorro",1)  // returns "Um cachorro"
//////////////////////////////////////////////////////////////////////////////////
function	script	F_GetArticle	{
	set .@str$, strtolower(getarg(0));

	// nem uma palavra
	if (!charisalpha(.@str$,0))
		return "um";

	// Palavras de 1 letra
	if (getstrlen(.@str$) == 1) {
		if (strpos("aefhilmnorsx",.@str$) > -1)
			return "uma";
		else
			return "um";
	}

	// casos especiais
	if (preg_match("(euler|hour(?!i)|heir|honest|hono)",.@str$))
		return "uma";

	// consoantes
	if (preg_match("^[^aeiouy]",.@str$))
		return "um";

	// formas vocálicas especiais
	if (preg_match("^e[uw]",.@str$) || preg_match("^onc?e\b",.@str$) || preg_match("^uni([^nmd]|mo)",.@str$) || preg_match("^u[bcfhjkqrst][aeiou]",.@str$))
		return "um";
	if (preg_match("^ut[th]",.@str$))
		return "uma";

	// special capitals (raro)
	//if (preg_match("^U[NK][AIEO]?",getarg(0)))
	//	return "um";

	// vogais
	if (preg_match("^[aeiou]",.@str$))
		return "uma";

	// y... (raro)
	//if (preg_match("^(y(b[lor]|cl[ea]|fere|gg|p[ios]|rou|tt))",.@str$))
	//	return "uma";

	return "um";
}
function	script	F_InsertArticle	{
	set .@article$, callfunc("F_GetArticle",getarg(0));
	return sprintf("%s %s", ((getarg(1,0)) ? replacestr(.@article$,"um","Um") : .@article$), getarg(0));
}

//////////////////////////////////////////////////////////////////////////////////
// Retorna um número com vírgulas entre cada três dígitos.
// -- callfunc "F_InsertComma",<number>
// Exemplos:
//    callfunc("F_InsertComma",7777777)  // retorna "7,777,777"
//////////////////////////////////////////////////////////////////////////////////
function	script	F_InsertComma	{
	set .@str$, getarg(0);
	for (set .@i,getstrlen(.@str$)-3; .@i>0; set .@i,.@i-3)
		set .@str$, insertchar(.@str$,",",.@i);
	return .@str$;
}

//////////////////////////////////////////////////////////////////////////////////
// Retorna um número com os sufíxos '-st', '-nd', '-rd', ou '-th'.
// -- callfunc "F_GetNumSuffix",<número>
// Examples:
//    callfunc("F_GetNumSuffix",1)   // retorna "1°"
//    callfunc("F_GetNumSuffix",11)  // retorna "11°"
//    callfunc("F_GetNumSuffix",32)  // retorna "32°"
//////////////////////////////////////////////////////////////////////////////////
function	script	F_GetNumSuffix	{
	set .@n, getarg(0);
	set .@mod, .@n % 10;
	if      (.@mod == 1 && .@n != 11) return .@n+"°";
	else if (.@mod == 2 && .@n != 12) return .@n+"°";
	else if (.@mod == 3 && .@n != 13) return .@n+"°";
	else return .@n+"°";
}
