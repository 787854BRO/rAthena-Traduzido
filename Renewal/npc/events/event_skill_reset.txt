//= Skill Reset with payment, minimum BaseLv.60 to reset skill.
//= Each 1 BaseLv need 20.000 Zeny, so be careful to use skill.
//============================================================
yuno,138,187,4	script	Hipnotizadora	124,{
	set @npcname$,"[^D5A500Hipnotizadora^000000]";
	mes @npcname$;

	if(MISC_QUEST & 1024){
		mes "Desculpe, mas você só pode completar este evento uma vez.";
		close;
	}
	if(BaseLevel < 60) {
		mes "Por favor, retorne quando chegar ao Nv. de Base 60.";
		close;
	}
	if(SkillPoint != 0) {
		mes "Você precisará usar todos os seus pontos de habilidade caso queira continuar.";
		mes "Por favor, volte logo!";
		close;
	}
	if(Weight || checkfalcon() || checkcart() || checkriding()) {
		mes "Você gostaria de resetar suas habilidades?";
		mes "Desculpe-me, mas...";
		next;
		mes @npcname$;
		if(Weight) {
			mes "Você não pode resetar suas habilidades";
			mes "enquanto estiver carregando";
			mes "muitos itens.";
		} else if(checkcart())
			mes "Por favor, livre-se do seu Carrinho para continuarmos.";
		else if(checkfalcon())
			mes "Por favor, livre-se do seu Falcão para continuarmos.";
		else if(checkriding())
			mes "Por favor, livre-se do seu Peco Peco para continuarmos.";
		next;
		mes @npcname$;
		mes "Volte em breve!";
		close;
	}
	mes "Olá, eu sou a Resetadora de Habilidades.";
	mes "Seu nome é ^0080FF"+strcharinfo(0)+"^000000.";
	mes "Como posso ajudá-lo?";
	next;
	menu "^009500Inf. sobre Resete de Habilidades.^000000",L_Info,"^00B6FFResetar Habilidades.^000000",L_Reset,"^000088Deixa para lá.^000000",-;

	mes @npcname$;
	mes "Você sabe onde me encontrar,";
	mes "se quiser um resete!!";
	close;

L_Info:
	mes @npcname$;
	mes "Resetar as habilidades não é gratuito!!";
	mes "O gasto para resetar as habilidades é de ^D5A50020.000 Zenys x Nível de Base^000000.";
	mes "Pois é... Cada Nível de Base custa 20.000 zenys para resetar suas habilidades.";
	next;
	mes @npcname$;
	mes "Ah sim, mais uma coisa!"; 
	mes "Quaisquer Carrinhos, Falcões ou Pecos que você tenha com você,"; 
	mes "serão removidos se você resetar suas habilidades.";
	next;
	mes @npcname$;
	mes "Lembre-se, você só terá essa vez para resetar seus pontos de habilidade.";
	mes "Então, tenha uidado com suas habilidades daqui em diante.";
	close;
	
L_Reset:
	mes @npcname$;
	mes "Antes de resetar suas habilidades,";
	mes "você deve me informar seu Nível de Base.";
	next;
	mes "^D5A500[" + strcharinfo(0) + "]^000000";
	mes "Meu Nv. de Base é ^AA00AA"+BaseLevel+"^000000."; 
	next;
	set @zeny,BaseLevel*20000;
	mes @npcname$;
	mes "O total de zenys necessários para pagar a taxa de serviço do resete de habilidades é de ^529DFF"+@zeny+" zenys^000000.";
	next;
	if(Zeny < @zeny) {
		mes @npcname$;
		mes "Parece que você não tem dinheiro suficiente.";
		mes "Além disso, aguarde pela oportunidade.";
		emotion ET_SCRATCH;
		close;
	}
	set Zeny, Zeny-@zeny;
	set MISC_QUEST,MISC_QUEST | 1024;
	resetskill;
	mes @npcname$;
	mes "Muito obrigado!";
	emotion ET_THANKS;
//	logmes "EVENTO DE RESETE DE HABILIDADES";

	close;
}
