payon,162,176,4	script	Mago do Halloween#iRO08	704,{
	if (Hallow08 < 1) {
		mes "[Mago do Halloween]";
		mes "Kkkkkkkkk!";
		mes "Eu tenho um evento especial nesse Halloween que testa sua sorte e agilidade.";
		mes "Gostaria de participar?";
		next;
		mes "[Mago do Halloween]";
		mes "Vamos! Não seja covarde!";
		mes "Se você conseguir coletar bastante Ticket você consegue bons prêmios!";
		mes "Então o que você diria?";
		next;
		goto MainMenu;
	}
	if (Hallow08Kill == 1) {
		mes "[Mago do Halloween]";
		mes "Você é 'medroso ou algo do tipo?!";
		mes "Você sabe que quer tentar de novo...";
		mes "Você sabe as regras?";
		next;
		set Hallow08Kill,0;
		if(select("Sim, eu sei.:Não, eu não sei.") == 2) {
			goto Rules;
		}
		goto Participate;
	}
	if (Hallow08Kill == 2) {
		mes "[Mago do Halloween]";
		mes "Oh, bem feito! Está tudo bem!";
		mes "Não é divertido com Zumbis??";
		mes "Você sabe, Zumbis eram pessoas também!";
		mes "Ha!";
		mes "Kkkkkkk.";
		next;
		getitem 7941,1;
		set Hallow08Kill,0;
		mes "[Mago do Halloween]";
		mes "Como eu prometi,";
		mes "você pode me trazer Tickets de Dia das Bruxas que eu troco por itens bem legais.";
		close;
	}
	if (Hallow08 > 0) {
		mes "[Mago do Halloween]";
		mes "Bem, você quer ouvir denovo as regras, ou simplesmente quer voltar...";
		next;
		goto MainMenu2;
	}

		Rules:
			mes "[Mago do Halloween]";
			mes "Essa aldeia é como uma Payon virtual.";
			mes "Há Zumbis e fantasmas que vagueiam ao redor e três saidas do sul, mas apenas uma funciona.";
			mes "Isso é até você descobrir.";
			next;
			mes "[Mago do Halloween]";
			mes "Você não pode usar habilidade nenhuma";
			mes "para matar os carniçais ou Zumbis.";
			mes "E mais uma coisa...";
			mes "você não pode esquecer...";
			next;
			mes "[Mago do Halloween]";
			mes "Todos os participantes deverão ficar sem algum tipo de equipamento.";
			mes "Coloque todas as suas coisas no armazém e volte aqui quando você estiver com o peso zerado.";
			next;
			mes "[Mago do Halloween]";
			mes "Ah e mais uma coisa!";
			mes "Você não pode montar no PecoPeco ou usar um carrinho.";
			mes "Se estiver, eu vou retirar antes de entrar no evento.";
			mes "Entendeu?";
			next;
			mes "[Mago do Halloween]";
			mes "Não se esqueça, há três saídas, mas apenas uma delas funciona de forma aleartória, os Zumbis e Carniçais não podem ser mortos e você não pode usar qualquer tipo de equipamento.";
			next;
			if (Hallow08 > 0) {
				mes "[Mago do Halloween]";
				mes "Ei...";
				mes "Volte quando estiver pronto.";
				close;
			} else {
				goto MainMenu;
			}

		Participate:
			mes "[Mago do Halloween]";
			mes "Ok, você está pronto.";
			mes "Deixe eu checar seu peso.";
			next;
			if (Weight > 0) {
				mes "[Mago do Halloween]";
				mes "Caramba!";
				mes "Há sempre uma ovelha negra em qualquer lugar.";
				next;
				mes "[Mago do Halloween]";
				mes "Você achou que eu não ia perceber que seu peso está acima de '0'?";
				mes "Você está acima do peso...";
				close;
			} else {
				mes "[Mago do Halloween]";
				mes "Você está bem para ir, e seu peso está zerado.";
				next;
				mes "[Mago do Halloween]";
				mes "Eu estou muito sobrecarregado com o meu trabalho, por isso estou exausto.";
				next;
				mes "[Mago do Halloween]";
				mes "Eu as vezes esqueço de enviar-lhe lá...";
				next;
				mes "[Mago do Halloween]";
				mes "Espero que volte bem.";
				close2;
				if (Hallow08 < 1) {
					set Hallow08,1;
				}
				set Hallow08Kill,1;
				set @Hallow08Warp, rand(1,3);
				percentheal -98,0;
				setriding 0;
				setcart 0;
				warp "evt_zombie",155,246;
				end;
			}

		TicketExchange:
			mes "[Mago do Halloween]";
			mes "Você quer trocar os Tickets pelos prêmios?";
			mes "Boa sorte! Kkkkkk!";
			next;
			mes "[Mago do Halloween]";
			mes "Vou te dizer os itens que você pode trocar.";
			next;
			mes "[Mago do Halloween]";
			mes "5 Tickets por uma Torta Assustadora.";
			mes "20 Tickets por um Zé-Lanterna.";
			mes "50 Tickets por uma Velha Caixa Azul.";
			mes "70 Tickets por uma Velha Caixa Roxa.";
			mes "200 Tickets por um Velho Álbum de Cartas.";
			next;
			mes "[Mago do Halloween]";
			mes "O que você gostaria de trocar?";
			next;
			switch(select("Torta Assustadora:Zé-Lanterna:Velha Caixa Azul:Velha Caixa Roxa:Velho Álbum de Cartas")) {
			case 1:
				if (countitem(7941) < 5) {
					goto NotEnough;
				} else {
					delitem 7941,5;
					getitem 12192,1;
					goto Enough;
				}
				break;
			case 2:
				if (countitem(7941) < 20) {
					goto NotEnough;
				} else {
					delitem 7941,20;
					getitem 5134,1;
					goto Enough;
				}
				break;
			case 3:
				if (countitem(7941) < 50) {
					goto NotEnough;
				} else {
					delitem 7941,50;
					getitem 603,1;
					goto Enough;
				}
				break;
			case 4:
				if (countitem(7941) < 70) {
					goto NotEnough;
				} else {
					delitem 7941,70;
					getitem 617,1;
					goto Enough;
				}
				break;
			case 5:
				if (countitem(7941) < 200) {
					goto NotEnough;
				} else {
					delitem 7941,200;
					getitem 616,1;
					goto Enough;
				}
			}

		NextTime:
			mes "[Mago do Halloween]";
			mes "Ok, até então.";
			mes "Kkkkkkkk.";
			close;

	MainMenu:
		menu "Regras.",Rules,"Participar.",Participate,"Trocar Tickets por prêmios.",TicketExchange,"Volto depois.",NextTime;

	MainMenu2:
		menu "Participar.",Participate,"Regras",Regras,"Trocar Tickets por prêmios.",TicketExchange,"Volto depois.",NextTime;

	NotEnough:
		mes "[Mago do Halloween]";
		mes "Você não tem o número suficiente de Tickets!";
		mes "Não é possível, você não conta?";
		mes "Por favor, venha aqui com o número certo de Tickets.";
		close;

	Enough:
		mes "[Mago do Halloween]";
		mes "Aqui está.";
		mes "Você precisa de...";
		mes "mais alguma coisa?";
		next;
		if (Hallow08 == 1) {
			goto MainMenu2;
		} else {
			goto MainMenu;
		}
}
