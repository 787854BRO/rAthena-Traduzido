// -------------------- Funções --------------------
// * Go("<mapa>",<x>,<y>);
//	~ Portais direto para um mapa.
//
// * Disp("<Menu de Opção>",<primeira Opção>,<última Opção>);
// * Pick("<mapa_prefixo>"{,<index offset>});
//	~ Menu dinâmico e seleção de mapa (numeração automática).
//	~ Campos e masmorras devem usar as funções Disp e Pick.
//
// * Disp("<Opção 1>:<Opção 2>:<etc.>");
// * Pick("","<mapa1>","<mapa2>","<etc.>");
//	~ Menu manual e seleção de mapa (listado).
//
// * Restrict("<RE | Pre-RE>"{,<números do menu de opções>});
//	~ Permite mapa apenas para os modos de renovação ou pré-renovação.
//     Se os números das opções do menu forem fornecidos, apenas esses mapas
//     serão restritos (ou seja, não para "Go").
//
// Outras notas:
//   ~ Array @c[] detém tudo (x,y) coordenadas.
//   ~ Use @c[2] EXCETO quando os mapas começarem dinamicamente
//	  em 0: use @c[0] e Pick() offset 1.
// --------------------------------------------------

function Go {
	set lastwarp$, getarg(0);
	set lastwarpx, getarg(1,0);
	set lastwarpy, getarg(2,0);
	warp getarg(0),getarg(1,0),getarg(2,0);
	end;
}
function Disp {
	if (getargcount() < 3)
		set @menu$, getarg(0);
	else {
		set @menu$,"";
		for (set .@i,getarg(1); .@i<=getarg(2); set .@i,.@i+1)
			set @menu$, @menu$+getarg(0)+" "+.@i+":";
	}
	return;
}
function Pick {
	set .@warp_block,@warp_block;
	set @warp_block,0;
	if((@f && .OnlyFirstFld) || (@d && .OnlyFirstDun)){
		set .@select,1;
		if(.@warp_block){
			while(.@warp_block & (1<<.@select)){
				.@select += 1;
			}
		}
	}else{
		set .@select, select(@menu$);
	}
	if (getarg(0) == "") {
		set .@i, .@select;
		set .@map$, getarg(.@i);
	} else {
		set .@i, .@select-getarg(1,0);
		set .@map$, getarg(0)+((.@i<10)?"0":"")+.@i;
	}
	if (.@warp_block & (1<<.@select)) {
		message strcharinfo(0),"Este mapa não está disponível na "+((checkre(0))?"":"Pré-")+"Renovação.";
		end;
	}
	set .@x, @c[.@i*2];
	set .@y, @c[.@i*2+1];
	deletearray @c[0],getarraysize(@c);
	@f = false; @d = false;
	Go(.@map$,.@x,.@y);
}
function Restrict {
	if ((getarg(0) == "RE" && !checkre(0)) || (getarg(0) == "Pre-RE" && checkre(0))) {
		if (getarg(1,0)) {
			set @warp_block,0;
			for (set .@i,1; .@i<getargcount(); set .@i,.@i+1)
				set @warp_block, @warp_block | (1<<getarg(.@i));
		} else {
			message strcharinfo(0),"Este mapa não está disponível na "+((checkre(0))?"":"Pré-")+"Renovação.";
			end;
		}
	}
	return;
}
