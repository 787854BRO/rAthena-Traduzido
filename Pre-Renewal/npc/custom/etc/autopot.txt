//===== rAthena Script =======================================
//= Auto-Potion
//===== Description: =========================================
//= Provides an @autopot command to automatically use potions.
//===== Changelogs: ==========================================
//= 1.0 First version. [Euphy]
//= 2.0 rewrite and adding SP option. [sader1992]
//============================================================
-	script	#autopot	-1,{
OnInit:
	.HP_Option = true;	//Permitir a opção HP
	.SP_Option = true;	//Permitir a opção SP
	.MinDelay = 100;	//Atraso mínimo permitido em milissegundos (padrão recomendado: 100) [valores mais baixos aumentarão a pressão do servidor]
	setarray .blackList[0],0;	//Array de IDs de itens listados na Lista Negra.

	if(!.HP_Option && !.SP_Option){
		debugmes "Perigo: @autopot de ambos HP e SP estão desabilitados.";
		debugmes "Perigo: O script @autopot foi desabilitado!";
		end;
	}
	bindatcmd("autopot",strnpcinfo(0)+"::OnCommand",0,99);
end;

L_Help:
	dispbottom "Comandos disponíveis:";
	if(.HP_Option && .SP_Option){
		dispbottom "    @autopot <hp|sp> <Id do Item> {<HP Mínimo|SP % [1..100]> {<Atraso [" + .MinDelay + "..1000]>}}";
		dispbottom "    @autopot <hp|sp> lista";
		dispbottom "    @autopot <hp|sp> <liga|desliga>";
	}else{
		dispbottom "    @autopot <" + (.HP_Option?"HP":"SP") + "> <Id do Item> {<Mínimo " + (.HP_Option?"HP":"SP") + " % [1..100]> {<Atraso [" + .MinDelay + "..1000]>}}";
		dispbottom "    @autopot <" + (.HP_Option?"HP":"SP") + "> lista";
		dispbottom "    @autopot <" + (.HP_Option?"HP":"SP") + "> <liga|desliga>";
	}
	dispbottom "    @autopot infor";
	dispbottom "    @autopot ajuda";
	dispbottom "    @autopot lista negra";
	return;

L_Info:
	dispbottom "----------- Informações de Poção-Auto ------------";
	if(.HP_Option){
		dispbottom "POÇÃO de HP: " + getitemname(@autopot_hp_ID) + " (" + @autopot_hp_ID + ")";
		dispbottom "HP Mínimo: " + @autopot_hp_Min + " %";
	}
	if(.SP_Option){
		dispbottom "POÇÃO de SP: " + getitemname(@autopot_sp_ID) + " (" + @autopot_sp_ID + ")";
		dispbottom "SP Mínimo: " + @autopot_sp_Min + " %";
	}
	dispbottom "Atraso: " + @autopot_delay + " milisegundos";
	dispbottom "--------------------------------------------------";
	return;

L_Start:
	deltimer strnpcinfo(3) + "::OnStart";
	.@type$  = getarg(0,0);
	.@potion = getarg(1,0);
	setd "@autopot_" + .@type$ + "_Min",getarg(2,0);
	@autopot_delay = getarg(3,0);

	if (getd("@autopot_" + .@type$ + "_Min")   < 1  || getd("@autopot_" + .@type$ + "_Min") > 100)  setd "@autopot_" + .@type$ + "_Min",90;
	if (@autopot_delay < 50 || @autopot_delay > 1000) @autopot_delay = .MinDelay;
	if (getiteminfo(.@potion, ITEMINFO_TYPE) != IT_HEALING) {
		dispbottom getitemname(.@potion) + " não é um item de cura.";
		end;
	}
	if(inarray(.blackList[0],.@potion) != -1){
		message strcharinfo(0), "'" + getitemname(.@potion) + "' está na lista negra e não pode ser usado.";
		end;
	}
	if (BaseLevel < getiteminfo(.@potion, ITEMINFO_EQUIPLEVELMIN)) {
		message strcharinfo(0), "Seu nível de base é muito baixo para usar '" + getitemname(.@potion) + "'.";
		end;
	}
	setd "@autopot_" + .@type$ + "_ID",.@potion;
	setd "@autopot_" + .@type$ + "_Active",true;
	addtimer .@delay,strnpcinfo(3) + "::OnStart";
	message strcharinfo(0), "O sistema de Poção-Auto " + strtoupper(.@type$) + " foi ativado.";
	callsub L_Info;
	return;

OnCommand:
	if (!getarraysize(.@atcmd_parameters$)) {
		message strcharinfo(0), "Sintaxe inválida.";
		callsub L_Help;
		end;
	}
	.@command$[0] = strtolower(.@atcmd_parameters$[0]);
	.@command$[1] = strtolower(.@atcmd_parameters$[1]);
	
	if(.@command$[0] == "hp" || .@command$[0] == "sp"){
		if((!.HP_Option && .@command$[0] == "hp") || (!.SP_Option && .@command$[0] == "sp")){
			message strcharinfo(0), "O sistema de Poção-Auto " + strtoupper(.@command$[0]) + " está Desativado no servidor.";
			end;
		}
		if(.@command$[1] == "liga"){
			if(getd("@autopot_" + .@command$[0] + "_Active")){
				message strcharinfo(0), "O sistema de Poção-Auto " + strtoupper(.@command$[0]) + " já está ativado.";
				end;
			}
			if(!getd("@autopot_" + .@command$[0] + "_ID")){
				dispbottom "Você precisa definir uma Poção " + strtoupper(.@command$[0]) + " para ser usada.";
				callsub L_Help;
				end;
			}
			callsub L_Start,.@command$[0],getd("@autopot_" + .@command$[0] + "_ID");
			end;
		}
		if(.@command$[1] == "desliga"){
			message strcharinfo(0), "O sistema de Poção-Auto " + strtoupper(.@command$[0]) + " foi desativado.";
			setd "@autopot_" + .@command$[0] + "_Active",false;
			if(!@autopot_hp_Active || !@autopot_sp_Active)
				deltimer strnpcinfo(3) + "::OnStart";
			end;
		}
		if(.@command$[1] == "lista"){
			getinventorylist;
			for(.@i = 0;.@i < @inventorylist_count;.@i++){
				if (getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) == IT_HEALING) {
					.@items[.@count] = @inventorylist_id[.@i];
					.@menu$ = .@menu$ + sprintf("~ ^0055FF%s^000000 (%dx)" + (inarray(.blackList[0],@inventorylist_id[.@i]) != -1? " ^364022Lista Negra^000000":"") + (BaseLevel < getiteminfo(@inventorylist_id[.@i],12)? " ^E82C0CPrecisa de Nível Superior^000000":"") + ":", getitemname(@inventorylist_id[.@i]), countitem(@inventorylist_id[.@i]));
					.@count++;
				}
			}
			if(.@count){
				mes "[ Poção-Auto ]";
				mes "Selecione o item " + strtoupper(.@command$[0]) + ".";
				.@select = select(.@menu$ + "   ^777777Cancelar!^000000") - 1;
				if (.@select != .@count)
					callsub L_Start,.@command$[0],.@items[.@select];
				close2;
			}else{
				message strcharinfo(0), "Não há itens de cura em seu inventário.";
			}
			end;
		}
		.@potion = atoi(.@command$[1]);
		.@percent = atoi(.@atcmd_parameters$[2]);
		.@delay = atoi(.@atcmd_parameters$[3]);
		if(getitemname(.@potion) != "null"){
			callsub L_Start,.@command$[0],.@potion,.@percent,.@delay;
			end;
		}
		message strcharinfo(0), "Poção-Auto : Algo deu errado!";
		callsub L_Help;
		end;
	}
	if(.@command$[0] == "lista negra"){
		dispbottom "------------ Lista Negra de Itens ------------";
		for(.@i=0;.@i<getarraysize(.blackList);.@i++){
			dispbottom " " + getitemname(.blackList[.@i]) + " (" + .blackList[.@i] + ")";
		}
		dispbottom "----------------------------------------------";
		end;
	}
	if(.@command$[0] == "infor"){
		if(@autopot_hp_Active || @autopot_sp_Active){
			message strcharinfo(0), "As informações do sistema de Poção-Auto serão exibidas abaixo.";
			callsub L_Info;
		}else{
			message strcharinfo(0), "O sistema de Poção-Auto não está habilitado.";
		}
		end;
	}
	if(.@command$[0] == "ajuda"){
		message strcharinfo(0), "A lista de comandos será exibida abaixo.";
		callsub L_Help;
		end;
	}
	
	message strcharinfo(0), "Sintaxe inválida.";
	callsub L_Help;
end;

OnStart:
	if (!getstatus(SC_BERSERK) && !getstatus(SC_SATURDAYNIGHTFEVER) && !getstatus(SC_GRAVITATION) &&
	    !getstatus(SC_TRICKDEAD) && !getstatus(SC_HIDING) && !getstatus(SC__SHADOWFORM) && !getstatus(SC__INVISIBILITY) &&
	    !getstatus(SC__MANHOLE) && !getstatus(SC_KAGEHUMI) && !getstatus(SC_HEAT_BARREL_AFTER) &&
		!getstatus(SC_STONE) && !getstatus(SC_FREEZE) && !getstatus(SC_STUN) && !getstatus(SC_SLEEP))
		{
		if(Hp){
			if(@autopot_hp_Active && Hp * 100 / MaxHp < @autopot_hp_Min) {
				if (countitem(@autopot_hp_ID)) {
					delitem @autopot_hp_ID,1;
					consumeitem @autopot_hp_ID;
					specialeffect2 207;
				}
			}
			if(@autopot_sp_Active && Sp * 100 / MaxSp < @autopot_sp_Min) {
				if (countitem(@autopot_sp_ID)) {
					delitem @autopot_sp_ID,1;
					consumeitem @autopot_sp_ID;
					specialeffect2 208;
				}
			}
		}
	}
	if(@autopot_hp_Active || @autopot_sp_Active)
		addtimer @autopot_delay,strnpcinfo(3) + "::OnStart";
end;
}
