prontera,151,256,5	script	Sala Privada MVP	100,{  //Novo local e nome adicionado pelo [ADM] Eragonat PK.

//prontera,148,174,5	script	Sala Privada MVP	100,{  Local Original.
	mes "[Sala Privada MVP]";
	mes "Por favor, selecione uma sala privada MVP.";
	if ( getvariableofnpc( .rentcost, "MVP Summoner" ) )
		mes "O custo para alugar uma sala durante "+ getvariableofnpc( .timeout, "MVP Summoner" ) +" minuto(s) é "+ callfunc("F_InsertComma", getvariableofnpc( .rentcost, "MVP Summoner" ) ) +" zeny(s).";
	else
		mes "Você só poderá utilizar a sala apenas durante "+ getvariableofnpc( .timeout, "MVP Summoner" ) +" minuto(s).";
	mes " ";
	for ( .@i = 1; .@i <= 8; .@i++ )
		if ( getvariableofnpc( .renttime[.@i], "MVP Summoner" ) )
			mes "Sala #"+ .@i +" = "+ .color$[ .type[.@i] ] + .whoinuse$[.@i] +"^000000";
	next;
	.@room = select(
		"Sala MVP 1 ["+ getmapusers("06guild_01") +"]",
		"Sala MVP 2 ["+ getmapusers("06guild_02") +"]",
		"Sala MVP 3 ["+ getmapusers("06guild_03") +"]",
		"Sala MVP 4 ["+ getmapusers("06guild_04") +"]",
		"Sala MVP 5 ["+ getmapusers("06guild_05") +"]",
		"Sala MVP 6 ["+ getmapusers("06guild_06") +"]",
		"Sala MVP 7 ["+ getmapusers("06guild_07") +"]",
		"Sala MVP 8 ["+ getmapusers("06guild_08") +"]");
	if ( getvariableofnpc( .renttime[.@room], "MVP Summoner" ) ) {
		if ( .inuseid[.@room] == getcharid( .type[.@room] ) ) {
			warp "06guild_0"+ .@room, 0,0;
			close;
		} else {
			mes "[Sala Privada MVP]";
			mes "Esta sala está reservada para ";
			mes .color$[ .type[.@room] ] + .whoinuse$[.@room] +"^000000.";
			mes "Por favor, selecione outra.";
			close;
		}
	}
	mes "[Sala Privada MVP]";
	mes "Reservar esta sala para...";
	next;
	set .@type, select( "Para os membros do meu grupo.", "Para os membros do meu Clã.", "Para o meu uso pessoal." );
	if ( !getcharid(.@type) ) {
		mes "[Sala Privada MVP]";
		mes "Você não possui um "+( ( .@type == 1 )? "Grupo" : "Clã" )+".";
		close;
	}
	else if ( Zeny < getvariableofnpc( .rentcost, "MVP Summoner" ) ) {
		mes "Você não tem zeny suficiente para alugar uma sala.";
		close;
	}
	else if ( getvariableofnpc( .renttime[.@room], "MVP Summoner" ) ) {
		mes "[Sala Privada MVP]";
		mes "Sinto muito, alguém já registrou esta sala, e foi mais rápido que você.";
		close;
	}
	for ( .@i = 1; .@i <= 8; .@i++ ) {
		if ( ( getvariableofnpc( .renttime[.@i], "MVP Summoner" ) ) && .@type == .type[.@i] && getcharid(.@type) == .inuseid[.@i] ) {
			mes "[Sala Privada MVP]";
			mes "Você já alugou a Sala#"+ .@i +". Use a sala já alugada.";
			close;
		}
	}
	set .type[.@room], .@type;
	set .inuseid[.@room], getcharid(.@type);
	set .whoinuse$[.@room], strcharinfo( ( .@type == 3 )? 0 : .@type );
	Zeny -= getvariableofnpc( .rentcost, "MVP Summoner" );
	warp "06guild_0"+ .@room, 0,0;
	killmonsterall "06guild_0"+ .@room;
	donpcevent "MVP Summoner#"+ .@room +"::OnEnterMap";
	close;
OnInit:
	.color$[1] =  "^EE8800"; // Cor do Grupo
	.color$[2] =  "^70CC11"; // Cor do Clã
	.color$[3] =  "^0000FF"; // Cor Pessoal
	end;
}

			set .@menu, select(.bossid_menu$) -1;
			mes "[Invocadora MVP]";
			if ( Zeny < .bosscost ) {
				mes "Você não tem zeny suficiente para convocar um mini-boss.";
				close;
			}
		} else {
			set .@menu, select(.bossid_menu$) -1;
			mes "[Invocadora MVP]";
		}
		mes "Por favor, prepare-se.";
		close2;
		if ( Zeny < .bosscost ) end;
		Zeny -= .bosscost;
		monster "this", 0, 0, "--já--", .bossid[.@menu], 1, strnpcinfo(3)+"::OnMobDead";
		end;
	case 4:
		if ( mobcount( "this", strnpcinfo(3)+"::OnMobDead" ) > 0 ) {
			mes "[Invocadora MVP]";
			mes "Eu não posso oferecer este serviço quando há monstros por perto.";
			close;
		}
		close2;
		callshop "MVP room#branch", 1;
		end;
	case 5:
		mes "[Invocadora MVP]";
		mes "Você tem certeza que quer sair desta sala?";
		next;
		if ( select( "Sim:Não") == 1 )
			warp .respawnmap$, .respawnx, .respawny;
		close;
	case 6:
		mes "[Invocadora MVP]";
		if ( getmapusers( strcharinfo(3) ) > 1 ) {
			mes "Ainda há alguns jogadores nesta sala. Certifique-se de que você é o último membro desta sala a usar esta opção.";

		}

		next;
