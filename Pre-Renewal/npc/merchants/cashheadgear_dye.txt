prt_in,253,168,4	script	Alora	862,{

// Mistura de Tinta Misteriosa
	set .@dyeID, 6220;
// Opções de Chapéus
	setarray .@hgn$[1], "Chapéu de Mestre Arcano","Gorro","Chapéu de Gatinho Dorminhoco","Chapéu de Deviruchi","Peruca","Laço","Chifres Majestosos";
// Chapéus de Mestre Arcano - Normal, Amarelo, Azul, Marrom, Cinza, Vermelho
	setarray .@hg_1[1], 5027, 5242, 5241, 5240, 5239, 5238;
// Gorros - Normal, Rosa, Azul, Marrom
	setarray .@hg_2[1], 5076, 5237, 5236, 5235;
// Chapéus de Gatinhos Dorminhocos - Normal, Azul, Marrom, Cinza, Rosa, Amarelo
	setarray .@hg_3[1], 5058, 5233, 5231, 5230, 5232, 5234;
// Chapéus de Deviruchi - Normal, Vermelho, Cinza, Marrom
	setarray .@hg_4[1], 5038, 5227, 5228, 5229;
// Perucas - Peruca Carnavalesca Feliz, Peruca Carnavalesca Brilhante, Peruca Carnavalesca Chamativa, Peruca Carnavalesca Fantástica
	setarray .@hg_5[1], 5273, 5274, 5275, 5276;
// Laços - Normal, Preto, Amarelo, Verde, Rosa, Vermelho, Laranja, Branco
	setarray .@hg_6[1], 2208, 5191, 5192, 5193, 5194, 5195, 5196, 5197;
// Chifres Majestosos - Normais, Evoluídos
	setarray .@hg_7[1], 2256, 5217;

	mes "[Alora]";
	mes "Olá, eu posso mudar a cor do seu chapéu se você me trouxer o item Mistura de Tinta Misteriosa.";
	next;
	mes "[Alora]";
	mes "Você tem algum chapéu que gostaria de tingir?";
	next;
	for(set .@i,1; .@i < getarraysize(.@hgn$); set .@i,.@i+1) {
		if(.@i == 1)    set .@menu$, .@hgn$[.@i];
		else            set .@menu$, .@menu$ +":"+ .@hgn$[.@i];
	}
	set .@menu$, .@menu$ +":Cancelar";
	set .@t, select(.@menu$);
	if(.@t == getarraysize(.@hgn$)) {
		mes "[Alora]";
		mes "Tenha uma boa jornada aventureiro!";
		mes "Se você estiver ancioso para experimentar uma nova cor em sua chapelaria Kafra ou naqueles chapéus que você encontrou em suas aventuras por ai, por favor, venha até mim!";
		close;
	}
	mes "[Alora]";
	mes "Ok, para qual cor você deseja mudá-lo?";
	next;
	for(set .@i,1; .@i < getarraysize(getd(".@hg_"+.@t+"")); set .@i,.@i+1) {
		if(.@i == 1 && .@t != 5)
			set .@menu$, "Normal";
		else if(.@i == 1 && .@t == 5)
			set .@menu$, getitemname(getd(".@hg_"+.@t+"["+.@i+"]"));
		else {
			sscanf(getitemname(getd(".@hg_"+.@t+"["+.@i+"]")),"%s "+ .@hgn$[.@t],.@colour$);
			set .@menu$, .@menu$ +":"+ .@colour$;
		}
	}
	set .@menu$, .@menu$ +":Cancelar";
	set .@c, select(.@menu$);
	if(.@c == getarraysize(getd(".@hg_"+.@t+""))) {
		mes "[Alora]";
		mes "Oh, certo sem problemas!";
		close;
	}
	set .@dyHG, getd(".@hg_"+.@t+"["+.@c+"]");
	mes "[Alora]";
	mes "Oh, estou tão animado e você não está?";
	mes "Só para ter certeza, é a cor "+ .@hgn$[.@t] +" que você quer que eu use?";
	next;
	mes "[Alora]";
	mes "Por favor, entenda que vou usar ^FF00001º "+ .@hgn$[.@t] +" dessa cor em seu inventário!^000000";
	next;
	mes "[Alora]";
	mes "Quaisquer refinamentos e cartas serão dissolvidos pela tinta misteriosa, então certifique-se de ter o item ^0000FF+0 "+ getitemname(.@dyHG) +" sem nenhuma carta equipada.^000000";
	next;
	mes "["+ strcharinfo(0) +"]";
	mes "Ok, obrigado pelo aviso, acho que vou dar o meu...";
	next;
	set .@menu$, "Deixa para lá";
	for(set .@i,1; .@i < getarraysize(getd(".@hg_"+.@t+"")); set .@i,.@i+1) {
		if(.@i == 1)    set .@menu$, .@menu$ +":Normal "+ getitemname(getd(".@hg_"+.@t+"["+.@i+"]"));
		else            set .@menu$, .@menu$ +":"+ getitemname(getd(".@hg_"+.@t+"["+.@i+"]"));
	}
	set .@mine, select(.@menu$) - 1;
	if(!.@mine) {
		mes "[Alora]";
		mes "É melhor estar muito certo disso, para que tenha uma viagem segura.";
		close;
	}
	set .@myHG, getd(".@hg_"+.@t+"["+.@mine+"]");
	if(.@dyHG == .@myHG) {
		mes "[Alora]";
		mes "O que aconteceu?";
		close;
	}
	if(!countitem(.@dyeID) || !countitem(.@myHG)) {
		mes "[Alora]";
		mes "Oh meu Deus, parece estar faltando alguma coisa.";
		close;
	}
	mes "[Alora]";
	mes "Parece ótimo, não é?!";
	mes "Espero que você volte para tingir mais lindos chapéus!";
	delitem .@dyeID,1; // Mistura de Tinta Misteriosa.
	delitem .@myHG,1;  // Seu Chapéu.
	getitem .@dyHG,1;  // Chapéu tingido.
	close;
}
