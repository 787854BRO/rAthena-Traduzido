		mes "[Funcionário do Hotel]";

	}

//===================== Função da Estalagem =========================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "Bem-vind"+(Sex?"o":"a")+"(a) à";
	mes getarg(1) + ".";
	mes "Em que posso ajudar?";
	next;
	switch(select("Salvar:Descançar -> 5000 zenys:Cancelar")) {
	case 1:
		mes .@npc_name$;
		mes "Seu Ponto de Retorno";
		mes "foi salvo.";
		mes "Obrigado e";
		mes "por favor, volte sempre.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "Sinto muito, mas a taxa de serviço é de 5.000 zenys. Por favor, certifique-se de ter dinheiro suficiente para fazer o check-in na próxima vez, ok?";
			close;
		}
		mes "Obrigado!";
		mes "Eu espero que você";
		mes "aproveite sua estadia~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
