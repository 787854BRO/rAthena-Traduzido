// Bandeiras das Guildas (Fora do Castelo)
//============================================================
-	script	Arsulf#1::OutsideFlagsG1	722,{

	set .@GID, GetCastleData("gefg_cas01",1);

	if (.@GID == 0) {
		mes "[ Decreto Divino do Reino de Rune-Midgard ]";
		mes " ";
		mes "1. Conforme o pacto divino do Reino de Rune-Midgard,";
		mes "nós declaramos que";
		mes "não há um mestre formal neste castelo.";
		mes " ";
		mes "2. Para aquele que conseguir";
		mes "superar todas as provações";
		mes "e destruir o Emperium,";
		mes "o rei o dotará com a";
		mes "posse deste castelo.";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ Voz Ecoando ]";
		mes "Bravo guerreiro...";
		mes "Você deseja retornar ao seu lugar de honra?";
		next;
		if(select("Retornar ao Castelo do Clã.:Sair.") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("gefg_cas01",1)) {
				warp "gefg_cas01",197,36;
				end;
			}
			end;
		}
		close;
	}
	mes "[ Decreto Divino do Reino de Rune-Midgard ]";
	mes " ";
	mes "1. Conforme o pacto divino do Reino de Rune-Midgard,";
	mes "nós declaramos que esse castelo";
	mes "pertence ao Clã ^ff0000"+GetGuildName(.@GID)+"^000000.";
	mes " ";
	mes "2. O Líder do clã ^ff0000"+GetGuildName(.@GID)+"^000000 se chama";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "Se houver alguém que se oponha a esse decreto,";
	mes "prove sua força e honra com uma lâmina de aço em suas mãos.";
	close;

OnRecvCastleG01:
	FlagEmblem GetCastleData("gefg_cas01",1);
	end;
}
