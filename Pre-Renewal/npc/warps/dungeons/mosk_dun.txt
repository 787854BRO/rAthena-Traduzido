mosk_dun02,53,220,4	script	#babayagain	WARPNPC,1,1,{
	end;
OnTouch:
	if (mos_nowinter == 11) {
		donpcevent "Soldado1#mos::OnEnable";
		donpcevent "Soldado2#mos::OnEnable";
		donpcevent "Soldado3#mos::OnEnable";
		mes "[Soldado]";
		mes "Pare aí mesmo!";
		next;
		mes "[Soldado]";
		mes "Nós temos a informação"; 
		mes "de que você tem se comunicado";
		mes "secretamente com";
		mes "Baba Yaga, o Terrível.";
		next;
		mes "[Soldado]";
		mes "Se você tiver algo a dizer,";
		mes "fale com o Csar.";
		mes "Venha conosco.";
		close2;
		donpcevent "Soldado1#mos::OnInit";
		donpcevent "Soldado2#mos::OnInit";
		donpcevent "Soldado3#mos::OnInit";
		set mos_nowinter,12;
		warp "mosk_in",125,92;
		end;
	}
	else if (mos_nowinter == 12 || mos_nowinter == 13) {
		donpcevent "Soldado1#mos::OnEnable";
		donpcevent "Soldado2#mos::OnEnable";
		donpcevent "Soldado3#mos::OnEnable";
		mes "[Soldado]";
		mes "Nós estavamos";
		mes "esperando que você aparecesse novamente.";
		mes "Qual foi o negócio com o Csar?";
		next;
		if(select("Fugir.:Explicar a situação.") == 1) {
			mes "["+strcharinfo(0)+"]";
			mes "Eu deveria dar o fora daqui...";
			close2;
			warp "mosk_dun02",100,100;
		}
		else {
			mes "-Você explica o que aconteceu até agora.-";
			next;
			mes "[Soldado]";
			mes "Se você tiver algo a dizer,";
			mes "fale com o Csar.";
			mes "Mas se deixar este lugar,";
			mes "você será considerado nosso inimigo.";
			close2;
		}
		donpcevent "Soldado1#mos::OnInit";
		donpcevent "Soldado2#mos::OnInit";
		donpcevent "Soldado3#mos::OnInit";
		end;
	}
	else {
		warp "mosk_in",215,41;
		end;
	}
}
