//Retorno: 1:vitória, 0:empate, -1:derrota
function	script	F_KO_Survival_rps	{
	setarray .@rps$[1], "Tesoura", "Pedra", "Papel";
	setarray .@emote[1], ET_SCISSOR, ET_ROCK, ET_WRAP;  //pode ser customizado no iRO(?)

	mes "Pedra! Papel! Tesoura!";
	next;
	set .@pc, select("Dar tesoura.:Dar pedra.:Dar papel.");
	set .@npc, rand(1,3);
	emotion .@emote[.@pc], playerattached();
	emotion .@emote[.@npc];
	mes "= Resultados do Jogo =";
	mes "Máquina: " + .@rps$[.@npc] + ".";
	mes "Você: " + .@rps$[.@pc] + ".";
	next;
	if ((.@npc == 1 && .@pc == 2) || (.@npc == 2 && .@pc == 3) || (.@npc == 3 && .@pc == 1)) {
		mes "Você Venceu";
		return 1;
	} else if (.@pc == .@npc) {
		mes "Empate";
		return 0;
	} else if ((.@npc == 1 && .@pc == 3) || (.@npc == 2 && .@pc == 1) || (.@npc == 3 && .@pc == 2)) {
		mes "Você Perdeu";
		return -1;
	} else {
		mes "Ocorreu um erro desconhecido.";
		mes "Por favor, entre em contato com o suporte ao jogador.";
		close;
	}
}
