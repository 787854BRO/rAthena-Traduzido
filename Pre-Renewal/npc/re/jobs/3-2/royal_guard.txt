		}
		close;
	case 3:
		mes "1~5";
		input .@input;
		next;
		if (.@input < 1 || .@input > 5) {
			mes "Defina o item para ajustar o Feiticeiro.";
			mes "Você só pode definir um número entre 1 e 5.";
		} else {
			set job_soc, .@input;
			mes "Concluído.";
		}
		close;
	}
}
