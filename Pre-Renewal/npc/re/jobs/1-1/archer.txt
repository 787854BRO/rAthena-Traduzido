payon_in02,64,71,4	script	Arqueiro#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Arqueiro]";
			mes "Ei, eu conheço você.";
			mes "Você fez este teste";
			mes "antes, não foi?";
			next;
			mes "[Arqueiro]";
			mes "Ah, você deve ter ido";
			mes "a Valhalla renascer.";
			mes "Uau, isso é tão impressionante!";
			next;
			if (getskilllv("NV_BASIC") < 9) {
				mes "[Arqueiro]";
				mes "Err...";
				mes "É melhor você aprender todas as habilidades básicas antes de se tornar um Arqueiro.";
				next;
				mes "[Arqueiro]";
				mes "Tudo bem, até mais.";
				close;
			}
			mes "[Arqueiro]";
			mes "Bem, então eu não";
			mes "preciso dizer mais nada.";
			mes "Eu sei que você será um grande Arqueiro...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_High;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Arqueiro]";
			mes "Embora não haja uma recompensa";
			mes "especial para você desta vez, espero que você entenda. Cuide-se.";
			close;
		}else{
			mes "[Arqueiro]";
			mes "Oh...?";
			mes "Ei, o que";
			mes "você está fazendo aqui...?";
			next;
			mes "[Arqueiro]";
			mes "Eu posso dizer que você não está pronto para ser um Arqueiro. Parece que você ainda deve fazer";
			mes "outra coisa...";
			close;
		}
	}
	mes "[Arqueiro]";
	mes "Prazer em conhecê-lo. Como posso ajudá-lo?";
	next;
	switch(select("Quero me tornar um Arqueiro.:Quais os requisitos, por favor.:Nada, obrigado.")) {
	case 1:
		mes "[Arqueiro]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Archer)
				mes "Haha, você está brincando comigo...";
			else {
				mes "Sinto muito, mas apenas Aprendizes podem mudar de classe.";
				mes "Você já tem a sua própria classe, não tem?";
			}
			close;
		}else{
			mes "Você é... "+strcharinfo(0)+", certo?";
			next;
			if (getskilllv("NV_BASIC") < 9) {
				mes "[Arqueiro]";
				mes "Bem, você não está no nível de habilidade certo.";
				mes "Seu nível de classe deve ser pelo menos ^4A4AFF10^000000 e seu nível de Habilidades Básicas deve atingir o ^4A4AFFnível 9^000000.";
				next;
				mes "[Arqueiro]";
				mes "Porque um Arqueiro precisa de uma concentração extremamente alta, então nós não aceitamos aqueles que têm pouca paciência.";
				close;
			}
			mes "[Arqueiro]";
			mes "Se nível de habilidades básicas agora é o suficiente...";
			mes "....Hm~~ então você está pronto para se tornar um Arqueiro. Vamos lá então, eu vou dar o próximo passo imediatamente.";
			next;
			mes "[Arqueiro]";
			mes "Parabéns! Você agora é um Arqueiro! Além disso, esperamos que você participe ativamente de muitos programas para o renascimento da Guilda dos Arqueiros.";
			mes "Ah, itens chegaram do Departamento de Produção. Aqui, pegue estes! São todos seus!";
			callfunc "Job_Change",Job_Archer;
			getitem 1742,1;		// Arco Composto de Iniciante
			getitem 12004,1;	// Aljave
			getitem 12009,1;	// Aljave de Flechas de Prata
			getitem 12008,1;	// Aljave de Flechas de Fogo
			next;
			mes "[Arqueiro]";
			mes "Tendo um arco e flechas, agora você se tornou um verdadeiro Arqueiro.";
			mes "Se você abrir o Aljave de Flechas, poderá ver que existem flechas nele e você poderá equipá-las.";
			mes "Bem, espero ouvir boas notícias de você. É hora de dizer adeus.";
			mes "Tchau.";
			close;
		}
	case 2:
		mes "[Arqueiro]";
		mes "Um Arqueiro usando um arco tem várias habilidades e talentos.";
		mes "A maior habilidade de um Arqueiro é atacar os inimigos a longa distância.";
		next;
		mes "[Arqueiro]";
		mes "Embora um Arqueiro tenha um HP mais fraco, ele pode atirar em inimigos a longa distância,";
		mes "então é mais seguro ser um Arqueiro em uma batalha real.";
		next;
		mes "[Arqueiro]";
		mes "Embora um Arqueiro em Ragnarok tenha o uma quantidade de HP baixa, ele ou ela tem alta precisão e taxa de ataque com arcos, e podem derrotar um monstro sem nem mesmo serem tocados.";
		next;
		mes "[Arqueiro]";
		mes "^8C2121Um Arqueiro pode mudar de classe para Caçador.^000000";
		mes "^8C2121Além de Caçador, se você é homem, você poderá mudar de classe para Bardo e se você for mulher, poderá mudar de classe para Odalisca.^000000";
		close;
	case 3:
		mes "[Arqueiro]";
		mes "Se você tiver alguma dúvida, sinta-se à vontade para vir e me perguntar.";
		close;
	}
}
