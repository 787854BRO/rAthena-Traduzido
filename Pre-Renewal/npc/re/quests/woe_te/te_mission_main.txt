			mes "Eles me disseram que um monte de "+ .@string$[2] +" está faltando no castelo em "+ .@fortress$ +".";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "O Gatuno provavelmente aproveitou as horas de GdE.";
			mes "É uma solicitação para encontrar ^4d4dff"+ .@num_items +" ou mais de "+ .@string$[2] +"^000000.";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Já que a região TE tem uma atmosfera mais casual,";
		mes "não é divertido fazer apenas batalhas GdE, certo?";
		mes "Então, eu trouxe ^4d4dffalgumas missões que você poderá fazer durante as horas de batalha GdE^000000~";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Você pode aceitar missões ^4d4dffcomeçando 1 hora antes do início da GdE e até o fim das batalhas de cerco TE^000000,";
		mes "e cada missão tem ^4d4dffum limite de tempo de 4 horas^000000 para ser concluída.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Simplificando, você aceita uma missão minha durante as horas de batalha GdE e";
		mes "^4d4dffvolta para me dar um relatório dentro do limite de tempo^000000.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Experimente por diversão.";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Entendo.";
		mes "Volte quando quiser, se mudar de ideia.";
		mes "Mas talvez eu não esteja disponível. Hehehe.";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "Parece que você ainda não completou a missão?";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Você quer cancelar por acaso?";
		next;
		switch( select( "Não.","Sim." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Boa sorte!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Como você queria, esta missão foi cancelada.";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "Hmm, você ainda tem algum tempo, mas ainda não é a hora da GdE.";
		mes "Eu não acho que você deva continuar.";
		mes "Você deveria cancelar esta missão. Eu diria que ela falhou.";
		next;
		switch( select( "Quero continuar.","Sim, vou cancelar." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Não acho que valha a pena. Bem, boa sorte!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Como você queria, esta missão foi cancelada.";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "Aceito.","Não aceito." ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dffO limite de tempo é de 4 horas.^000000";
		mes "Cumpra a missão dentro desse tempo e me informe.";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Tente outra missão então.";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Anel de Rune, Manto de Rune, Manteau de Mana, Botas de Mana
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "Você não encontrou nada.";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "Você encontrou '"+ getitemname(.@item_id[ .@miss_item01 ]) +" !'";
		}
	}
	close;

S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + " ^4d4dffVerificar resultado^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
