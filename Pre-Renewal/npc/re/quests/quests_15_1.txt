function checkstatusquest {
	switch( getarg(0) ) {
	case 4297:
		if (lhz_curse >= 31)// Missão do Espírito Amaldiçoado
			return true;
		break;
	case 4298:
		if (lhz_boss >= 43)// Missão do Presidente
			return true;
		break;
	case 4299:
		if (yuno_hist >= 10)// Missão da História das Ruínas de Juperos
			return true;
		break;
	case 4300:
		if (metto_q >= 9)// Cientista Maluco
			return true;
		break;
	case 4301:
		if (hg_bio >= 9)// Missão da Biopesquisa
			return true;
		break;
	case 4302:
		if (hg_odin >= 60)// Inspeção no Templo de Odin
			return true;
		break;
	}
	return false;
}
