L_CheckHunting:
	for (set .@quest,getarg(0)+1; .@quest<getarg(1)+getarg(2); set .@quest,.@quest+1) {
		if (checkquest(.@quest,HUNTING) == 2) {
			mes "[Reno]";
			mes "Concluída a missão de caça à coleção "+getarg(3)+".";
			next;
			setquest getarg(0);
			erasequest .@quest;
			getitem getarg(4),getarg(5);
			mes "Você recebeu "+getarg(5)+" "+getitemname(getarg(4))+"s como recompensa da coleção "+getarg(3)+".";
			close;
		}
	}
	return;
