//============================================================
-	script	Armazém do Éden#00::pggc	-1,{
	mes "Isto é um Armazém para Armazenamento de Mercadorias.";
	mes "Uma mensagem escrita em um pedaço de papel.";
	next;
	mes "+ Esse Armazém é exclusivamente +";
	mes "+ para membros do Grupo do Éden +";
	mes "Se você quiser usar este";
	mes "armazém, por favor verifique";
	mes "o seguinte:";
	next;
	mes "1.Você é um membro";
	mes " do Grupo do Éden?";
	mes "2.Você já aprendeu";
	mes " habilidades básicas suficientes?";
	mes "3.A taxa para o Armazém é de";
	mes " ^4d4dff500 zenys^000000!";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "Você precisa inserir zeny para usar o Armazém.";
		mes "Custo: 500 zenys.";
		mes "Você gostaria de usá-lo?";
		next;
		switch(select("Usar o Armazém.:Cancelar.")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "O Armazém não está funcionando para mim.";
						mes "Talvez eu ainda não esteja qualificado para usar o Armazém.";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "Eu não tenho zeny o suficiente.";
				mes "Preciso de pelo menos 500 zenys para usar o Armazém.";
				close;
			case 2:
				mes "Vou usá-lo na próxima vez.";
				close;
		}
	}
	mes "Eu preciso de um Emblema do Éden para usar este Armazém.";
	close;
}
