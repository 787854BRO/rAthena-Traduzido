moc_para01,38,38,3	script	Missões [26 - 40]	857,{

		mes "Você não é um membro qualificado. Você não pode acessar o Quadro de Avisos.";
		close;
	}
	mes "[Quadro de Missões do Nv.26 ~ 40]";
	mes "- Muitas missões estão -";
	mes "- no Quadro de Avisos. -";
	next;
	if ((BaseLevel < 26) || (BaseLevel > 40)) {
		mes "Seu nível não é qualificado. Você não pode acessar o Quadro de Avisos.";
		close;
	}
	switch(select("Missão de Caça:Missão de Entrega:Missão de Coleta:Missão de Escolta")) {
	case 1:
		mes "[Conselho de Missão de Caça]";
		mes "- Os detalhes da Missão de Caça -";
		mes "- estão no quadro. -";
		next;
		switch(select("Caçar Esporo Venenoso:Caçar Fumacento:Caçar Salgueiro Ancião(Acima do Nv.30):Caçar Coco(Acima do Nv.35):Caçar Sapo de Thara(Acima do Nv.38)")){
			case 1: callsub L_Quest,8266,0,"Caçar Esporo Venenoso","Sponiac","Geffen","Cace 10 Esporos Venenosos";
			case 2: callsub L_Quest,8267,0,"Caçar Fumacento","Sponiac","Geffen","Cace 10 Fumacentos";
			case 3: callsub L_Quest,8268,30,"Caçar Salgueiro Ancião","Sponiac","Geffen","Cace 10 Salgueiros Anciões";
			case 4: callsub L_Quest,8269,35,"Caçar Coco","Sponiac","Geffen","Cace 10 Cocos";
			case 5: callsub L_Quest,8270,38,"Caçar Sapo Thara","Sponiac","Geffen","Cace 10 Sapos de Thara";
		}
	case 2:
		mes "[Quadro de Missão de Entrega]";
		mes "- Os detalhes da missão de entrega -";
		mes "- estão no quadro. -";
		next;
		switch(select("Entregar Inseticida:Entregar Mel:Entregar Cobertor(Acima do Nv.35)")) {
			case 1: callsub L_Quest,8271,0,"Entregar um Inseticida","Dieshin Man","Ruínas de Morroc","Obtenha um 'Inseticida' do Dieshin Man e entregue-o a sudeste de Payon";
			case 2: callsub L_Quest,8272,0,"Entregar Mel","Dieshin Man","Ruínas de Morroc","Obtenha 'Mel' do Dieshin Man e entregue-o a oeste da abadia de St. Capitolina";
			case 3: callsub L_Quest,8273,35,"Entregar Cobertor","Dieshin Man","Ruínas de Morroc","Obter Cobertores do Dieshin Man e entregar no campo a sul de Lutie";
		}
	case 3:
		mes "[Quadro de Missão de Coleta]";
		mes "- Os detalhes da Missão de Coleta -";
		mes "- estão no quadro. -";
		next;
		switch(select("Coletar Ossos:Coletar Couro:Coletar Peles(Acima do Nv.30):Coletar Caudas(Acima do Nv.32):Coletar Biscoitos(Acima do Nv.35):Coletar Bigodes(Acima do Nv.35)")) {
			case 1: callsub L_Quest,8274,0,"Coletar Ossos","Colete '10 Ossos' e entregue-os a Dashia";

	end;

	//QuestID,ReqLvl,Quest,Client,Loc,Desc
	.@string$ = getarg(2);
	if (BaseLevel < getarg(1,26)) {
		mes "O seu nível não satisfaz os requisitos. Não pode acessar a missão '"+ .@string$ +"'.";
		close;
	}
	mes "["+ .@string$ +" !!]";
	mes "Cliente : "+getarg(3);
	mes "Localização do cliente : "+getarg(4);
	mes "Missão solicitada : "+getarg(5);
	mes " ";
	mes "Aceitará esta missão?";
	next;
	if(select("Não, não aceitarei.:Sim, vou aceitá-la.") == 1) {
		mes "Não aceitou a Missão "+ .@string$ +".";
		close;
	}
	if (checkquest(getarg(0)) > -1) mes "Você já está a cumprir esta missão.";
	else {
		setquest getarg(0);
		mes "Aceitou a Missão "+ .@string$ +".";
	}
	mes "Pode receber uma recompensa do cliente depois de terminar a Missão.";
	next;
	mes "Para mais detalhes, por favor pressione 'Alt + U', para obter mais informações da Janela de Missões.";
	close;
}
