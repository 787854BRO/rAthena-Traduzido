S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest(getarg(0));
	setquest(getarg(1));
	if (getarg(2,0))
		getexp getarg(2),getarg(3);
	return;

S_Erasequest:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 0:
		case 1:
			mes "[Ragi]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "O restante da história de "+ getarg(.@i+2) +"... Tudo bem, pedirei para outro aventureiro.";
				break;
			case 2:
			case 3:
				mes "Pedirei a outros aventureiros a história de "+ getarg(.@i+2) +".";
				break;
			}
			break;
		case 2:
			mes "[Ragi]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "Você ouviu a história de "+ getarg(.@i+2) +"? Obrigado. Prometo compor uma boa música.";
				getexp 1100000,300000;
				break;
			case 2:
				mes "A história de Morroc é digna de uma canção? Você fez um ótimo trabalho.";
				getexp 1100000,200000;
				break;
			case 3:
				mes "Como foi sua aventura? Foi divertida? Me conte sua história de aventura algum dia.";
				getexp 1100000,200000;

				break;
			}
		}
		erasequest .@quest_id;
		next;
	}
	return;
