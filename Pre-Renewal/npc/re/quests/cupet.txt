function	script	treinador_pet_fofos	{
	// getarg(0) -> .@tame_id Array
	// getarg(1) -> .@tame_amount Array
	// getarg(2) -> .@hunt_id Array
	// getarg(3) -> .@hunt_amount Array
	// getarg(4) -> .@mob_id Array
	// getarg(5) ->  Attempt Variable
	// getarg(6) -> .@tame_gets Variable
	// getarg(7) -> .@hunt_id2 Array
	// getarg(8) -> .@hunt_amount2 Array

	mes "[Treinador de Pets Fofos]";
	mes "Olá~! Eu sou o Treinador de Pets Fofos,";
	mes "sou o responsável pelas relações públicas";
	mes "do sistema de Evoluçã de Novos Pets.";
	next;
	mes "[Treinador de Pets Fofos]";
	mes "Se você tiver alguma dúvida sobre";
	mes "o Sistema Pet Bonito, eu estou aqui para guiá-lo.";
	mes "Deixe-me saber qual é o item";
	mes "de domesticar que deseja,";
	mes "então eu lhe direi qual o monstro";
	mes "que você poderá domar com esse item.";
	next;
	mes "[Treinador de Pets Fofos]";
	mes "Também vou lhe explicar quais os materiais";
	mes "que você precisa levar a fim de";
	mes "obter o item domesticador.";
	next;
	mes "[Treinador de Pets Fofos]";
	mes "Se você já trouxe os materiais";
	mes "para trocar pelo item de domesticar,";
	mes "poderei trocá-los pelo item imediatamente.";
	next;
	mes "[Treinador de Pets Fofos]";
	mes "No entanto, há um limite";
	mes "para a troca de itens de domesticar.";
	mes "Você só pode obter itens de domesticar";
	mes getarg(6) + " vezes. Então, por favor certifique-se";
	mes "quantas vezes você já trocou.";
	next;
	mes "[Treinador de Pets Fofos]";
	mes "Então, qual o item de dosmesticar que você quer?";
	mes "Por favor, escolha um da lista.";
	next;
	
	// Cria o Sistema de Menu
	set .@size, getarraysize(getarg(0));
	for (set .@a, 0; .@a < .@size; set .@a, .@a + 1)
		set .@menu$, .@menu$ + (.@menu$ == "" ? "" : ":") + getitemname(getelementofarray(getarg(0), .@a));
	
	// Consultar a escolha do jogador
	set .@choice, select(.@menu$) - 1;

	// Variáveis do Armazém (menos pesquisa)
	set .@tame_id, getelementofarray(getarg(0), .@choice);
	set .@tame_amount, getelementofarray(getarg(1), .@choice);
	set .@hunt_id, getelementofarray(getarg(2), .@choice);
	set .@hunt_amount, getelementofarray(getarg(3), .@choice);
	set .@mob_id, getelementofarray(getarg(4), .@choice);
	set .@hunt_id2, getelementofarray(getarg(7), .@choice);
	set .@hunt_amount2, getelementofarray(getarg(8), .@choice);

	dispbottom "ID Comida: " + .@tame_id;
	dispbottom "Quantidade Comida: " + .@tame_amount;
	dispbottom "ID Item de Caça: " + .@hunt_id;
	dispbottom "Nome do Item de Caça: " + getitemname(.@hunt_id);
	dispbottom "Quantidade de Caça: " + .@hunt_amount;
	dispbottom "ID do Monstro: " + .@mob_id;
	if (.@hunt_id2) { 
	dispbottom "ID de Caça 2: " + .@hunt_id2;
	dispbottom "Quantidade de Caça 2: " + .@hunt_amount2;
		set .@hunt2_count, countitem(.@hunt_id2);
	}
	if (countitem(.@hunt_id) >= .@hunt_amount && countitem(6083) > 0 && .@hunt2_count >= .@hunt_amount2) {
		mes "[Treinador de Pets Fofos]";
		mes "Ah, você juntou todos os itens";
		mes "com seus amigos. Agora você pode";
		mes "trocá-los por um " + getitemname(.@tame_id) + ".";
		mes "Você quer trocar comigo?";
		next;
		if  (select("Trocar:Não Trocar") == 2) { 
			mes "[Treinador de Pets Fofos]";
			mes "Você não quer? Então até logo.";
			close;
		}
		delitem .@hunt_id, .@hunt_amount;
		delitem 6083, 1;
		if (.@hunt_id2)
			delitem .@hunt_id2, .@hunt_amount2;
		set getarg(5), getarg(5) + 1;
		getitem .@tame_id, .@tame_amount;
		
		mes "[Treinador de Pets Fofos]";
		mes "Sábia escolha.";
		mes "Espero que você e seu animal de estimação se deêm bem.";
		close;
	} else {
		mes "[Treinador de Pets Fofos]";
		mes "Você escolheu " + getitemname(.@tame_id) + ".";
		mes "Você pode usá-lo para domar";
		mes "o monstro ^FF0000" + strmobinfo(1,.@mob_id) + "^000000.";
		next;
		mes "[Treinador de Pets Fofos]";
		mes "Traga ^FF0000" + .@hunt_amount + " " + getitemname(.@hunt_id) + "^000000s " + (.@hunt_id2 ? ", ^FF0000" + .@hunt_amount2 + " " + getitemname(.@hunt_id2) : "") + " ^000000 e ^FF00001 Capsula de Bonecos^000000";
		mes "então você pode trocá-los";
		mes "por uma " + getitemname(.@tame_id) + ".";
		close;
	}
}
