c_tower1,232,222,5	script	Belljamin Botão#ct	968,{
	mes "[Belljamin Botão]";
	mes "Os monstros aqui são capazes de controlar o tempo.";
	if( BaseLevel < 150 ){
		mes "Volte quando tiver alcançado o ^0000FFNível 150^000000.";
		close;
	}
	mes "Tempo, memórias, ambientes... tudo pode ser distorcido aqui.";
	next;
	mes "[Belljamin Botão]";
	mes "Estou implorando para que não pergunte mais sobre isso. Pode colocar sua vida em perigo.";
	next;
	.@menu = select("Não perguntar.: Ir ao 2ºandar da Torre Distorcida.:Ir ao 3º andar da Torre Distorcida.");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[Belljamin Botão]";
			mes "Tenha cuidado com a sua escolha.";
			next;
			mes "[Belljamin Botão]";
			mes "Preciso de ^0000FF10.000z^000000 para deixar você entrar. É um lugar muito perigoso para se estar.";
			next;
			switch( select( "Sair:Pagar 10,000z.:Por que você precisa de dinheiro?" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[Belljamin Botão]";
						mes "Sério? Você não tem dinheiro e me pediu para te levar lá?";
						close;
					}
					Zeny -= 10000;
					mes "[Belljamin Botão]";
					mes "Não tenho certeza de como você vai controlar o tempo aí, mas seja o mais cuidadoso possível.";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[Belljamin Botão]";
					mes "Estou aqui o dia todo pela segurança dos aventureiros. Eu sempre os alerto sobre o perigo. É por isso que preciso de alguma forma de pagamento.";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[Belljamin Botão]";
	mes "Ufa...Você fez uma boa escolha. Este lugar é brutal, então você provavelmente deveria voltar para uma área segura.";
	close;
}
