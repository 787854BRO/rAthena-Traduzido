pay_arche,44,124,5	script	Mariaju	4_F_JOB_BLACKSMITH,{
	mes "[Mariaju]";
	if (BaseLevel < 60) {
		mes "Seu nível de base é muito baixo, volte quando alcançar o nível 60+.";
		close;
	}
	mes "^ff0000Se você tentar domar qualquer monstro, isso não contará como uma morte. Por favor, tome cuidado.^000000";
	next;
	.@day = gettime(DT_DAYOFWEEK);
	if (.@day != SATURDAY && .@day != SUNDAY) {
		mes "[Mariaju]";
		mes "Oh, um curioso espectador? Eu sou Mariaju, a pessoa encarregada dos fins de semana entre os dias da semana.";
		next;
		mes "[Mariaju]";
		mes "O que é a Cripta? Na verdade, é um espaço especialmente projetado para novos aventureiros treinarem.";
		next;
		mes "^0000FFOlhando ao redor, fica óbvio que Mariaju está falando com você.^000000";
		next;
		mes "[Mariaju]";
		mes "Para aqueles que não têm um grupo nos fins de semana, este lugar é um campo de caça e ganho de experiência para aventureiros solitários!";
		next;
		mes "[Mariaju]";
		mes "Além disso, o ganho de experiência aqui é muito generoso.";
		next;
		select( "Qual é o horário de abertura?" );
		mes "[Mariaju]";
		mes "A masmorra estará aberta durante:";
		mes "^0000FFSábado 00:00 ~ Domingo 23:59^000000";
		next;
		mes "[Mariaju]";
		mes "Espero ver você quando o fim de semana chegar~";
		close;
	}
	.@md_name$ = "Cripta";

	if (getcharid(1) < 1) {
		mes "[Mariaju]";
		mes "Você precisa formar um grupo com um membro ou mais.";
		close;
	}
	switch( checkquest(12378,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		switch( checkquest(12377,PLAYTIME) ) {
		case -1:
		case 2:
			mes "[Mariaju]";
			mes "Desculpe, mas o período de entrada para a " + .@md_name$ + " expirou.";
			mes "Descanse enquanto espera o Calabouço ficar disponível novamente.";
			close;
		case 0:
		case 1:
			break;
		}
		break;
	case 2:
		erasequest 12377;
		erasequest 12378;
		mes "[Mariaju]";
		mes "O tempo de espera expirou.";
		mes "Você pode entrar novamente no Calabouço Memorial.";
		close;
	}

	if (is_party_leader() == true)
		.@menu$ = "Preparar Calabouço Memorial";

	mes "[Mariaju]";
	mes "Acho que você já está praticamente pronto.";
	mes "Agora, você gostaria de entrar agora?";
	next;
	switch( select( .@menu$, "Entrar no Cripta", "Cancelar" ) ) {
	case 1:
		instance_create(.@md_name$);
		mes "[Mariaju]";
		mes "O Calabouço Memorial foi criado.";
		mes "Agora você pode entrar.";
		close;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			end;
		case IE_NOINSTANCE:
			mes "Não há nenhum Calabouço Memorial registrado.";
			close;
		case IE_OTHER:
			mes "Ocorreu um erro desconhecido.";
			close;
		case IE_OK:
			mes "[Mariaju]";
			mes "Vamos lá!";
			mapannounce "pay_arche", "" + strcharinfo(0) + " do grupo, " + getpartyname( getcharid(1) ) + ", está entrando na " + .@md_name$ + ".", bc_map, 0xFF99;
			if (isbegin_quest(12377) == 0)
				setquest 12377;
			if (isbegin_quest(12378) == 0)
				setquest 12378;
			end;
		}
		end;
	case 3:
		mes "[Mariaju]";
		mes "Você não se preparou?";
		mes "Acho que não tem muito o que fazer então.";
		close;
	}
}

		break;

		mes "Por favor, volte quando tiver materiais suficientes.";
		close;

		mes "Por favor, volte quando tiver Zeny suficiente.";
		mes F_InsertComma(.@zeny) + "z para ser mais exato.";
		close;
	}
	if (checkweight(23144,1) == 0) {
		mes "Você tem muitos tipos de itens. Libere espaço no seu inventário.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Não é possível continuar porque você está carregando muitos itens pesados. Reduza seu peso total liberando espaço no inventário.";
		close;
	}
	mes "Confirma gastar " + F_InsertComma(.@zeny) + "z para embalar 5 <ITEM>" + getitemname(.@item) + "<INFO>" + .@item + "</INFO></ITEM> em <ITEM>" + getitemname(.@box) + "<INFO>" + .@box + "</INFO></ITEM>?";
	next;
	if (select( "Sim", "Não" ) == 2) {
		mes "[Atendente de Suprimentos]";
		mes "Volte sempre que precisar dos meus serviços de embalagem!";
		close;
	}
	Zeny -= .@zeny;
	delitem .@item,5;
	getitem .@box,1;
	end;
}
