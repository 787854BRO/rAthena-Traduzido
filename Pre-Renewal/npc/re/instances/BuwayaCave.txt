ma_fild02,312,317,5	script	Guarda#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[Guarda]";
		mes "Este lugar é muito perigoso, e pessoas com ^ff0000nível menor que 130^000000"; 
		mes "não têm permissão para entrar.";
        mes "Por favor, retorne depois.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"Caverna de Buawaya";

	if (!.@party_id) {
		mes "[Guarda]";
		mes "Buwaya é um monstro perigoso.";
		mes "Por favor, retorne apenas quando tiver";
		mes "formado uma equipe com mais pessoas.";
		close;
	}
	if (is_party_leader() == true) {
		mes "[Guarda]";
		mes "Este lugar é perigoso. Por favor, volte.";
		mes "Se você não tem nada para tratar aqui, faça o favor de voltar.";
		next;
		while(1) {
			switch(select("Por que é tão perigoso?:Estou aqui para caçar Buwaya.:Terminar conversa.")) {
			case 1:
				mes "[Guarda]";
				mes "Há pouco tempo atrás, tem havido casos"; 
				mes "em que moradores das redondezas desapareceram.";
				mes "Após as investigações dos desaparecimentos,"; 
				mes "descobrimos que o gigante monstro Buwaya"; 
				mes "foi o responsável.";
				next;
				mes "[Guarda]";
				mes "Diversos soldados e mercenários foram"; 
				mes "encaminhados à caça de Buwaya, mas nunca"; 
				mes "obtiveram sucesso.";
				next;
				mes "[Guarda]";
				mes "Buwaya ainda está sequestrando"; 
				mes "os moradores nos dias de hoje.";
				mes "Por favor, reuna algumas pessoas para destruí-lo.";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// cai através
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[Guarda]";
					mes "Buwaya continua escondido.";
					mes "Mesmo que você entre agora, Bawaya não vai aparecer.";
					mes "Por favor, volte mais tarde.";
					close;
				} else {
					erasequest 4229;
					// cai através
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[Guarda]";
					mes "Nome do Grupo: "+getpartyname(.@party_id)+".";
					mes "Líder do Grupo: "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000, não pode ser aberta agora.";
					mes "Por favor, tente mais tarde.";
					close;
				}
				mes "[Guarda]";
				mes "^0000ff"+.@md_name$+"^000000, vou abrir o túnel por algum tempo.^000000.";
				mes "Desejamos boa sorte.";
				close;
			case 3:
				mes "[Guarda]";
				mes "Esse lugar é perigoso, então você não pode entrar.";
				close;
			}
		}
	} else {
		mes "[Guarda]";
		mes "Preciso conversar com o líder do grupo.";
		mes "Por favor, chame o representante do grupo.";
		close;
	}
}

ma_fild02,315,323,0	script	Entrada da Caverna#buwaya	45,2,2,{
	end;
OnTouch:
	if (BaseLevel < 130) {
		mes "[Guarda]";
		mes "Este lugar é perigoso,";
		mes "e pessoas com ^ff0000nível menor do que 130^000000 não tem permissão para entrar.";
      	mes "Por favor, retorne num outro momento.";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// cai através
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[Guarda]";
		mes "Buwaya está dormindo no momento.";
		mes "Mesmo que você entre agora, Bawaya não acordará.";
		mes "Por favor, retorne numa outra oportunidade.";
		close;
	} else {
		erasequest 4229;
		// cai através
	}
	switch(select("Entrar.:Retornar.")) {
	case 1:
		switch(instance_enter("Caverna de Buawaya")) {
		case IE_OTHER:
			mes "[Guarda]";
			mes "Oh, agora não é uma boa hora.";
			mes "Por favor, tente novamente mais tarde.";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[Guarda]";
			mes "Este lugar é perigoso. Por favor, não entre.";
			close;
		case IE_OK:
            mapannounce "ma_fild02",getpartyname(getcharid(1))+", membro do grupo "+strcharinfo(0)+" entrou na instância Caverna de Buwaya para caçar Buwaya.",bc_map,"0x00ff99";
			setquest 4229;
			//warp "1@ma_c",35,57;
			end;
		default:
			mes "- Não foi possível entrar devido a uma força não identificada. -";
			close;
		}
	case 2:
		mes "[Guarda]";
		mes "Você acaba de preservar sua vida.";
		mes "É sempre bom pensar duas vezes antes de agir.";
		close;
	}
}
