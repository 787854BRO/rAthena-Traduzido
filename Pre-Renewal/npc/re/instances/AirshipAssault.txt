dali02,142,81,3	script	Dispositivo Dimensional#air1	PORTAL,{
	if (checkquest(15050,PLAYTIME) == 2) {
		erasequest 15050;// Instância: Invasão ao Aeroplano
		erasequest 15051;
	}
	else if (checkquest(15051,PLAYTIME) == 2) {// Instância: Invasão ao Aeroplano
		mes "^ff0000A Instância não pode ser reentrada dentro de 23 horas após sua última entrada.^000000";
		close;
	}
	.@party_id = getcharid(1);
	.@p_name$ = getpartyname(.@party_id);
	.@md_name$ = "Invasão ao Aeroplano";

	if (BaseLevel < 125) {
		mes "^ff0000Jogadores abaixo do nível 125 não podem entrar nesta instância.^000000";
		close;

	if (.@party_id < 1) {
		mes "^ff0000Para entrar, você deve criar ou entrar em um grupo de pelo menos 1 membro.^000000";
		close;
	}
	if (is_party_leader() == false) {
		mes "^ff0000Esta Instância é apenas para uma pessoa. Somente o líder do grupo pode entrar.^000000";
		close;
	}
	.@s = select("Ativar o Dispositivo.", "Usar o Dispositivo.", "Cancelar");
	if (is_party_leader(.@party_id) == false) {
		mes "^ff0000Esta Instância é apenas para uma pessoa. Somente o líder do grupo pode entrar.^000000";
		close;
	}
	switch(.@s) {
	case 1:
		if (instance_create(.@md_name$) < 0) {
			mes "Nome do Grupo: " + getpartyname(getcharid(1));
			mes "Nome do Personagem: " + strcharinfo(0);
			mes "^0000ff" + .@md_name$ + "^000000 - Falha na criação!";
			close;
		}
		mes "^ff0000Após o dispositivo dimensional ser formado^000000";
		mes "^ff0000clique para entrar novamente.^000000";
		close;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_OK:
			if (isbegin_quest(15050) == 0) {
				setquest 15050;// Instância: Invasão ao Aeroplano
				setquest 15051;// Instância: Invasão ao Aeroplano
			}
			mapannounce "dali02", "" + strcharinfo(0) + ", membro do grupo " + getpartyname(getcharid(1)) + " entrou na instância " + .@md_name$ + ".", bc_map,"0x00ff99";
			specialeffect2 EF_WARP;
			// warp "1@air1",244,73;
			end;
		case IE_NOMEMBER:
			mes "^ff0000Para entrar, você deve criar ou entrar em um grupo de pelo menos 1 membro.^000000";
			close;
		case IE_NOINSTANCE:
			mes "^ff0000O líder do grupo não usou o dispositivo.^000000";
			close;
		case IE_OTHER:
			mes "Ocorreu um erro desconhecido.";// Customizado
			close;
		}
	case 3:
		end;
	}
}
