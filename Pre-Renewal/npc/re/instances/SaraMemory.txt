dali,139,118,4	script	Dispositivo Dimencional#sara	10007,{
	if (BaseLevel < 99) {
		mes "[Aventureiro Leon]";
		mes "Ei! És muito novo para brincares com isso! Volte quando tiver pelo menos o ^0000FFNível 99^000000!";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[Aventureiro Leon]";
		mes "Antes de brincares com isso; porque não vens até aqui e falas comigo?";
		close;
	}

	// 20 horas para recarga.
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"Memórias de Sarah";

		if (!.@party_id) {
			mes "[Aventureiro Leon]";
			mes "Antes de entrar, é preciso criar uma Grupo! Não sabe como? Escreva /organize PARTYNAMEHERE.";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "Inicializar Dispositivo Dimensional.:Usar Dispositivo Dimensional.:Cancelar.";
		else
			set .@menu$, ":Usar Dispositivo Dimensional.:Cancelar.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "Nome do Grupo: "+.@p_name$;
				mes "Líder do Grupo: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+"^000000 - falhadas Reservas!";
				close;
			}
			mes "^FF0000O dispositivo dimensional inicializa corretamente. Use o dispositivo para entrar na fenda do espaço e do tempo.^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "Ocorreu um erro desconhecido.";
				close;
			case IE_NOINSTANCE:
				mes "O calabouço memorial "+.@md_name$+" não existe.";
				mes "O líder do Grupo ainda não gerou o calabouço.";
				close;
			case IE_NOMEMBER:
				mes "Apenas os membros registados podem entrar na instância "+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+", membro do grupo "+.@p_name$+" entrou na instância "+.@md_name$+".",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000O líder do grupo não inicializou o Dispositivo Dimensional.^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[Aventureiro Leon]";
		mes "Acho que dentro de cerca de 20 horas após a sua última entrada na fenda, ela estará suficientemente estável para a atravessar de novo.";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ Memórias de Sarah ] O arrefecimento foi limpo. Agora é possível utilizar novamente o Dispositivo Dimensional.^000000";
		close;
	}
}
