ein_fild04,275,342,5	script	Comandante da Tropa#	4_M_REBELLION3,{
	if (terra_gloria_main < 19) {
		mes "[Comandante da Tropa]";
		mes "Ei, quem está aí?";
		next;
		select("Estou executando minha missão com a unidade Rebelião.");
		mes "[Comandante da Tropa]";
		mes "Este lugar é perigoso e você deve sair rapidamente. Estamos perto da base inimiga.";
		close;
	}
	switch( checkquest(12375,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[Comandante da Tropa]";
		mes "Estamos organizando a situação dentro da base. Ainda não podemos deixar você entrar.";
		close;
	case 2:
		mes "^0000ffParece que a base já foi organizada por dentro. ^000000";
		erasequest 12375;// Manutenção da Base
		break;
	}
	if (isbegin_quest(12374) == 0) {
		mes "[Comandante da Tropa]";
		mes "Quando estivermos prontos para a operação, você entrará comigo.";
		next;
		mes "[Comandante da Tropa]";
		mes "A propósito, você recebeu um relatório do general Julian? Volte depois de receber o relatório.";
		close;
	}
	mes "[Comandante da Tropa]";
	mes "Quando estivermos prontos para a operação, você entrará comigo.";
	next;
	if (terra_gloria_main == 19)
		.@md_name$ = "Base Militar 1";
	else
		.@md_name$ = "Base Militar 2";
	switch( instance_enter(.@md_name$) ) {
	case IE_NOMEMBER:
		mes "^0000ffParece que um aventureiro sem sua unidade (grupo) não pode entrar.^000000";
		close;
	case IE_NOINSTANCE:
		mes "[Comandante da Tropa]";
		mes "O campo de operação é a " + .@md_name$ + " Verifique se a operação foi aplicada corretamente.";
		next;
		mes "^ff0000Ir para o limite externo da Base Militar dos Caça-Coração sem concluir a Base Militar dos Caça-Coração é proibido. Além disso, você não pode entrar em uma Base Militar dos Caça-Coração que já foi concluída. Por favor, verifique seu ponto de operação.^000000";
		next;
		mes "^ff0000Isso acontece quando seu progresso é diferente do líder do grupo ou de outro membro. Nesse caso, encontre outro membro do grupo que possa ir ao mesmo lugar com você.^000000";
		next;
		mes "^ff0000A entrada também é impossível se o Calabouço Memorial não tiver sido criada como campo de operação.^000000";
		close;
	case IE_OTHER:
		mes "Ocorreu um erro desconhecido.";
		close;
	case IE_OK:
		mapannounce "ein_fild04", "" + strcharinfo(0) + " do grupo " + getpartyname( getcharid(1) ) + ", está entrando na " + .@md_name$ + ".", bc_map, 0xFF99;
		setquest 12375;// Manutenção da Base
		// if (terra_gloria_main == 20)
		// 	warp "1@swat",35,65;
		// else
		// 	warp "1@swat",73,118;
		end;
	}
}
