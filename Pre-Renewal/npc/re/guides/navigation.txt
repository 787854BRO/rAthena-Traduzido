function	script	F_Navi	{

	if ( PACKETVER < 20111010 )
		set .@disabled,1;

	// Esta função tem de 0 ~ 3 parâmetros.
	switch(getargcount()) {
	case 0:	// Verifica se o sistema está desabilitado.
		return .@disabled;
	case 1:	// Exibi mensagem informativa, se ativado.
		if (!.@disabled) {
			next;
			mes getarg(0);
			mes "Quando você clicar na ^B9062F[Localização]^000000 desejada, receberá o mais avançado serviço de ^B9062FNavegação^000000!";
		}
		return;
	default: // Exibi o link de navegação, se ativado; caso contrário, define a cor do texto, se disponível.
		if (!.@disabled)
			return "<NAVI>[" + getarg(0) + "]<INFO>" + getarg(1) + ",0,000,0</INFO></NAVI>";
		else if (getargcount() == 3)
			return getarg(2)+getarg(0)+"^000000";
		else
			return getarg(0);
	}
}
