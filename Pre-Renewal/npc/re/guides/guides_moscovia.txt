		next;

			break;
		case 2:
			mes "[Guia de Moscóvia]";
			mes "No "+F_Navi("Palácio","moscovia,255,140","^006400")+" você pode encontrar nosso Senhor, o Czar e outros servos.";

			viewpoint 1,255,140,1,0x0A82FF;

			mes "[Guia de Moscóvia]";
			mes "A "+F_Navi("Capela","moscovia,253,179","^008080")+" está localizada aqui.";
			mes .@str$;
			viewpoint 1,253,179,2,0xAAFF00;
			break;

			mes "[Guia de Moscóvia]";
			mes "Você pode comprar armas dos melhores artesãos em Moscóvia se você for a "+F_Navi("Loja de Armas","moscovia,186,191","^FF1493")+".";

			viewpoint 1,186,191,3,0x008080;
			break;

			mes "[Guia de Moscóvia]";
			mes "Você encontrará apenas os melhores equipamentos de Moscóvia na";
			mes ""+F_Navi("Loja de Armaduras","moscovia,204,169","^8B4513")+".";
			mes .@str$;
			viewpoint 1,204,169,4,0xB200FF;
			break;
		case 6:
			mes "[Guia de Moscóvia]";
			mes "A "+F_Navi("Loja de Utilidades","moscovia,221,176","^FF1493")+" está localizado ao sul do centro da cidade.";

			mes "[Guia de Moscóvia]";
			mes "A "+F_Navi("'Erva de Árvore Pegajosa'","moscovia,227,206")+", a Estalagem, está ao norte do centro da cidade.";
			mes "Se você precisa descansar, não há lugar melhor para ficar.";
			mes .@str$;
			viewpoint 1,227,206,6,0x8B4513;
			break;
		case 8:
			mes "[Guia de Moscóvia]";
			mes "Você pode encontrar o";
			mes F_Navi("R. Públicas de Moscóvia","moscovia,166,53")+" nas docas se";

			mes "Aproveite sua viagem~!";
			close;
		}
	}
