	next;

			if(!callfunc("F_CanOpenStorage")){
				mes "[Gerente de Logística]";
				mes "Você ainda não está qualificado para usar o Armazém.";

				Zeny -= 200;

			mes "[Gerente de Logística]";
			mes "Você precisa de pelo menos 200 zenys para usar o Armazém.";
		}
		break;
	case 2:
		mes "[Gerente de Logística]";
		mes "Deixe-me ver se você mudou de ideia.";
	}
	close;
}
