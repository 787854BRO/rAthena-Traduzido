prontera,80,106,5	script	Remover Buffs de Roupas	4_EP16_COOK2,{
	mes "[Atirador Derek]";
	mes "Você não consegue participar de batalhas por causa de buffs de roupas? Deve ser irritante.";
	next;
	mes "[Atirador Derek]";
	mes "Embora fique bem em você, também pode ser muito irritante. Eu posso me livrar deles com um único gesto de mão!! Irei organizá-lo perfeitamente. Se estiver pronto, escolha sua opção!";
	next;
	switch( select( "Mantenha-o.", "Remover imediatamente." ) ){
		case 1:
			mes "[Atirador Derek]";
			mes "Volte a falar comigo se mudar de ideia.";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[Atirador Derek]";
			mes "A remoção foi concluída. Certifique-se de que você está podendo se mover livremente agora.";
			atcommand "@changedress";
			close;
	}
}
