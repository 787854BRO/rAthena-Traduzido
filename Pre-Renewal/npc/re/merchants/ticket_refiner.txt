//= 1.1 Do not refine above ticket level. [Euphy]

// NPC Principal :: safety_Ref_NPC
//============================================================
prontera,184,177,6	script	Mestre do Refino	851,{
	disable_items;
	setarray .@cert_weapon[0],
		6864,	// Certificado de Arma Nv.19
		6875,	// Certificado de Arma Nv.18
		6874,	// Certificado de Arma Nv.17
		6873,	// Certificado de Arma Nv.16
		6872,	// Certificado de Arma Nv.15
		6871,	// Certificado de Arma Nv.14
		6870,	// Certificado de Arma Nv.13
		6584,	// Certificado de Arma Nv.12
		6238,	// Certificado de Arma Nv.11
		6228,	// Certificado de Arma Nv.09
		6229,	// Certificado de Arma Nv.08
		6230,	// Certificado de Arma Nv.07
		6231,	// Certificado de Arma Nv.06
		6456;	// Certificado de Arma Nv.05
	setarray .@cert_armor[0],
		6865,	// Certificado de Armadura Nv.19
		6881,	// Certificado de Armadura Nv.18
		6880,	// Certificado de Armadura Nv.17
		6879,	// Certificado de Armadura Nv.16
		6878,	// Certificado de Armadura Nv.15
		6877,	// Certificado de Armadura Nv.14
		6876,	// Certificado de Armadura Nv.13
		6585,	// Certificado de Armadura Nv.12
		6239,	// Certificado de Armadura Nv.11
		6232,	// Certificado de Armadura Nv.09
		6233,	// Certificado de Armadura Nv.08
		6234,	// Certificado de Armadura Nv.07
		6235,	// Certificado de Armadura Nv.06
		6457;	// Certificado de Armadura Nv.05
	setarray .@cert_level[0],19,18,17,16,15,14,13,12,11,9,8,7,6,5;
	.@size_cert = getarraysize(.@cert_weapon);

	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@cert_weapon[.@i]) > 0 || countitem(.@cert_armor[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}

	if (.@check == 0) {
		mes "[Mestre do Refino]";
		mes "Olá!";
		mes "E aí, novidades?";
		mes "Sou especialista";
		mes "em refinar itens,";
		mes "mas eu não trabalho mais.";
		next;
		switch(select("Eu seguirei meu caminho.:Hmm... isso me deixa curioso.")) {
		case 1:
			mes "[Mestre do Refino]";
			mes "Tome cuidado, aventureiro.";
			close;
		case 2:
			mes "[Mestre do Refino]";
			mes "Na verdade, às vezes eu forneço serviços de refino para aventureiros com um ^006400Pergaminho de Refino^000000...";
			mes "Tchau tchau~!";
			close;
		}
	}
	emotion ET_SURPRISE;
	mes "[Mestre do Refino]";
	mes "Saudações!";
	mes "Eu posso refinar um item até o ^006400mesmo nível do seu Pergaminho^000000.";
	mes "Você não precisa se preocupar! Não há chance de quebrar seu item.";
	next;
	if(select("Eu voltarei mais tarde.:Refinar item com Pergaminho.") == 1) {
		mes "[Mestre do Refino]";
		mes "Ok.";
		mes "Você pode vir depois.";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Qual equipamento você gostaria de refinar?";
	next;
	setarray .@position$[1],"Cabeça Topo","Vestimenta","Mão esquerda","Mão direita","Capa","Sapatos","Acessório 1","Acessório 2","Cabeça Meio","Cabeça Baixo";
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ += (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : .@position$[.@i] + "- [Empty]") + ":";

	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Mestre do Refino]";
		mes "Você tem que equipar o item que você quer refinar.";
		close;
	}
	if (!getequipisenableref(.@part)) {
		emotion ET_OTL;
		mes "[Mestre do Refino]";
		mes "Oh, me desculpe.";
		mes "É impossível refinar este item.";
		close;
	}
	.@refineitemid = getequipid(.@part); //salva o id do item
	.@refinerycnt = getequiprefinerycnt(.@part); //salva a contagem de refinamento
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	if( .@itemtype == IT_WEAPON ){
		switch( getequipweaponlv( .@part ) ){
			case 1:
			case 2:
			case 3:
			case 4:
				copyarray .@tickets[0], .@cert_weapon[0], .@size_cert;
				.@type$ = "Arma";
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_ARMOR ){
		switch( getequiparmorlv( .@part ) ){
			case 1:
				copyarray .@tickets[0], .@cert_armor[0], .@size_cert;
				.@type$ = "Armadura";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	.@check = 0;
	for ( .@i = 0; .@i < .@size_cert; ++.@i ) {
		if (countitem(.@tickets[.@i]) > 0) {
			.@check = 1;
			break;
		}
	}
	if (.@check == 0) {
		emotion ET_THINK;
		mes "[Mestre do Refino]";
		mes "Se você quiser refinar este ^006400"+.@type$+"^000000, por favor, entre com ^006400"+.@type$+" Pergaminho de Refino^000000.";
		mes "Até logo!";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Por favor, escolha qual ^006400"+.@type$+" Pergaminho de Refino^000000 você quer usar.";
	next;
	.@menu$ = "";
	for ( .@i = 0; .@i < .@size_cert; ++.@i )
		.@menu$ += getitemname(.@tickets[.@i]) + ":";
	.@select = select(.@menu$)-1;
	.@ticket_lv = .@cert_level[.@select];
	.@ticket_id = .@tickets[.@select];

	if (countitem(.@ticket_id) == 0) {
		emotion ET_QUESTION;
		mes "[Mestre do Refino]";
		mes getitemname(.@ticket_id)+" não está no seu inventário. Você colocou no seu armazém?";
		mes "Por favor, verifique novamente.";
		mes "Até logo!";
		close;
	}
	if (getequiprefinerycnt(.@part) >= .@ticket_lv) {
		emotion ET_PROFUSELY_SWEAT;
		mes "[Mestre do Refino]";
		mes "^8B4513Este item já foi refinado tanto quanto possível.^000000";
		mes "Por favor, volte com um item menos refinado que seu Pergaminho.";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Eu vou refinar ^006400"+getequipname(.@part)+"^8B4513 até o nível +"+.@ticket_lv+" ^000000 com ^006400"+getitemname(.@ticket_id)+"^000000.";
	mes "Posso prosseguir?";
	next;
	if(select("Não.:Sim.") == 1) {
		emotion ET_THINK;
		mes "[Mestre do Refino]";
		mes "Oh, você mudou de idéia.";
		mes "Ok.";
		mes "Você pode voltar mais tarde.";
		close;
	}
	mes "[Mestre do Refino]";
	mes "Ótimo.";
	mes "Como desejar!";
	mes "Eu tenho meu próprio jeito especial de refinar...";
	mes ".......ka boom!";
	specialeffect EF_SUI_EXPLOSION;
	if (countitem(.@ticket_id) < 1) {
		next;
		mes "Erro!";
		mes "Por favor, reporte isto.";
		close;
	}
	delitem .@ticket_id,1;

	// anti-roubo
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		mes "[Refine Master]";
		emotion ET_FRET;
		mes "Espere um segundo...";
		mes "Você acha que eu sou estúpido?!";
		mes "Você trocou o item enquanto eu não estava olhando! Saia daqui!";
		close;
	}
	successrefitem .@part, .@ticket_lv - getequiprefinerycnt(.@part);
	next;
	emotion ET_DELIGHT;
	mes "[Mestre do Refino]";
	mes "Tudo bem, aqui está~";
	mes "Bem, ^0000FF"+strcharinfo(0)+"^000000!";
	mes "Parabéns pelo seu brilho "+.@type$+".";
	mes "Você fica ÓTIMO!";
	mes "Adeus~!";
	close;
}
