//= arg(1): How many of each 'getarg(0)' arrow per quiver.
//= arg(2): The cost of making a 'getarg(0)' quiver.
//= arg(3): The quiver given by the NPC (item ID).

// Mora :: ep14_1_tre
//============================================================
mora,106,117,3	script	Fabricante de Aljaves#mora	4_M_RAFLE_GR,{
	if (checkweight(1201,1) == 0) {
		mes "[Fabricante de Aljaves]";
		mes "Você está carregando muitas coisas com você.";
		mes "Abra algum espaço no seu inventário e volte. Vou lhe contar uma coisa interessante.";
		close;
	}
	if (MaxWeight - Weight < 2000) {
		mes "[Fabricante de Aljaves]";
		mes "Você parece esgotado carregando todas essas coisas.";
		mes "Abra algum espaço no seu inventário e volte. Vou lhe contar uma coisa interessante.";
		close;
	}
	mes "[Fabricante de Aljaves]";
	mes "Os aldeões da vila de Mora perguntam o que são aljaves. Eles simplesmente não sabem como o mundo funciona.";
	mes "Não é de se admirar que eles não saibam nada sobre aljaves - Ficam esparramados sobre folhas o dia todo.";
	next;
	if (getbattleflag("feature.barter")) {
		.@menu$ = "Eu quero flechas.";
	}
	switch(select("Por favor, faça-me um Aljave.", .@menu$, "O Que é um Aljave?")) {
	case 1:
		if (getbattleflag("feature.barter")) {
		mes "[Fabricante de Aljaves]";
			mes "Finalmente, apareceu alguém que sabe o verdadeiro valor de uma aljava. Eu farei isso por aqueles que sabem o seu valor.";
			close2;
			callshop "Quiver_Archer", 1;
			end;
		}
		mes "[Fabricante de Aljaves]";
		mes "Finalmente alguém aprecia um Aljave!";
		mes "Eu posso fazer Aljave de Elfo e Aljave de Caça.";
		mes "De qual você precisa?";
		next;
		switch(select("Um Aljave de Elfo:Um Aljave de Caça:Não preciso de um Aljave.")) {
		case 1: callsub S_BuyQuiver,1773,500,500,12575; //Flecha_de_Elfo_Cntr
		case 2: callsub S_BuyQuiver,1774,500,500,12576; //Flecha_de_Caça_Cntr
		case 3:
			mes "[Fabricante de Aljaves]";
			mes "Você pode comprar flechas ao lado do Vendedor de Utilidades.";
			close;
		}
		end;
	case 2:
		if (getbattleflag("feature.barter")) {
			mes "[Fabricante de Aljaves]";
			mes "Uma flecha pode ser fina e leve, mas carregar centenas ou milhares de flechas é como carregar uma tora inteira.";
			close2;
			callshop "Catch_Arrows", 1;
		}
		end;
	case 3:
		mes "[Fabricante de Aljaves]";
		mes "Uma flecha pode ser fina e leve, mas carregar centenas e milhares de flechas é como carregar um tronco inteiro.";
		next;
		mes "[Fabricante de Aljaves]";
		mes "Mas quando você tiver usando Aljaves, você pode colocar flechas neles e economizar espaço e peso.";
		mes "Se você estiver interessado em um, costuro um para você.";
		close;
	}
	end;

S_BuyQuiver:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);
	if (countitem(getarg(0)) < .@req_amount) {
		mes "[Fabricante de Aljaves]";
		mes "Traga mais do que "+ .@req_amount +" "+getitemname(.@item_req) +" e eu farei um aljave para você.";
		close;
	}
	mes "[Fabricante de Aljaves]";
	mes "Oh, eu vejo que você tem "+getitemname(.@item_req) +" na sua mão!";
	mes "Você está interessado em usar um aljave? É realmente muito conveniente!";
	mes "Se você estiver interessado, eu posso trocar "+ .@req_amount +" daquelas flechas por um desses aljaves por ^FF3131"+getarg(2)+" zeny(s)^000000.";
	next;
	switch(select("Negociar todas as flechas.:Consiga apenas um aljave.:Não negociar!")) {
	case 1:
		set .@arrows, countitem(.@item_req);
		set .@quiver, .@arrows / .@req_amount;
		set .@arrows_used, .@quiver * .@req_amount;
		set .@arrow_zeny01, .@quiver * getarg(2);
		mes "O número de flechas que você tem : ^3131FF"+.@arrows+"^000000";
		mes "O número de aljaves disponíveis : ^3131FF"+.@quiver+"^000000";
		mes "Zeny necessário para a troca : ^3131FF"+.@arrow_zeny01+" zeny(s)^000000";
		next;
		mes "Negociar?";
		next;
		if(select("Negociar:Não Negociar") == 2) {
			mes "[Fabricante de Aljaves]";
			mes "Ei, você não duvida das minhas habilidades, não é?";
			close;
		}
		break;
	case 2:
		set .@quiver, 1;
		set .@arrows_used, .@req_amount;
		set .@arrow_zeny01, getarg(2);
		set .@zeny_mes,1;
		break;
	case 3:
		mes "[Fabricante de Aljaves]";
		mes "Ei, você não duvida das minhas habilidades, não é?";
		close;
	}
	if (Zeny < .@arrow_zeny01) {
		mes "[Fabricante de Aljaves]";
		if (.@zeny_mes == 1)
			mes "Eu disse que aceitaria moedas humanas só para você, e você ainda não quer gastar o dinheiro?";
		else
			mes "Você realmente não espera colocar suas mãos em uma obra prima por nada, não é?";
		close;
	}
	mes "[Fabricante de Aljaves]";
	mes "Ei, aqui está você.";
	mes "Há ^3131FFalgo que você precisa saber^000000 - Tente sempre se lembrar.";
	mes "^FF0000Você não pode usar aljaves quando o seu peso estiver acima de 70%.^000000";
	mes "É melhor você manter isso em mente, ou você poderá entrar em apuros mais tarde.";
	set Zeny, Zeny-.@arrow_zeny01;
	delitem .@item_req,.@arrows_used;
	getitem getarg(3),.@quiver;
	close;
}
