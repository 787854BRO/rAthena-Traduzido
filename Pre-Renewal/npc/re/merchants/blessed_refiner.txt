//===== rAthena Script ======================================= 
//= Blessed Refiner
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use Blessed ores to refine equipment.
//= Upon failure, the equipment is not destroyed. The success
//= rate is identical to that for Enriched ores.
//= - "Blacksmith Dister" only refines from +6~12.
//= NOTE: This NPC is currently disabled on official servers.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// NPC Principal :: new_smelting612
//============================================================
-	script	::FerreiroDister	-1,{
	disable_items;
	mes "[Ferreiro Dister]";
	mes "Nesta sociedade altamente competitiva, nós devemos ser diferentes para sobreviver!";
	mes "Eu refino equipamentos em nível +6 ~ +12.";
	next;
	mes "[Ferreiro Dister]";
	mes "Eu precisarei de ^ff9999Omni-Oridecon^000000 para armas, ou de ^ff9999Mega-Elunium^000000 para armaduras.";
	mes "Refinamentos fracassados ^FF0000não quebrarão ou reduzirão o nível de refino^000000.";
	next;
	mes "[Ferreiro Dister]";
	mes "Que tal? Você quer refinar?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Não equipado]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Ferreiro Dister]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Sou um ferreiro, não um cabeleireiro.";
			break;
		case EQI_ARMOR:
			mes "Com meu martelo, Farei você uma estrela do céu.";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fazer mãos artificiais não é minha especialidade.";
			break;
		case EQI_GARMENT:
			mes "Traga o item para que eu possa refiná-lo!";
			break;
		case EQI_SHOES:
			mes "De onde está vindo esse cheiro estranho?";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Onde está o acessório?";
			break;
		case EQI_HEAD_MID:
			mes "O que quer que eu refine?";
			break;
		case EQI_HEAD_LOW:
			mes "Huh? O que quer que eu faça?";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Ferreiro Dister]";
		mes "Este item não pode ser refinado.";
		close;
	}
	.@equip_id = getequipid(.@part);
	.@itemtype = getiteminfo( .@equip_id, ITEMINFO_TYPE );
	.@equip_refine = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if (.@equip_refine < 6 || .@equip_refine > 12) {
		mes "[Ferreiro Dister]";
		mes "Este equipamento está refinado no nível "+.@equip_refine+". Eu só dou conta de itens com níveis de refino de +6 a +12!";
		close;
	}
	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "Armadura";

		switch(.@equip_lv) {
			case 1:
				set .@price,20000;
				set .@material,6439; //Mega-Elunium
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "Arma";

		switch( .@equip_lv ){
			case 1:
				set .@price,1000;
				set .@material,6438; //Omni-Oridecon
				break;
			case 2:
				set .@price,2000;
				set .@material,6438; //Omni-Oridecon
				break;
			case 3:
				set .@price,20000;
				set .@material,6438; //Omni-Oridecon
				break;
			case 4:
				set .@price,40000;
				set .@material,6438; //Omni-Oridecon
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	set .@ore$,"^ff9999Minério Abençoado "+.@type$+"^000000";
	mes "[Ferreiro Dister]";
	mes "Esta "+.@type$+" está refinada em "+.@equip_refine+". Para refiná-la, precisarei de "+.@ore$+" e "+callfunc("F_InsertComma",.@price)+" zenys como taxa de refinamento.";
	mes "Quer continuar?";
	next;
	if(select("Sim:Não") == 2) {
		mes "[Ferreiro Dister]";
		mes "Estou ocupado, não me incomode!";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[Ferreiro Dister]";
		mes "Esta "+.@type$+" foi refinada muitas vezes. Embora ela não vá desaparecer no caso de falha, o "+.@ore$+" vai!";
		next;
		mes "[Ferreiro Dister]";
		mes "É claro, a taxa de refinamento não será devolvida! Quer continuar?";
		next;
		if(select("Sim:Não") == 2) {
			mes "[Ferreiro Dister]";
			mes "Bom. Eu também ficaria triste se falhasse.";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Ferreiro Dister]";
		mes "Material insuficiente.";
		mes "Para refinar esta "+.@type$+" até o nível "+.@equip_lv+", precisarei de "+.@ore$+" e de "+callfunc("F_InsertComma",.@price)+" zenys como taxa de refinamento.";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-roubo
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
		close;

	mes "[Ferreiro Dister]";
	mes "Tac! Tac! Tac!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		specialeffect EF_BLESSING;
		successrefitem .@part;
		next;
		mes "[Ferreiro Dister]";
		mes "Hahahahahaha!!!";
		next;
		mes "[Ferreiro Dister]";
		mes "O refino foi um grande sucesso~ Hahaha!";
		close;
	}
	specialeffect EF_CURSEATTACK;
	specialeffect2 EF_SUI_EXPLOSION;
	next;
	emotion (!rand(5))?ET_ANGER:ET_HUK;
	mes "[Ferreiro Dister]";
	mes "Oh Meu Deuss!!!!";
	next;
	mes "[Ferreiro Dister]";
	mes "Oh! Embora tenha sido abençoado pelo "+.@ore$+" para preservar o equipamento...";
	next;
	mes "[Ferreiro Dister]";
	mes "Meu humor piora quando eu falho!!";
	close;
}
