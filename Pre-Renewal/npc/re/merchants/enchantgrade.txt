grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[Sratos]";
	mes "Olá querido cliente que acaba de entrar, vamos torcer por um milagre hoje.";
	mes "Como posso ajudá-lo?";
	next;
	switch( select( "Melhorar Equipamento.", "Obter Joias Etéricas", "Encantamento de Grade?" ) ){
		case 1:
			mes "[Sratos]";
			mes "Nosso cliente. Você quer melhorar a qualidade do seu equipamento.";
			mes "Não é fácil manipular o poder mágico das joias.";
			next;
			mes "[Sratos]";
			mes "O poder mágico pode explodir no local.";
			mes "Então a arma que herdará o poder mágico também explodirá!";
			next;
			switch( select( "Ainda quero fazer!", "Vou pensar sobre isso." ) ){
				case 1:
					mes "[Sratos]";
					mes "Desejo boa sorte aos nossos corajosos clientes!";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[Sratos]";
					mes "Quando você tiver coragem de desafiar, volte.";
					close;
			}
		case 2:
			mes "[Sratos]";
			mes "Estas são as joias usadas para o Encantamento de Grade.";
			mes "Pó de Éter, o ingrediente que será a base, e se você me der uma pequena taxa eu trocarei por Joias Étéricas.";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[Sratos]";
			mes "Ocasionalmente, são encontradas essas joias instáveis com muito poder mágico e puro.";
			mes "Chamamos elas de joias Étéricas, certo?";
			next;
			mes "[Sratos]";
			mes "Você pode transferir o poder mágico da joia etérica para armas, mas é claro que existem riscos.";
			mes "O processo é delicado, então se você não tomar cuidado, pou! Vai explodir.";
			next;
			mes "[Sratos]";
			mes "De qualquer forma, se você usar esta joia etérica, poderá aprimorar a arma que está utilizando.";
			mes "O desempenho melhora. Nós chamamos isso de Encantamento de Grade.";
			next;
			mes "[Sratos]";
			mes "Você também não ficaria feliz se as armas que você ama crescessem um passo adiante?";
			mes "Nós somos as pessoas que o ajudaram com isso. Mas é um pouco arriscado. ahahaha.";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Encantamento de Grade>");
	end;
}
