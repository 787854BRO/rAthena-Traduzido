		}
		.@amount_coag = .@en_amount_coag[.@num];
		.@amount_poll = .@en_amount_polluted[.@num];

		mes "[Mestre da Magia Negra]";
		if (.@num == 4) {
			.@card[2] = callfunc("F_Rand",4875,4876,4877,4878,4879,4880);// Poder do Urso, Magia Incontrolável, Velocidade da Luz, Tolo Musculoso, Olho de Águia, Sorte Grande
			.@string$ = "^990000Bônus de Efeito^000000 melhorado.";
			mes "Gostaria de receber um bônus de efeito aleatório para o 3º slot? Você precisará de ^0000ff" + .@amount_poll + " " + getitemname(.@polluted_id) + ", " + .@amount_coag + " " + getitemname(.@coagulated_id) + ", e " + callfunc("F_InsertComma",.@cost_zeny) + " zenys^000000. Mas há um grande risco de falha...";
		}
		else {
			.@level = .@num + 1;
			.@card[3] = getd( ".@enchant_" + (.@num+1) + "[" + .@type + "]" );
			.@string$ = "Encantamento número ^990000" + .@level + "^000000.";
			mes "Encantando o 4º slot da ^0000ff" + .@equip_name$ + " (Com Slot)^000000 e com efeito de nível ^0000ff" + .@level + "^000000. Requer ^0000ff" + .@amount_poll + " " + getitemname(.@polluted_id) + ", " + .@amount_coag + " " + getitemname(.@coagulated_id) + ", e " + callfunc("F_InsertComma",.@cost_zeny) + " z^000000. Existe um risco de falha...";

			mes "Ok, volte quando estiver pronto.";

		close;
	}
	delitem .@coagulated_id, .@amount_coag;
	delitem .@polluted_id, .@amount_poll;
	Zeny -= .@cost_zeny;

	if (callfunc("F_IsEquipIDHack", EQI_SHOES, .@equip_id) || callfunc("F_IsEquipCardHack", EQI_SHOES, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]) || callfunc("F_IsEquipRefineHack", EQI_SHOES, .@equip_refine))

		mes "[Mestre da Magia Negra]";
		mes "Arghhh, falhamos. Melhor sorte na próxima vez.";
		close;
	}
	specialeffect2 EF_REPAIRWEAPON;
	mes "[Mago do Hugin]";
	mes "Tentando o " + .@string$;
	getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,.@card[2],.@card[3];
	close;
