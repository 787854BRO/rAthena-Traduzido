aldeba_in,96,181,4	script	Serviços Kafra	113,{
	cutin "kafra_05",2;
	mes "[Kafra Leila]";
	mes "Hm...?";
	mes "Oh, bem-vind"+((Sex)?"o":"a")+" ao";
	mes "Quartel General";
	mes "da Corporação Kafra";
	mes "precisa de algo?";	
	next;
	switch(select("Salvar.:Usar Armazém.:Alugar Carrinho.:Cancelar")) {
	case 1:
		mes "[Kafra Leila]";
		mes "Seu ponto de retorno";
		mes "foi salvo aqui, dentro";
		mes "do Quartel General";
		mes "da Corporação Kafra, Obrigado.";
		next;
		savepoint "aldeba_in",96,179,1,1;
		mes "[Kafra Leila]";
		mes "Por favor, faça uso de";
		mes "nossos Serviços Kafra que estão";
		mes "disponíveis em toda a";
		mes "Midgard. Obrigado por";
		mes "visitar a sede da Corp. Kafra.";
		close2;
		break;
	case 2:
		if (Zeny < 20) {
			mes "[Kafra Leila]";
			mes "Com licença, mas me";
			mes "parece que você não";
			mes "tem os 20 zenys para pagar";
			mes "a taxa de acesso ao armazém...";
			close;
		}
		mes "[Kafra Leila]";
		mes "Embora esta instalação seja";
		mes "destinada exclusivamente para";
		mes "o treinamento de Funcionárias Kafra";
		mes "e suas funções administrativas,";
		mes "irei acessar seu armazém para você.";
		next;
		set Zeny, Zeny-20;
		set RESRVPTS, RESRVPTS + (20/5);
		mes "[Kafra Leila]";
		mes "No futuro por favor,";
		mes "pergunte à Funcionária Kafra";
		mes "se você deseja usar";
		mes "qualquer um dos Serviços da Corp. Kafra.";
		mes "Obrigado pelo seu patrocínio.";
		callfunc("F_CheckKafCode");    //Verifica se o armazém possui senha.
		close2;
		openstorage;
		break;
	case 3:
		mes "[Kafra Leila]";
		mes "Minhas desculpas, mas";
		mes "não estou em serviço. Eu te ajudaria";
		mes "se eu pudesse, mas na verdade não";
		mes "tenho carrinhos disponíveis.";
		mes "Por que você não procura outra Funcionária Kafra";
		mes "para obter assistência?";
		next;
		mes "[Kafra Leila]";
		mes "As Funcionárias Kafra estão";
		mes "trabalhando em todo o";
		mes "Continente Midgard,";
		mes "e você deverá ser capaz de encontrá-las";
		mes "até fora de Al De Baran.";
		close2;
		break;
	case 4:
		mes "[Kafra Leila]";
		mes "^666666*Nossa...*^000000";
		mes "Ótimo, porque na verdade";
		mes "estou no meu intervalo";
		mes "agora mesmo. Escolhendo";
		mes "''Cancelar'' será muito";
		mes "bom de sua parte.";
		close2;
		break;
	}
	cutin "",255;
	end;
}
