//=== Função Principal =====================================================
//=   arg(0): Usado para determinar qual mensagem de boas-vindas será mostrada.
//=   arg(1): Usado para determinar qual menu será exibido.
//=   arg(2): Usado para determinar se o menu de informações é mostrado em F_KafInfo.
//=   arg(3): Custo para usar o Armazém.
//=   arg(4): Custo para alugar um Carrinho.
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Limpa as variáveis desatualizadas e não utilizadas.
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Exibe mensagem de boas-vindas da Kafra.
	switch(.@welcome) {
	default:// Mensagem padrão (obsoleta)
		mes "[Funcionária Kafra]";
		mes "Bem-vind"+(Sex?"o":"a")+" à Corporação Kafra.";
		mes "Os serviços da corporação estão";
		mes "aqui para dar suporte à você.";
		mes "Como posso ajudá-lo hoje?";
		break;
	case 1:// Mensagem específica de Niflheim
		mes "[Funcionária Kafra]";
		mes "^666666B-beeeem-v-iiindo";
		mes "à C-corpoo-r-raçããoooo";
		mes "K-kaaafrrrra...^000000";
		break;
	case 2:// Mensagem da Kafra no Castelo do Clã 
		mes "[Funcionária Kafra]";
		mes "Bem-vind"+(Sex?"o":"a")+" membro do clã ^ff0000" + getguildname( getcharid(2) ) + "^000000.";
		mes "A Corporação Kafra, está aqui para melhor lhe atender.";
		break;
	case 3:// Mensagem específica do Amatsu (obsoleta)
		mes "[Funcionária Kafra]";
		mes "Então, você veio de uma terra distante para estudar nossa cultura, ou você está só de turista?";
		mes "Em ambos os casos, por que não ficar por um tempo?";
		mes "O ar é eternamente carregado";
		mes "com o agradável aroma de flores selvagens.";
		break;
	case 4:// Mensagem específica de Luoyang e Ayothaya (obsoleta)
		mes "[Funcionária Kafra]";
		mes "Com nossas muitas";
		mes "localidades de Serviços Kafra, você nunca estará";
		mes "longe de casa.";
		break;
	case 5:// O NPC tem sua própria mensagem de boas vindas. (Exibe nada)
		break;
	}
	next;
	if (.@welcome == 2)// Substituição do menu Clã da Kafra (teletransporte gratuito, armazenamento gratuito, carrinho gratuito)
		setarray .@K_Menu0$[0], "Usar Armazém", "Usar Armazém do Clã", "Alugar Carrinho", "Usar Serviço de Teleporte", "Cancelar";
	else {
		switch(.@menu_num) {
			// Opções de Salvar e Usar Armazém.
			case 1:	setarray .@K_Menu0$[0],"Salvar","Usar Armazém","Cancelar"; break;
			// Apenas a opção de usar Armazém.
			case 2:	setarray .@K_Menu0$[0],"Usar Armazém","Cancelar"; break;
			// Sem a opção Teleporte (Comum)
			case 3: setarray .@K_Menu0$[0],"Salvar","Usar Armazém","Alugar Carrinho","Checar Outras Informações","Cancelar"; break;
			// Case 4 em Einbroch não tem telemensagem.
			// Sem as opções Salvar, ou Teleporte. (Comum)
			case 5:	setarray .@K_Menu0$[0],"Usar Armazém","Alugar Carrinho","Checar Outras Informações","Cancelar"; break;
			// Apenas Armazém e Chegar Outras Informações.
			case 6:	setarray .@K_Menu0$[0],"Usar Armazém","Checar Outras Informações","Cancelar"; break;
			// Salvar, Armazém, e Alugar Carrinho (Armazém da Kafra)
			case 7:	setarray .@K_Menu0$[0],"Salvar","Usar Armazém","Alugar Carrinho","Cancelar"; break;
			// Salvar, Armazém, Chegar Outras Informações. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"Salvar","Usar Armazém","Checar Outras Informações","Cancelar"; break;
			// Não salvar (Cavaleiro Rúnico)
			case 9: setarray .@K_Menu0$[0],"Usar Armazém","Alugar Carrinho","Usar Serviço de Teleporte","Checar Outras Informações","Cancelar"; break;
			// Armazém, Salvar, e Alugar Carrinho (Dewata, reordenação do case 7)
			case 10: setarray .@K_Menu0$[0],"Usar Armazém","Salvar","Alugar Carrinho","Cancelar"; break;
			// Mensagens padrão (Obsoleta)
			default: setarray .@K_Menu0$[0],"Salvar","Usar Armazém","Usar Serviço de Teleporte","Alugar Carrinho","Checar Outras Informações","Cancelar"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "Salvar")
			return;
		else if (.@K_Menu0$[.@j] == "Usar Armazém") {
			if (.@welcome == 2) // Sem Cobrança para usar o Armazém do Clã
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "Usar Serviço de Teleporte") {
			if (.@menu_num != 4)// Exibir em Einbroch "Sem aviso de serviço de teleporte".
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[Funcionária Kafra]";
				mes "Devido ao ^FF0000Limitado";
				mes "Acordo de Transporte^000000, a";
				mes "Corporação Kafra não pode";
				mes "providenciar os Serviços de";
				mes "Teletransporte na República Schwaltzvalt.";
				next;
				mes "[Funcionária Kafra]";
				mes "Nós pedimos que por favor,";
				mes "use o serviço aéreo.";
				mes "Obrigado por sua compreenção";
				mes "e cooperação.";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "Alugar Carrinho") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "Checar Outras Informações") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "Cancelar") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "Usar Armazém do Clã") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Função do Armazém
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Desabilitado para acesso ao Armazém do Clã (Ocupado).
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[Funcionária Kafra]";
			mes "Sinto muito, mas outro membro do Clã está usando o armazém neste instante.";
			mes "Por favor, aguarde até que esta pessoa termine.";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Desabilitado para acesso ao Armazém Normal (Nível de Habilidades Básicas Insuficiente).
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Funcionária Kafra]";
		// Mensagem Específica de Niflheim
		if (getarg(2) == 1) {
			mes "^666666D-d-esscullpeeee,";
			mes "v-vocêê éé u-u-ummmm";
			mes "Aaaprendizzz... preecisaa";
			mes "ter habilidadessss báásicass ní-nível 6...^000000";
			return;
		}
		mes "Me desculpe, mas você";
		mes "precisa das Habilidades";
		mes "Básicas no Nível 6 para";
		mes "poder usar o armazém.";
		return;
	}
	// Acessando o armazenamento normal (ignorado se estiver acessando o armazenamento do castelo do Clã)
	if (.@type != 2) {
		// Consome o "Vale-Armazém para armazenamento Kafra" se disponível.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[Funcionária Kafra]";
				// Mensagem Específica de Niflheim
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Zeeeeeny...";
					mes "P-Preeciso de Z-Zeny...!";
					mes "P-Preciiiso de 150... Zenys...";
					mes "Ergh!!! T-tomando s-angue!";
					return;
				}
				// Mensagem Padrão.
				mes "Lamento, mas você não possui";
				mes "Zenys suficientes para usar";
				mes "o serviço de armazém.";
				mes "O custo de acesso é de " + .@fee + " zenys.";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Suprime a mensagem "Fim" se estiver no castelo do clã.
	if (.@type != 2) {
		// Mensagem específica de Niflheim
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Funcionária Kafra]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666Obrigado.. por... usar...^000000";
		}
		// Mensagem Normal
		else {
			mes "[Funcionária Kafra]";
			mes "Pronto, deixe-me abrir";
			mes "seu armazém para você.";
			mes "Obrigado por usar";
			mes "os Serviços Kafra!";
		}
	}
	callfunc("F_CheckKafCode");	// Verifica sua senha do Armazém, se definida.
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Função de Teletransporte
//===================================================
function	script	F_KafTele	{
	mes "[Funcionária Kafra]";
	mes "Por favor,";
	mes "escolha seu destino.";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "Cancelar")
		return;
	// Consome o "Vale-Teletransporte para transporte Kafra" se disponível.
	// Não consome se estiver no castelo do Clã
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[Funcionária Kafra]";
			mes "Me desculpe, mas você não tem";
			mes "zenys suficientes para o teletransporte.";
			mes "O valor do teletransporte";
			mes "para " + @wrpD$[.@j] + " é de " + @wrpP[.@j] + " zenys.";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Ilha do Farol, Pharos") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "Geffen") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "Izlude") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "Juno") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "Mina Abandonada de Mjolnir") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "Morroc") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "Caverna dos Orcs") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "Payon") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "Prontera") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "Umbala") warp "umbala", 100, 154;
	end;
}

//===================================================
// Função do Carrinho
//===================================================
function	script	F_KafCart	{
	// Certifica-se de que a classe que deseja alugar um carrinho de mão seja um comerciante.
	if (BaseClass != Job_Merchant) {
		mes "[Funcionária Kafra]";
		mes "Me desculpe, mas o";
		mes "Serviço de Aluguel de Carrinho";
		if (checkre(0) == true)
			mes "está disponível somente para Mercadores,";
		else {
			mes "está disponível somente para Ferreiros, Mestre-Ferreiros,";
			mes "Alquimistas e Criadores.";
		}
		return 1;
	}
	// Certifica-se de que o personagem invocador ainda não tenha um carrinho
	else if (checkcart() == true) {
		mes "[Funcionária Kafra]";
		mes "Você já tem um";
		mes "carrinho equipado.";
		mes "Infelizmente, não podemos";
		mes "alugar mais de um para";
		mes "cada cliente.";
		return 1;
	}
	// Verifica se o jogador tem a habilidade Usar Carrinho
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[Funcionária Kafra]";
		mes "Você só pode alugar um carrinho depois de aprender a habilidade \"Usar Carrinho\".";
		return 1;
	}
	// Consome o "Vale-Carrinho para o serviço de Carrinho" se disponível.
	// Não Consome se estiver no castelo do Clã.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[Funcionária Kafra]";
		mes "O aluguel do carrinho de mão";
		mes "custará " + .@rental_fee + " zenys.";
		mes "Você deseja alugar um carrinho de mão?";
		next;
		if (select("Alugar Carrinho.:Cancelar") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[Funcionária Kafra]";
			mes "Me desculpe, mas você";
			mes "não tem zenys suficientes";
			mes "para pagar o preço do aluguel";
			mes "de carrinho que é de " + .@rental_fee + " zenys.";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

// Função - Pontos de Reserva Especial
//===================================================
function	script	F_KafInfo	{
// Remova o comentário da próxima linha para remover a Proteção Kafra
//	.@block = 1;
	if (getarg(0) == 2)// Porto Malaya, Cavaleiro Rúnico
		setarray .@menu$[0], "Checar Reserva de Pontos Especiais.", "", "", "Cancelar";
	else if (.@block)
		setarray .@menu$[0], "Checar Reserva de Pontos Especiais.", "", "Localização das Funcionárias Kafra", "Cancelar";
	else
		setarray .@menu$[0], "Checar Reserva de Pontos Especiais.", "Serviço de Senha do Armazém", "Localização das Funcionárias Kafra", "Cancelar";
	.@menu_list$ = implode(.@menu$,":");
	while (true) {
		.@j = select(.@menu_list$) - 1;
		switch(.@j) {
		case 0:// Checa a reserva de pontos especiais.
			mes "[Funcionária Kafra]";
			mes "Vamos ver...";
			mes "" + strcharinfo(0) + "...";
			mes "Ah, você tem um total de";
			mes "" + RESRVPTS + " Ponto's de Reserva.";
			next;
			mes "[Funcionária Kafra]";
			mes "Você pode trocar seus";
			mes "Pontos de Reserva por";
			mes "prêmios no Escritório Oficial da Kafra em Al De Baran. Por favor, use nossos";
			mes "serviços convenientes para ver os benefícios do programa de prêmios.";
			if (getarg(0) == 1)
				return;
			next;
			break;
		case 1:// Serviço de Senha do Armazém
			callfunc("F_SetKafCode","[Funcionária Kafra]","Serviços Kafra");
		case 2:// Localização das Funcionárias Kafra
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 1,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			next;
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 2,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			break;
		default:
			return;
		}
	}
}

		mes "os Serviços Kafra.";
	} else if (.@type == 0 || .@type == 5) {// Fim Genérico
		mes "Nós, aqui da Corporação Kafra,";
		mes "estaremos sempre ao seu redor para prover um melhor serviço. Esperamos poder suprir as necessidades de um bravo aventureiro como você.";
	} else if (.@type == 1) {// Niflheim Fim
		percentheal 0,-25;
		mes "^666666Kaffffra nn-nuncaa";
		mes "mmooooorreeee. See...";
		mes "Sempre aoo seuuu laaadooo~^000000";
	} else if (.@type == 2) {// Salvar em Pedra Cume
		mes "Foi salvo.";
		mes "Obrigado pelo seu patrocínio.";
	}
	close3;
}

//===================================================
// Função para Verificar o Armazém
//===================================================
function	script	F_CheckKafCode	{
	if (!#Senha_do_Armazem_Kafra)
		return;
	mes "Digite a senha do armazém:";
	input .@code_;
	if (.@code_ != #Senha_do_Armazem_Kafra) {
		dispbottom "Senha errada.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Definir / Mudar / Limpar Senha do Armazém ===
// getarg(0): Nome do NPC
// getarg(1): Nome da Compania
//==================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#Senha_do_Armazem_Kafra == 0) {
		mes "" + .@comp_name$ + "orgulhosamente apresenta um novo serviço:";
		mes "Proteção adicional do seu armazém com senha.";
		next;
		if (select( "Definir nova senha -> 5000z", "Cancelar" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "Seu armazém está protegido com uma senha. O que você fará agora?";
	next;
	switch( select( "Mudar senha antiga -> 5000z", "Remover senha do Armazém -> 1000z", "Cancelar" ) ) {
	case 1:
		mes .@npc_name$;
		mes "Primeiro, digite sua ^0000FFsenha antiga^000000.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #Senha_do_Armazem_Kafra) {
			mes "Senha errada. Você não pode inserir uma nova senha.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "Por favor, entre com a senha antes de sua retirada.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "A senha não foi removida.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "Você não possui zeny suficiente.";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #Senha_do_Armazem_Kafra) {
			#Senha_do_Armazem_Kafra = 0;
			mes "Você teve êxito ao colocar a senha em seu armazém.";
			mes "Muito obrigado por usar os " + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "Senha errada! Não devolverei seus 1.000 zenys.";
			mes "Por favor, na próxima entre com a senha correta.";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "Agora entre com sua ^FF0000nova senha^000000 para proteger seu armazém de roubos.";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "A senha não foi trocada.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "Você não possui zeny suficiente.";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#Senha_do_Armazem_Kafra = .@code;
	mes "Seu armazém está protegido com uma senha secreta.";
	mes "Obrigado por usar os " + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Função Básica de Validação - Senha Função
//===================================================
function	script	F_EntKafCode	{
	mes "Digite um número entre 1000~10000000:";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tentou colocar senha de armazenamento.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "Você não pode usar uma senha tão grande.";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "Você não pode usar uma senha tão curta.";
		return 0;
	}
	return .@code_;
}
