	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Mestre-Ferreiro]";
			mes "Parabéns!";
			mes "Honra aos guerreiros!";
			close;
		}
		mes "[Mestre-Ferreiro]";
		mes "Ah...";
		mes "O melhor em";
		mes "forja e refinamento.";
		mes "Esse é o trabalho";
		mes "do Mestre-Ferreiro.";
		next;
		mes "[Mestre-Ferreiro]";
		mes "Sendo um você será";
		mes "o melhor dos melhores,";
		mes "você começará a forjar com";
		mes "o objetivo de descobrir";
		mes "novas e melhores formas de";
		mes "forja...";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[Mestre-Ferreiro]";
		mes "Chegou a hora!";
		mes "Nosso mundo precisa de coragem,";
		mes "e de esforçados aventureiros";
		mes "como você...";
		next;
		mes "[Mestre-Ferreiro]";
		mes "Você gostaria de se tornar";
		mes "um Mestre-Ferreiro?";
		next;
		if (select("Não.:Sim.") == 1) {
			mes "[Mestre-Ferreiro]";
			mes "Quando estiver pronto,";
			mes "sinta-se a vontade para voltar.";
			mes "Honra aos guerreiros!";
			close;
		}
		if (SkillPoint) {
			mes "[Mestre-Ferreiro]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "todos os seus Pontos de Habilidade antes de voltar a falar comigo.";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[Mestre-Ferreiro]";
		mes "Parabéns!";
		mes "Sendo um Mestre-Ferreiro,";
		mes "eu acredito que você irá construir";
		mes "um futuro brilhante para";
		mes "a cidade de Rune-Midgard.";
		close;
	}
	mes "[Mestre-Ferreiro]";
	mes "Bem-vind"+((Sex) ? "o":"a")+"";
	mes "à Valhalla,";
	mes "o Salão de Honras.";
	next;
	mes "[Atirador de Elite]";
	mes "Por favor, sinta-se";
	mes "a vontade enquanto";
	mes "estiver por aqui.";
	mes "Honra aos guerreiros.";
	close;
}
